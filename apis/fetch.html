Fetch API - Conceptos Básicos

¿Qué es Fetch?
Fetch es la API moderna de JavaScript para hacer peticiones HTTP. Reemplaza a XMLHttpRequest y trabaja con Promises.
1Sintaxis Básica de Fetch
// ============================================
// FETCH API - SINTAXIS BÁSICA
// ============================================

// La forma más simple de fetch (GET por defecto)
fetch('https://api.example.com/data')
    .then(response => response.json())
    .then(data => console.log(data))
    .catch(error => console.error('Error:', error));


// ====== ANATOMÍA DE FETCH ======

// fetch() retorna una Promise que resuelve a un objeto Response
fetch(url, options)
    .then(response => {
        // El objeto Response contiene:
        console.log(response.ok);        // true si status 200-299
        console.log(response.status);    // código de estado (200, 404, etc)
        console.log(response.statusText); // texto del estado ("OK", "Not Found")
        console.log(response.headers);   // headers de la respuesta
        console.log(response.url);       // URL final (después de redirects)
        
        // Métodos para leer el body (solo puedes usar UNO):
        return response.json();   // Para JSON
        // return response.text();   // Para texto plano
        // return response.blob();   // Para archivos binarios
        // return response.formData(); // Para FormData
    });


// ====== OPTIONS DE FETCH ======

const options = {
    method: 'POST',  // GET, POST, PUT, PATCH, DELETE
    
    headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer tu_token_aqui',
        'Accept': 'application/json'
    },
    
    body: JSON.stringify({
        nombre: 'Juan',
        email: 'juan@email.com'
    }),
    
    // Otras opciones:
    mode: 'cors',              // cors, no-cors, same-origin
    credentials: 'include',    // include, same-origin, omit (para cookies)
    cache: 'no-cache',         // default, no-cache, reload, force-cache
    redirect: 'follow',        // follow, error, manual
};

fetch('https://api.example.com/users', options);

    );
    const user = await response.json();
    return user;
}

// Uso:
const user = await getUserById(1);


// ====== EJEMPLO 3: GET con Query Parameters ======
async function searchProducts(query, page = 1, limit = 10) {
    // Construir URL con URLSearchParams
    const params = new URLSearchParams({
        q: query,
        _page: page,
        _limit: limit,
        _sort: 'price',
        _order: 'asc'
    });
    
    const url = `https://api.example.com/products?${params}`;
    // Resultado: /products?q=laptop&_page=1&_limit=10&_sort=price&_order=asc
    
    const response = await fetch(url);
    return await response.json();
}

// Uso:
const results = await searchProducts('laptop', 1, 20);


// ====== EJEMPLO 4: GET con Headers de Autenticación ======
async function getProtectedData() {
    const token = localStorage.getItem('authToken');
    
    const response = await fetch('https://api.example.com/me', {
        method: 'GET',
        headers: {
            'Authorization': `Bearer ${token}`,
            'Content-Type': 'application/json'
        }
    });
    
    if (!response.ok) {
        if (response.status === 401) {
            // Token expirado - redirigir a login
            window.location.href = '/login';
        }
        throw new Error('Error de autenticación');
    }
    
    return await response.json();
}


// ====== EJEMPLO 5: GET múltiples recursos en paralelo ======
async function getDashboardData() {
    try {
        // Ejecutar todas las peticiones en paralelo
        const [users, posts, comments] = await Promise.all([
            fetch('https://jsonplaceholder.typicode.com/users')
                .then(r => r.json()),
            fetch('https://jsonplaceholder.typicode.com/posts')
                .then(r => r.json()),
            fetch('https://jsonplaceholder.typicode.com/comments')
                .then(r => r.json())
        ]);
        
        return { users, posts, comments };
    } catch (error) {
        console.error('Error cargando dashboard:', error);
        throw error;
    }
}

// Uso:
const dashboard = await getDashboardData();
console.log(`${dashboard.users.length} usuarios`);
console.log(`${dashboard.posts.length} posts`);

async function createUser(userData) {
    const response = await fetch('https://jsonplaceholder.typicode.com/users', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify(userData)

    const newUser = await response.json();
    console.log('Usuario creado:', newUser);
    return newUser;
}

// Uso:
const nuevoUsuario = await createUser({
    nombre: 'Juan Pérez',
    email: 'juan@email.com',
    password: 'secreto123'
});


// ====== EJEMPLO 2: POST con FormData (archivos) ======
async function uploadFile(file) {
    const formData = new FormData();
    formData.append('file', file);
    formData.append('description', 'Mi archivo');
    
    // ⚠️ NO pongas Content-Type header con FormData
    // El navegador lo pone automáticamente con el boundary
    const response = await fetch('/api/upload', {
        method: 'POST',
        body: formData
    });
    
    return await response.json();
}

// Uso con input file:
const fileInput = document.querySelector('input[type="file"]');
fileInput.addEventListener('change', async (e) => {
    const file = e.target.files[0];
    const result = await uploadFile(file);
    console.log('Archivo subido:', result);
});


// ====== EJEMPLO 3: POST de formulario completo ======
async function submitForm(formElement) {
    const formData = new FormData(formElement);
    const data = Object.fromEntries(formData);
    
    try {
        const response = await fetch('/api/contact', {
            method: 'POST',
            headers: {
                'Content-Type': 'application/json'
            },
            body: JSON.stringify(data)
        });
        
        if (!response.ok) {
            throw new Error('Error al enviar formulario');
        }
        
        return await response.json();
    } catch (error) {
        console.error(error);
        throw error;
    }
}

// Uso:
document.getElementById('contactForm').addEventListener('submit', async (e) => {
    e.preventDefault();
    const result = await submitForm(e.target);
    alert('¡Mensaje enviado!');
});


// ====== EJEMPLO 4: POST Login con manejo de sesión ======
async function login(email, password) {
    const response = await fetch('/api/auth/login', {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({ email, password }),
        credentials: 'include' // Para manejar cookies
    });
    
    if (!response.ok) {
        const error = await response.json();
        throw new Error(error.message || 'Error de login');
    }
    
    const { token, user } = await response.json();
    
    // Guardar token
    localStorage.setItem('authToken', token);
    localStorage.setItem('user', JSON.stringify(user));
    
    return user;
}


// ====== EJEMPLO 5: POST con retry automático ======
async function postWithRetry(url, data, maxRetries = 3) {
    let lastError;
    
    for (let attempt = 1; attempt <= maxRetries; attempt++) {
        try {
            const response = await fetch(url, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(data)
            });
            
            if (response.ok) {
                return await response.json();
            }
            
            // Si es error del servidor (5xx), reintentar
            if (response.status >= 500) {
                throw new Error(`Server error: ${response.status}`);
            }
            
            // Si es error del cliente (4xx), no reintentar
            const errorData = await response.json();
            throw new Error(errorData.message);
            
        } catch (error) {
            lastError = error;
            console.log(`Intento ${attempt} fallido: ${error.message}`);
            
            if (attempt < maxRetries) {
                // Esperar antes de reintentar (exponential backoff)
                await new Promise(r => setTimeout(r, 1000 * attempt));
            }
        }
    }
    
    throw lastError;
}

// ====== PATCH - Actualización parcial ======
async function patchUser(id, partialData) {
    const response = await fetch(`/api/users/${id}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(partialData) // Solo campos a actualizar
    });
    return await response.json();
}

// Uso: Solo envías lo que quieres cambiar
await patchUser(1, { email: 'nuevo@email.com' });


// ====== DELETE - Eliminar recurso ======
async function deleteUser(id) {
    const response = await fetch(`/api/users/${id}`, {
        method: 'DELETE'
    });
    
    if (!response.ok) {
        throw new Error('Error al eliminar');
    }
    
    return true; // DELETE típicamente retorna 204 No Content
}

// Uso con confirmación:
async function handleDelete(id) {
    if (confirm('¿Seguro que deseas eliminar?')) {
        await deleteUser(id);
        alert('Usuario eliminado');
        reloadUserList();
    }
}

Manejo de Errores 

// ============================================
// MANEJO DE ERRORES EN FETCH
// ============================================

// ⚠️ IMPORTANTE: fetch() solo rechaza la Promise en errores de red
// Los errores HTTP (404, 500, etc) NO causan throw

// ====== FORMA CORRECTA DE MANEJAR ERRORES ======
async function fetchWithErrorHandling(url) {
    try {
        const response = await fetch(url);
        
        // Verificar si la respuesta es exitosa
        if (!response.ok) {
            // Intentar obtener mensaje de error del servidor
            let errorMessage = `Error HTTP: ${response.status}`;
            
            try {
                const errorData = await response.json();
                errorMessage = errorData.message || errorMessage;
            } catch {
                // Si no hay JSON en la respuesta de error
            }
            
            throw new Error(errorMessage);
        }
        
        return await response.json();
        
    } catch (error) {
        // Aquí llegan:
        // - Errores de red (sin conexión, DNS, CORS)
        // - Errores que lanzamos arriba
        
        if (error.name === 'TypeError') {
            // Error de red
            console.error('Error de conexión:', error.message);
        }
        
        throw error;
    }
}


// ====== ERRORES HTTP COMUNES ======
async function handleHttpErrors(response) {
    switch (response.status) {
        case 400:
            throw new Error('Datos inválidos');
        case 401:
            // No autorizado - token inválido/expirado
            localStorage.removeItem('authToken');
            window.location.href = '/login';
            break;
        case 403:
            throw new Error('No tienes permisos para esta acción');
        case 404:
            throw new Error('Recurso no encontrado');
        case 422:
            const errors = await response.json();
            throw new ValidationError(errors);
        case 429:
            throw new Error('Demasiadas peticiones. Intenta más tarde.');
        case 500:
            throw new Error('Error del servidor. Intenta más tarde.');
        default:
            throw new Error(`Error: ${response.status}`);
    }
}


// ====== TIMEOUT PARA FETCH ======
async function fetchWithTimeout(url, options = {}, timeout = 5000) {
    const controller = new AbortController();
    const id = setTimeout(() => controller.abort(), timeout);
    
    try {
        const response = await fetch(url, {
            ...options,
            signal: controller.signal
        });
        clearTimeout(id);
        return response;
    } catch (error) {
        clearTimeout(id);
        if (error.name === 'AbortError') {
            throw new Error('La petición tardó demasiado');
        }
        throw error;
    }
}


// ====== USO EN COMPONENTES ======
async function loadUsers() {
    const container = document.getElementById('users');
    container.innerHTML = '<div class="loading">Cargando...</div>';
    
    try {
        const users = await fetchWithErrorHandling('/api/users');
        container.innerHTML = renderUsers(users);
    } catch (error) {
        container.innerHTML = `
            <div class="error-state">
                <p>Error: ${error.message}</p>
                <button onclick="loadUsers()">Reintentar</button>
            </div>
        `;
    }
}

Clase ApiService Reutilizable

Mejor Práctica: Crea una clase o módulo para centralizar todas tus llamadas a API. 
// ============================================
// API SERVICE CLASS - CÓDIGO REUTILIZABLE
// ============================================

class ApiService {
    constructor(baseURL) {
        this.baseURL = baseURL;
        this.defaultHeaders = {
            'Content-Type': 'application/json'
        };
    }
    
    // Obtener headers con token
    getHeaders() {
        const headers = { ...this.defaultHeaders };
        const token = localStorage.getItem('authToken');
        
        if (token) {
            headers['Authorization'] = `Bearer ${token}`;
        }
        
        return headers;
    }
    
    // Método principal para todas las peticiones
    async request(endpoint, options = {}) {
        const url = `${this.baseURL}${endpoint}`;
        
        const config = {
            ...options,
            headers: {
                ...this.getHeaders(),
                ...options.headers
            }
        };
        
        try {
            const response = await fetch(url, config);
            
            // Manejar respuestas sin contenido
            if (response.status === 204) {
                return null;
            }
            
            const data = await response.json();
            
            if (!response.ok) {
                throw new ApiError(data.message || 'Error de API', response.status);
            }
            
            return data;
            
        } catch (error) {
            if (error instanceof ApiError) {
                throw error;
            }
            throw new ApiError('Error de conexión', 0);
        }
    }
    
    // Métodos de conveniencia
    async get(endpoint, params = {}) {
        const queryString = new URLSearchParams(params).toString();
        const url = queryString ? `${endpoint}?${queryString}` : endpoint;
        return this.request(url);
    }
    
    async post(endpoint, data) {
        return this.request(endpoint, {
            method: 'POST',
            body: JSON.stringify(data)
        });
    }
    
    async put(endpoint, data) {
        return this.request(endpoint, {
            method: 'PUT',
            body: JSON.stringify(data)
        });
    }
    
    async patch(endpoint, data) {
        return this.request(endpoint, {
            method: 'PATCH',
            body: JSON.stringify(data)
        });
    }
    
    async delete(endpoint) {
        return this.request(endpoint, { method: 'DELETE' });
    }
}

// Clase de error personalizada
class ApiError extends Error {
    constructor(message, status) {
        super(message);
        this.name = 'ApiError';
        this.status = status;
    }
}

// ====== INSTANCIAR Y USAR ======
const api = new ApiService('http://localhost:3000');

// GET usuarios
const users = await api.get('/users');

// GET con parámetros
const filteredUsers = await api.get('/users', { 
    role: 'admin', 
    active: true 
});

// POST crear
const newUser = await api.post('/users', {
    nombre: 'Juan',
    email: 'juan@mail.com'
});

// PATCH actualizar
const updated = await api.patch('/users/1', { nombre: 'Juan Actualizado' });

// DELETE eliminar
await api.delete('/users/1');
