1. Fetch Wrapper con Interceptores

Cliente HTTP personalizado con interceptores, manejo de errores y autenticación automática.

/**
 * HTTP CLIENT AVANZADO
 * Con interceptores, retry, timeout y manejo de errores
 */

class HttpClient {
    constructor(baseURL = '') {
        this.baseURL = baseURL;
        this.interceptors = {
            request: [],
            response: []
        };
    }
    
    // Agregar interceptor de request
    addRequestInterceptor(fn) {
        this.interceptors.request.push(fn);
    }
    
    // Agregar interceptor de response
    addResponseInterceptor(fn) {
        this.interceptors.response.push(fn);
    }
    
    // Método principal de fetch
    async request(endpoint, options = {}) {
        let config = {
            url: this.baseURL + endpoint,
            method: 'GET',
            headers: {
                'Content-Type': 'application/json'
            },
            ...options
        };
        
        // Ejecutar interceptores de request
        for (const interceptor of this.interceptors.request) {
            config = await interceptor(config);
        }
        
        try {
            let response = await fetch(config.url, config);
            
            // Ejecutar interceptores de response
            for (const interceptor of this.interceptors.response) {
                response = await interceptor(response);
            }
            
            if (!response.ok) {
                throw new HttpError(response.status, response.statusText);
            }
            
            return response.json();
            
        } catch (error) {
            throw error;
        }
    }
    
    // Métodos de conveniencia
    get(url, options) { return this.request(url, { ...options, method: 'GET' }); }
    post(url, data, options) { return this.request(url, { ...options, method: 'POST', body: JSON.stringify(data) }); }
    put(url, data, options) { return this.request(url, { ...options, method: 'PUT', body: JSON.stringify(data) }); }
    delete(url, options) { return this.request(url, { ...options, method: 'DELETE' }); }
}

// Clase de error personalizada
class HttpError extends Error {
    constructor(status, message) {
        super(message);
        this.status = status;
    }
}

// ====== USO ======
const api = new HttpClient('https://api.ejemplo.com');

// Interceptor: Agregar token automáticamente
api.addRequestInterceptor((config) => {
    const token = localStorage.getItem('token');
    if (token) {
        config.headers['Authorization'] = `Bearer ${token}`;
    }
    return config;
});

// Interceptor: Log de requests
api.addRequestInterceptor((config) => {
    console.log(`📡 ${config.method} ${config.url}`);
    return config;
});

// Interceptor: Refresh token si expira
api.addResponseInterceptor(async (response) => {
    if (response.status === 401) {
        console.log('🔄 Token expirado, renovando...');
        await refreshToken();
    }
    return response;
});

// Usar el cliente
const users = await api.get('/users');
const newUser = await api.post('/users', { name: 'Juan' });

2. Retry con Exponential Backoff

Reintentar peticiones fallidas con espera exponencial.

/**
 * FETCH CON RETRY Y EXPONENTIAL BACKOFF
 */

async function fetchWithRetry(url, options = {}, maxRetries = 3) {
    let lastError;
    
    for (let attempt = 0; attempt < maxRetries; attempt++) {
        try {
            const response = await fetch(url, options);
            
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }
            
            return await response.json();
            
        } catch (error) {
            lastError = error;
            
            // Calcular delay exponencial: 1s, 2s, 4s, 8s...
            const delay = Math.pow(2, attempt) * 1000;
            
            console.log(`⚠️ Intento ${attempt + 1} fallido. Reintentando en ${delay}ms...`);
            
            // Esperar antes de reintentar
            await new Promise(r => setTimeout(r, delay));
        }
    }
    
    throw lastError;
}

// ====== VERSIÓN AVANZADA CON OPCIONES ======
async function fetchWithRetryAdvanced(url, {
    maxRetries = 3,
    baseDelay = 1000,
    maxDelay = 30000,
    retryOn = [408, 429, 500, 502, 503, 504],
    onRetry = null,
    ...fetchOptions
} = {}) {
    let lastError;
    
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
        try {
            const response = await fetch(url, fetchOptions);
            
            // Solo reintentar en ciertos códigos de error
            if (!response.ok && retryOn.includes(response.status) && attempt < maxRetries) {
                throw new Error(`HTTP ${response.status}`);
            }
            
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}: ${response.statusText}`);
            }
            
            return response;
            
        } catch (error) {
            lastError = error;
            
            if (attempt >= maxRetries) break;
            
            // Exponential backoff con jitter (aleatoridad)
            const exponentialDelay = Math.min(
                baseDelay * Math.pow(2, attempt),
                maxDelay
            );
            const jitter = Math.random() * 0.3 * exponentialDelay;
            const delay = exponentialDelay + jitter;
            
            // Callback opcional
            if (onRetry) {
                onRetry({ attempt, delay, error });
            }
            
            await new Promise(r => setTimeout(r, delay));
        }
    }
    
    throw lastError;
}

// Uso
const data = await fetchWithRetryAdvanced('https://api.ejemplo.com/data', {
    maxRetries: 5,
    baseDelay: 1000,
    onRetry: ({ attempt, delay }) => {
        console.log(`🔄 Retry #${attempt + 1} en ${delay}ms`);
    }
});

3. AbortController y Timeout

Cancelar peticiones y establecer timeouts.

/**
 * FETCH CON TIMEOUT Y CANCELACIÓN
 */

// Fetch con timeout
async function fetchWithTimeout(url, options = {}, timeout = 5000) {
    const controller = new AbortController();
    const timeoutId = setTimeout(() => controller.abort(), timeout);
    
    try {
        const response = await fetch(url, {
            ...options,
            signal: controller.signal
        });
        
        clearTimeout(timeoutId);
        return response;
        
    } catch (error) {
        clearTimeout(timeoutId);
        
        if (error.name === 'AbortError') {
            throw new Error(`⏱️ Timeout: Request exceeded ${timeout}ms`);
        }
        
        throw error;
    }
}

// Uso
try {
    const data = await fetchWithTimeout('/api/slow-endpoint', {}, 3000);
} catch (error) {
    console.log(error.message); // ⏱️ Timeout: Request exceeded 3000ms
}


// ====== CANCELAR PETICIONES MANUALMENTE ======
class CancellableRequest {
    constructor() {
        this.controller = null;
    }
    
    async fetch(url, options = {}) {
        // Cancelar petición anterior si existe
        if (this.controller) {
            this.controller.abort();
        }
        
        this.controller = new AbortController();
        
        try {
            const response = await fetch(url, {
                ...options,
                signal: this.controller.signal
            });
            return response.json();
        } catch (error) {
            if (error.name === 'AbortError') {
                console.log('🚫 Request cancelado');
                return null;
            }
            throw error;
        }
    }
    
    cancel() {
        if (this.controller) {
            this.controller.abort();
        }
    }
}

// Uso: Búsqueda que cancela peticiones anteriores
const searchRequest = new CancellableRequest();

searchInput.addEventListener('input', async (e) => {
    const query = e.target.value;
    
    // Automáticamente cancela la búsqueda anterior
    const results = await searchRequest.fetch(`/api/search?q=${query}`);
    
    if (results) {
        displayResults(results);
    }
});


// ====== ABORT EN COMPONENTES (cleanup) ======
// Para React/Vue/etc cuando el componente se desmonta

function useAbortableFetch() {
    const controller = new AbortController();
    
    const fetchData = async (url) => {
        try {
            const res = await fetch(url, { signal: controller.signal });
            return res.json();
        } catch (e) {
            if (e.name !== 'AbortError') throw e;
        }
    };
    
    const abort = () => controller.abort();
    
    return { fetchData, abort };
}

// En un componente
const { fetchData, abort } = useAbortableFetch();

// Al montar: cargar datos
fetchData('/api/data').then(setData);

// Al desmontar: cancelar
abort();

4. Sistema de Caché para APIs

Cachear respuestas para evitar peticiones innecesarias.

/**
 * SISTEMA DE CACHÉ PARA FETCH
 */

class FetchCache {
    constructor(defaultTTL = 60000) { // 1 minuto por defecto
        this.cache = new Map();
        this.defaultTTL = defaultTTL;
    }
    
    // Generar key única para la petición
    _getCacheKey(url, options = {}) {
        return `${options.method || 'GET'}:${url}:${JSON.stringify(options.body || '')}`;
    }
    
    // Verificar si el cache es válido
    _isValid(entry) {
        return entry && Date.now() < entry.expiresAt;
    }
    
    // Fetch con caché
    async fetch(url, options = {}, ttl = this.defaultTTL) {
        const key = this._getCacheKey(url, options);
        
        // Verificar caché
        const cached = this.cache.get(key);
        if (this._isValid(cached)) {
            console.log('📦 Desde caché:', url);
            return cached.data;
        }
        
        // Hacer petición
        console.log('🌐 Fetch:', url);
        const response = await fetch(url, options);
        const data = await response.json();
        
        // Guardar en caché
        this.cache.set(key, {
            data,
            expiresAt: Date.now() + ttl
        });
        
        return data;
    }
    
    // Invalidar caché
    invalidate(urlPattern) {
        for (const key of this.cache.keys()) {
            if (key.includes(urlPattern)) {
                this.cache.delete(key);
            }
        }
    }
    
    // Limpiar todo el caché
    clear() {
        this.cache.clear();
    }
}


// ====== CACHÉ CON STALE-WHILE-REVALIDATE ======
class SWRCache {
    constructor(freshTime = 30000, staleTime = 300000) {
        this.cache = new Map();
        this.freshTime = freshTime;   // Tiempo que se considera "fresh"
        this.staleTime = staleTime;   // Tiempo máximo de stale
    }
    
    async fetch(url, options = {}) {
        const key = url;
        const cached = this.cache.get(key);
        const now = Date.now();
        
        // Si está fresh, devolver caché directamente
        if (cached && now < cached.freshUntil) {
            console.log('✨ Fresh cache');
            return cached.data;
        }
        
        // Si está stale pero válido, devolver y revalidar en background
        if (cached && now < cached.staleUntil) {
            console.log('📦 Stale cache (revalidando...)');
            this._revalidate(url, options, key); // No await!
            return cached.data;
        }
        
        // Si no hay caché o expiró, fetch normal
        return this._fetchAndCache(url, options, key);
    }
    
    async _fetchAndCache(url, options, key) {
        const response = await fetch(url, options);
        const data = await response.json();
        const now = Date.now();
        
        this.cache.set(key, {
            data,
            freshUntil: now + this.freshTime,
            staleUntil: now + this.staleTime
        });
        
        return data;
    }
    
    async _revalidate(url, options, key) {
        try {
            await this._fetchAndCache(url, options, key);
            console.log('✅ Revalidación completada');
        } catch (e) {
            console.log('❌ Revalidación fallida');
        }
    }
}

// Uso
const swrCache = new SWRCache();

// Primera llamada: fetch normal
await swrCache.fetch('/api/users');

// Llamada inmediata: desde fresh cache
await swrCache.fetch('/api/users');

// Después de 30s: devuelve stale y revalida en background
await swrCache.fetch('/api/users');

5. Cliente API Completo

Todo junto: interceptores, retry, timeout, caché.

/**
 * API CLIENT PROFESIONAL
 * Combina: Interceptores, Retry, Timeout, Caché, Cancelación
 */

class APIClient {
    constructor(config = {}) {
        this.baseURL = config.baseURL || '';
        this.timeout = config.timeout || 10000;
        this.maxRetries = config.maxRetries || 3;
        this.cache = new Map();
        this.cacheTTL = config.cacheTTL || 60000;
        this.interceptors = { request: [], response: [] };
        this.pendingRequests = new Map();
    }
    
    // Interceptores
    useRequestInterceptor(fn) { this.interceptors.request.push(fn); return this; }
    useResponseInterceptor(fn) { this.interceptors.response.push(fn); return this; }
    
    // Request principal
    async request(endpoint, options = {}) {
        const url = this.baseURL + endpoint;
        const cacheKey = `${options.method || 'GET'}:${url}`;
        
        // Verificar caché (solo GET)
        if ((!options.method || options.method === 'GET') && options.cache !== false) {
            const cached = this._getFromCache(cacheKey);
            if (cached) return cached;
        }
        
        // Deduplicar peticiones idénticas en vuelo
        if (this.pendingRequests.has(cacheKey)) {
            return this.pendingRequests.get(cacheKey);
        }
        
        const requestPromise = this._executeRequest(url, options, cacheKey);
        this.pendingRequests.set(cacheKey, requestPromise);
        
        try {
            return await requestPromise;
        } finally {
            this.pendingRequests.delete(cacheKey);
        }
    }
    
    async _executeRequest(url, options, cacheKey) {
        let config = {
            url,
            method: 'GET',
            headers: { 'Content-Type': 'application/json' },
            ...options
        };
        
        // Ejecutar interceptores de request
        for (const interceptor of this.interceptors.request) {
            config = await interceptor(config);
        }
        
        // Ejecutar con retry
        let lastError;
        for (let attempt = 0; attempt <= this.maxRetries; attempt++) {
            try {
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), this.timeout);
                
                let response = await fetch(config.url, {
                    ...config,
                    signal: controller.signal
                });
                
                clearTimeout(timeoutId);
                
                // Ejecutar interceptores de response
                for (const interceptor of this.interceptors.response) {
                    response = await interceptor(response, config);
                }
                
                if (!response.ok) {
                    throw new Error(`HTTP ${response.status}`);
                }
                
                const data = await response.json();
                
                // Guardar en caché
                if (config.method === 'GET') {
                    this._saveToCache(cacheKey, data);
                }
                
                return data;
                
            } catch (error) {
                lastError = error;
                if (attempt < this.maxRetries) {
                    await new Promise(r => setTimeout(r, Math.pow(2, attempt) * 1000));
                }
            }
        }
        
        throw lastError;
    }
    
    _getFromCache(key) {
        const cached = this.cache.get(key);
        if (cached && Date.now() < cached.expiresAt) {
            return cached.data;
        }
        return null;
    }
    
    _saveToCache(key, data) {
        this.cache.set(key, {
            data,
            expiresAt: Date.now() + this.cacheTTL
        });
    }
    
    // Métodos de conveniencia
    get(url, opts) { return this.request(url, { ...opts, method: 'GET' }); }
    post(url, data, opts) { return this.request(url, { ...opts, method: 'POST', body: JSON.stringify(data) }); }
    put(url, data, opts) { return this.request(url, { ...opts, method: 'PUT', body: JSON.stringify(data) }); }
    delete(url, opts) { return this.request(url, { ...opts, method: 'DELETE' }); }
    
    clearCache() { this.cache.clear(); }
}

// ====== USO ======
const api = new APIClient({
    baseURL: 'https://api.ejemplo.com',
    timeout: 5000,
    maxRetries: 3,
    cacheTTL: 60000
});

// Agregar interceptor de auth
api.useRequestInterceptor((config) => {
    config.headers['Authorization'] = `Bearer ${getToken()}`;
    return config;
});

// Agregar interceptor de errores
api.useResponseInterceptor(async (response, config) => {
    if (response.status === 401) {
        await refreshToken();
        // Reintentar request original
    }
    return response;
});

// Usar la API
const users = await api.get('/users');
const user = await api.post('/users', { name: 'Juan' });
