1. Cart Manager Profesional

Sistema completo de carrito con eventos, persistencia y validaciones.

/**
 * CART MANAGER PROFESIONAL
 * Sistema completo con eventos, persistencia y validaciones
 */

class CartManager {
    static STORAGE_KEY = 'cart_data';
    
    constructor(options = {}) {
        this.items = [];
        this.listeners = {};
        this.maxQuantityPerItem = options.maxQuantityPerItem || 99;
        this.autoSave = options.autoSave !== false;
        
        // Cargar datos guardados
        this._loadFromStorage();
    }
    
    // ====== SISTEMA DE EVENTOS ======
    on(event, callback) {
        if (!this.listeners[event]) {
            this.listeners[event] = [];
        }
        this.listeners[event].push(callback);
        return () => this.off(event, callback);
    }
    
    off(event, callback) {
        if (this.listeners[event]) {
            this.listeners[event] = this.listeners[event]
                .filter(cb => cb !== callback);
        }
    }
    
    _emit(event, data) {
        if (this.listeners[event]) {
            this.listeners[event].forEach(cb => cb(data));
        }
        // Evento genérico de cambio
        if (event !== 'change') {
            this._emit('change', { event, data });
        }
    }
    
    // ====== OPERACIONES DE CARRITO ======
    addItem(product, quantity = 1) {
        const existingIndex = this.items.findIndex(item => 
            item.id === product.id && 
            JSON.stringify(item.variant || {}) === JSON.stringify(product.variant || {})
        );
        
        if (existingIndex > -1) {
            const newQty = this.items[existingIndex].quantity + quantity;
            this.updateQuantity(existingIndex, newQty);
        } else {
            const cartItem = {
                ...product,
                quantity: Math.min(quantity, this.maxQuantityPerItem),
                addedAt: Date.now()
            };
            this.items.push(cartItem);
            this._emit('itemAdded', cartItem);
        }
        
        this._persist();
        return this;
    }
    
    removeItem(index) {
        const removed = this.items.splice(index, 1)[0];
        if (removed) {
            this._emit('itemRemoved', removed);
            this._persist();
        }
        return removed;
    }
    
    removeByProductId(productId) {
        const index = this.items.findIndex(item => item.id === productId);
        if (index > -1) {
            return this.removeItem(index);
        }
        return null;
    }
    
    updateQuantity(index, quantity) {
        if (this.items[index]) {
            const newQty = Math.max(0, Math.min(quantity, this.maxQuantityPerItem));
            
            if (newQty === 0) {
                return this.removeItem(index);
            }
            
            this.items[index].quantity = newQty;
            this._emit('quantityUpdated', { index, quantity: newQty });
            this._persist();
        }
        return this;
    }
    
    clear() {
        this.items = [];
        this._emit('cartCleared', {});
        this._persist();
        return this;
    }
    
    // ====== CÁLCULOS ======
    get subtotal() {
        return this.items.reduce((sum, item) => {
            return sum + (item.price * item.quantity);
        }, 0);
    }
    
    get totalItems() {
        return this.items.reduce((sum, item) => sum + item.quantity, 0);
    }
    
    get isEmpty() {
        return this.items.length === 0;
    }
    
    getItemTotal(index) {
        const item = this.items[index];
        return item ? item.price * item.quantity : 0;
    }
    
    // ====== PERSISTENCIA ======
    _persist() {
        if (this.autoSave) {
            localStorage.setItem(CartManager.STORAGE_KEY, JSON.stringify({
                items: this.items,
                updatedAt: Date.now()
            }));
        }
    }
    
    _loadFromStorage() {
        try {
            const saved = localStorage.getItem(CartManager.STORAGE_KEY);
            if (saved) {
                const data = JSON.parse(saved);
                this.items = data.items || [];
            }
        } catch (e) {
            console.error('Error loading cart:', e);
        }
    }
    
    toJSON() {
        return {
            items: this.items,
            subtotal: this.subtotal,
            totalItems: this.totalItems
        };
    }
}

// ====== USO ======
const cart = new CartManager();

// Escuchar eventos
cart.on('itemAdded', (item) => {
    console.log('🛒 Agregado:', item.name);
    showNotification(`${item.name} agregado al carrito`);
});

cart.on('change', () => {
    updateCartBadge(cart.totalItems);
    renderCart(cart.items);
});

2. Sistema de Cupones y Descuentos

Manejo de cupones con diferentes tipos de descuento.

/**
 * SISTEMA DE CUPONES
 */

class CouponManager {
    constructor() {
        this.coupons = new Map();
        this.appliedCoupon = null;
    }
    
    // Registrar cupón
    registerCoupon(coupon) {
        this.coupons.set(coupon.code.toUpperCase(), {
            ...coupon,
            code: coupon.code.toUpperCase()
        });
        return this;
    }
    
    // Validar cupón
    validateCoupon(code, cartData) {
        const coupon = this.coupons.get(code.toUpperCase());
        
        if (!coupon) {
            return { valid: false, error: 'Cupón no encontrado' };
        }
        
        // Verificar fechas
        const now = Date.now();
        if (coupon.startDate && now < coupon.startDate) {
            return { valid: false, error: 'Cupón aún no está activo' };
        }
        if (coupon.endDate && now > coupon.endDate) {
            return { valid: false, error: 'Cupón expirado' };
        }
        
        // Verificar usos
        if (coupon.maxUses && coupon.usedCount >= coupon.maxUses) {
            return { valid: false, error: 'Cupón agotado' };
        }
        
        // Verificar mínimo de compra
        if (coupon.minPurchase && cartData.subtotal < coupon.minPurchase) {
            return { 
                valid: false, 
                error: `Mínimo de compra: $${coupon.minPurchase}` 
            };
        }
        
        // Verificar productos aplicables
        if (coupon.applicableProducts) {
            const hasApplicable = cartData.items.some(item => 
                coupon.applicableProducts.includes(item.id)
            );
            if (!hasApplicable) {
                return { valid: false, error: 'No aplicable a estos productos' };
            }
        }
        
        return { valid: true, coupon };
    }
    
    // Calcular descuento
    calculateDiscount(coupon, cartData) {
        let discount = 0;
        
        switch (coupon.type) {
            case 'percentage':
                discount = (cartData.subtotal * coupon.value) / 100;
                break;
                
            case 'fixed':
                discount = coupon.value;
                break;
                
            case 'freeShipping':
                discount = cartData.shipping || 0;
                break;
                
            case 'buyXgetY':
                // Compra X, lleva Y gratis
                const eligible = cartData.items
                    .filter(item => coupon.applicableProducts.includes(item.id));
                const totalQty = eligible.reduce((sum, i) => sum + i.quantity, 0);
                const freeItems = Math.floor(totalQty / (coupon.buyX + coupon.getY)) * coupon.getY;
                const cheapest = Math.min(...eligible.map(i => i.price));
                discount = freeItems * cheapest;
                break;
        }
        
        // Aplicar máximo descuento si existe
        if (coupon.maxDiscount) {
            discount = Math.min(discount, coupon.maxDiscount);
        }
        
        return Math.round(discount * 100) / 100;
    }
    
    // Aplicar cupón
    applyCoupon(code, cartData) {
        const validation = this.validateCoupon(code, cartData);
        
        if (!validation.valid) {
            return validation;
        }
        
        const discount = this.calculateDiscount(validation.coupon, cartData);
        this.appliedCoupon = { ...validation.coupon, discount };
        
        return {
            valid: true,
            coupon: this.appliedCoupon,
            discount
        };
    }
    
    removeCoupon() {
        this.appliedCoupon = null;
    }
}

// ====== USO ======
const couponManager = new CouponManager();

3. Control de Inventario

Sistema de control de stock con reservas y alertas.

/**
 * SISTEMA DE INVENTARIO
 */

class InventoryManager {
    constructor() {
        this.products = new Map();
        this.reservations = new Map();
        this.lowStockThreshold = 5;
    }
    
    // Agregar producto al inventario
    addProduct(product) {
        this.products.set(product.id, {
            ...product,
            stock: product.stock || 0,
            reserved: 0
        });
        return this;
    }
    
    // Obtener stock disponible (total - reservado)
    getAvailableStock(productId) {
        const product = this.products.get(productId);
        if (!product) return 0;
        return product.stock - product.reserved;
    }
    
    // Verificar disponibilidad
    checkAvailability(productId, quantity) {
        const available = this.getAvailableStock(productId);
        return {
            available: available >= quantity,
            currentStock: available,
            requested: quantity
        };
    }
    
    // Reservar stock (para checkout)
    reserveStock(sessionId, items) {
        const reservationId = `res_${Date.now()}_${Math.random().toString(36).slice(2)}`;
        const reserved = [];
        const failed = [];
        
        for (const item of items) {
            const product = this.products.get(item.productId);
            
            if (!product || this.getAvailableStock(item.productId) < item.quantity) {
                failed.push({ ...item, reason: 'Stock insuficiente' });
                continue;
            }
            
            // Reservar
            product.reserved += item.quantity;
            reserved.push(item);
        }
        
        if (reserved.length > 0) {
            this.reservations.set(reservationId, {
                sessionId,
                items: reserved,
                createdAt: Date.now(),
                expiresAt: Date.now() + (15 * 60 * 1000) // 15 minutos
            });
        }
        
        return { reservationId, reserved, failed };
    }
    
    // Confirmar compra (descontar del stock)
    confirmPurchase(reservationId) {
        const reservation = this.reservations.get(reservationId);
        if (!reservation) {
            return { success: false, error: 'Reserva no encontrada' };
        }
        
        for (const item of reservation.items) {
            const product = this.products.get(item.productId);
            if (product) {
                product.stock -= item.quantity;
                product.reserved -= item.quantity;
                
                // Verificar bajo stock
                if (product.stock <= this.lowStockThreshold) {
                    this._emitLowStockAlert(product);
                }
            }
        }
        
        this.reservations.delete(reservationId);
        return { success: true };
    }
    
    // Cancelar reserva
    cancelReservation(reservationId) {
        const reservation = this.reservations.get(reservationId);
        if (!reservation) return;
        
        for (const item of reservation.items) {
            const product = this.products.get(item.productId);
            if (product) {
                product.reserved -= item.quantity;
            }
        }
        
        this.reservations.delete(reservationId);
    }
    
    // Limpiar reservas expiradas
    cleanExpiredReservations() {
        const now = Date.now();
        for (const [id, reservation] of this.reservations) {
            if (now > reservation.expiresAt) {
                this.cancelReservation(id);
            }
        }
    }
    
    _emitLowStockAlert(product) {
        console.warn(`⚠️ BAJO STOCK: ${product.name} (${product.stock} unidades)`);
        // Aquí podrías enviar notificación, email, etc.
    }
}

// ====== USO ======
const inventory = new InventoryManager();

// Agregar productos

    .addProduct({ id: 1, name: 'iPhone 15', stock: 50 })
    .addProduct({ id: 2, name: 'MacBook Pro', stock: 20 });

}

    { productId: 1, quantity: 2 }
]);

// Al confirmar pago

4. Proceso de Checkout Completo

Flujo de checkout con validaciones, cálculos y orden final.

/**
 * CHECKOUT MANAGER
 * Proceso completo de pago
 */

class CheckoutManager {
    constructor(cart, inventory, couponManager) {
        this.cart = cart;
        this.inventory = inventory;
        this.couponManager = couponManager;
        this.shippingMethods = [];
        this.taxRate = 0.16; // 16% IVA
        this.reservationId = null;
    }
    
    // Configurar métodos de envío
    setShippingMethods(methods) {
        this.shippingMethods = methods;
        return this;
    }
    
    // Iniciar checkout
    async initCheckout() {
        // Verificar carrito no vacío
        if (this.cart.isEmpty) {
            throw new Error('El carrito está vacío');
        }
        
        // Reservar inventario
        const items = this.cart.items.map(item => ({
            productId: item.id,
            quantity: item.quantity
        }));
        
        const reservation = this.inventory.reserveStock('checkout', items);
        
        if (reservation.failed.length > 0) {
            throw new Error('Algunos productos no tienen stock suficiente');
        }
        
        this.reservationId = reservation.reservationId;
        
        return {
            success: true,
            reservationId: this.reservationId,
            expiresIn: 15 * 60 // 15 minutos
        };
    }
    
    // Calcular totales
    calculateTotals(shippingMethodId) {
        const subtotal = this.cart.subtotal;
        
        // Envío
        const shipping = this.shippingMethods
            .find(m => m.id === shippingMethodId)?.price || 0;
        
        // Descuento de cupón
        const couponDiscount = this.couponManager.appliedCoupon?.discount || 0;
        
        // Subtotal después de descuento
        const subtotalAfterDiscount = subtotal - couponDiscount;
        
        // Impuestos
        const tax = subtotalAfterDiscount * this.taxRate;
        
        // Total
        const total = subtotalAfterDiscount + shipping + tax;
        
        return {
            subtotal: this._round(subtotal),
            shipping: this._round(shipping),
            couponDiscount: this._round(couponDiscount),
            subtotalAfterDiscount: this._round(subtotalAfterDiscount),
            tax: this._round(tax),
            total: this._round(total)
        };
    }
    
    // Crear orden
    async createOrder(customerData, shippingMethodId, paymentMethod) {
        const totals = this.calculateTotals(shippingMethodId);
        
        const order = {
            id: `ORD-${Date.now()}-${Math.random().toString(36).slice(2, 8).toUpperCase()}`,
            customer: customerData,
            items: this.cart.items.map(item => ({
                productId: item.id,
                name: item.name,
                price: item.price,
                quantity: item.quantity,
                total: item.price * item.quantity
            })),
            shipping: {
                method: this.shippingMethods.find(m => m.id === shippingMethodId),
                address: customerData.shippingAddress
            },
            payment: {
                method: paymentMethod,
                status: 'pending'
            },
            coupon: this.couponManager.appliedCoupon?.code || null,
            totals,
            status: 'pending',
            createdAt: new Date().toISOString()
        };
        
        return order;
    }
    
    // Confirmar orden (después del pago)
    async confirmOrder(order) {
        // Confirmar reserva de inventario
        this.inventory.confirmPurchase(this.reservationId);
        
        // Limpiar carrito
        this.cart.clear();
        
        // Limpiar cupón
        this.couponManager.removeCoupon();
        
        // Guardar orden
        this._saveOrder(order);
        
        order.status = 'confirmed';
        order.payment.status = 'completed';
        
        return order;
    }
    
    // Cancelar checkout
    cancelCheckout() {
        if (this.reservationId) {
            this.inventory.cancelReservation(this.reservationId);
            this.reservationId = null;
        }
    }
    
    _saveOrder(order) {
        const orders = JSON.parse(localStorage.getItem('orders') || '[]');
        orders.push(order);
        localStorage.setItem('orders', JSON.stringify(orders));
    }
    
    _round(num) {
        return Math.round(num * 100) / 100;
    }
}

// ====== USO ======
const checkout = new CheckoutManager(cart, inventory, couponManager);

checkout.setShippingMethods([
    { id: 'standard', name: 'Estándar (5-7 días)', price: 5.99 },
    { id: 'express', name: 'Express (2-3 días)', price: 12.99 },
    { id: 'overnight', name: 'Siguiente día', price: 24.99 }
]);

// Iniciar checkout
await checkout.initCheckout();

// Calcular totales
const totals = checkout.calculateTotals('express');

// Crear y confirmar orden
const order = await checkout.createOrder(
    { name: 'Juan', email: 'juan@email.com', shippingAddress: 'Calle 123' },
    'express',
    'card'
);

const confirmedOrder = await checkout.confirmOrder(order);
