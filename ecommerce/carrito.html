// ============================================
// CARRITO DE COMPRAS COMPLETO - VANILLA JS
// ============================================

// Estado del carrito (array de productos)
let cart = [];

// Productos de ejemplo (normalmente vendrían de una API)
const products = [
    { id: 1, nombre: 'iPhone 15 Pro', precio: 999.99, imagen: 'iphone.jpg', stock: 10 },
    { id: 2, nombre: 'MacBook Pro', precio: 1999.99, imagen: 'macbook.jpg', stock: 5 },
    { id: 3, nombre: 'AirPods Pro', precio: 249.99, imagen: 'airpods.jpg', stock: 20 },
    { id: 4, nombre: 'Apple Watch', precio: 399.99, imagen: 'watch.jpg', stock: 15 }
];

// ====== INICIALIZAR CARRITO ======
function initCart() {
    // Cargar carrito desde localStorage
    const savedCart = localStorage.getItem('cart');
    if (savedCart) {
        cart = JSON.parse(savedCart);
    }
    
    // Actualizar UI
    updateCartUI();
    updateCartCount();
}

// ====== AGREGAR AL CARRITO ======
function addToCart(productId, quantity = 1) {
    // Buscar el producto
    const product = products.find(p => p.id === productId);
    
    if (!product) {
        console.error('Producto no encontrado');
        return;
    }
    
    // Verificar stock
    if (product.stock < quantity) {
        showNotification('Stock insuficiente', 'error');
        return;
    }
    
    // Verificar si ya existe en el carrito
    const existingItem = cart.find(item => item.id === productId);
    
    if (existingItem) {
        // Actualizar cantidad
        if (existingItem.cantidad + quantity <= product.stock) {
            existingItem.cantidad += quantity;
        } else {
            showNotification('No hay más stock disponible', 'error');
            return;
        }
    } else {
        // Agregar nuevo item
        cart.push({
            id: product.id,
            nombre: product.nombre,
            precio: product.precio,
            imagen: product.imagen,
            cantidad: quantity
        });
    }
    
    // Guardar y actualizar UI
    saveCart();
    updateCartUI();
    updateCartCount();
    showNotification(`${product.nombre} agregado al carrito`, 'success');
}

// ====== REMOVER DEL CARRITO ======
function removeFromCart(productId) {
    cart = cart.filter(item => item.id !== productId);
    saveCart();
    updateCartUI();
    updateCartCount();
}

// ====== ACTUALIZAR CANTIDAD ======
function updateQuantity(productId, newQuantity) {
    const item = cart.find(item => item.id === productId);
    const product = products.find(p => p.id === productId);
    
    if (!item || !product) return;
    
    if (newQuantity <= 0) {
        removeFromCart(productId);
        return;
    }
    
    if (newQuantity > product.stock) {
        showNotification('Stock insuficiente', 'error');
        return;
    }
    
    item.cantidad = newQuantity;
    saveCart();
    updateCartUI();
    updateCartCount();
}

// ====== VACIAR CARRITO ======
function clearCart() {
    cart = [];
    saveCart();
    updateCartUI();
    updateCartCount();
}

// ====== CALCULAR TOTALES ======
function getCartTotals() {
    const subtotal = cart.reduce((sum, item) => {
        return sum + (item.precio * item.cantidad);
    }, 0);
    
    const itemCount = cart.reduce((sum, item) => {
        return sum + item.cantidad;
    }, 0);
    
    const shipping = subtotal > 100 ? 0 : 9.99; // Envío gratis > $100
    const tax = subtotal * 0.16; // 16% IVA
    const total = subtotal + shipping + tax;
    
    return {
        subtotal: subtotal.toFixed(2),
        itemCount,
        shipping: shipping.toFixed(2),
        tax: tax.toFixed(2),
        total: total.toFixed(2)
    };
}

// ====== GUARDAR EN LOCALSTORAGE ======
function saveCart() {
    localStorage.setItem('cart', JSON.stringify(cart));
}

// ====== ACTUALIZAR UI DEL CARRITO ======
function updateCartUI() {
    const cartContainer = document.getElementById('cart-items');
    const totals = getCartTotals();
    
    if (cart.length === 0) {
        cartContainer.innerHTML = `
            <div class="text-center py-5">
                <i class="bi bi-cart-x display-1 text-muted"></i>
                <p class="mt-3 text-muted">Tu carrito está vacío</p>
                <a href="/tienda.html" class="btn btn-primary">Ver productos</a>
            </div>
        `;
        return;
    }
    
    cartContainer.innerHTML = cart.map(item => `
        <div class="cart-item d-flex gap-3" data-id="${item.id}">
            <img src="${item.imagen}" alt="${item.nombre}" 
                 class="rounded" style="width: 80px; height: 80px; object-fit: cover;">
            <div class="flex-grow-1">
                <h6 class="mb-1">${item.nombre}</h6>
                <p class="text-success mb-2">$${item.precio.toFixed(2)}</p>
                <div class="quantity-control">
                    <button class="btn btn-sm btn-outline-secondary btn-minus">
                        <i class="bi bi-dash"></i>
                    </button>
                    <span class="fw-bold">${item.cantidad}</span>
                    <button class="btn btn-sm btn-outline-secondary btn-plus">
                        <i class="bi bi-plus"></i>
                    </button>
                    <button class="btn btn-sm btn-outline-danger ms-auto btn-remove">
                        <i class="bi bi-trash"></i>
                    </button>
                </div>
            </div>
        </div>
    `).join('');
    
    // Actualizar subtotal
    document.getElementById('cart-subtotal').textContent = `$${totals.subtotal}`;
}

// ====== ACTUALIZAR CONTADOR ======
function updateCartCount() {
    const totals = getCartTotals();
    const badge = document.getElementById('cart-count');
    badge.textContent = totals.itemCount;
    badge.style.display = totals.itemCount > 0 ? 'flex' : 'none';
}

// ====== EVENT DELEGATION PARA BOTONES DEL CARRITO ======
document.getElementById('cart-items').addEventListener('click', (e) => {
    const cartItem = e.target.closest('.cart-item');
    if (!cartItem) return;
    
    const productId = parseInt(cartItem.dataset.id);
    const item = cart.find(i => i.id === productId);
    
    if (e.target.closest('.btn-plus')) {
        updateQuantity(productId, item.cantidad + 1);
    }
    
    if (e.target.closest('.btn-minus')) {
        updateQuantity(productId, item.cantidad - 1);
    }
    
    if (e.target.closest('.btn-remove')) {
        removeFromCart(productId);
    }
});

// ====== ABRIR/CERRAR CARRITO SIDEBAR ======
function openCart() {
    document.getElementById('cart-sidebar').classList.add('open');
    document.getElementById('cart-overlay').classList.add('open');
    document.body.style.overflow = 'hidden';
}

function closeCart() {
    document.getElementById('cart-sidebar').classList.remove('open');
    document.getElementById('cart-overlay').classList.remove('open');
    document.body.style.overflow = '';
}

// ====== CHECKOUT ======
async function checkout() {
    if (cart.length === 0) {
        showNotification('El carrito está vacío', 'error');
        return;
    }
    
    const totals = getCartTotals();
    
    // Aquí enviarías la orden a tu backend
    const order = {
        items: cart,
        totals,
        fecha: new Date().toISOString()
    };
    
    try {
        const response = await fetch('/api/orders', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(order)
        });
        
        if (response.ok) {
            clearCart();
            showNotification('¡Orden creada exitosamente!', 'success');
            window.location.href = '/checkout-success.html';
        }
    } catch (error) {
        showNotification('Error al procesar la orden', 'error');
    }
}

// Inicializar al cargar la página
document.addEventListener('DOMContentLoaded', initCart);

4Sistema de Cupones de Descuento 

// ============================================
// SISTEMA DE CUPONES DE DESCUENTO
// ============================================

// Cupones disponibles (normalmente vendrían del backend)
const coupons = {
    'WELCOME10': { tipo: 'porcentaje', valor: 10, minimo: 50 },
    'SAVE20': { tipo: 'porcentaje', valor: 20, minimo: 100 },
    'FLAT15': { tipo: 'fijo', valor: 15, minimo: 75 },
    'FREESHIP': { tipo: 'envio_gratis', valor: 0, minimo: 0 }
};

let appliedCoupon = null;

// ====== APLICAR CUPÓN ======
function applyCoupon(code) {
    const coupon = coupons[code.toUpperCase()];
    const totals = getCartTotals();
    
    if (!coupon) {
        showNotification('Cupón inválido', 'error');
        return false;
    }
    
    if (parseFloat(totals.subtotal) < coupon.minimo) {
        showNotification(
            `Mínimo de compra: $${coupon.minimo}`, 
            'error'
        );
        return false;
    }
    
    appliedCoupon = { code: code.toUpperCase(), ...coupon };
    updateCartUI();
    showNotification('¡Cupón aplicado!', 'success');
    return true;
}

// ====== REMOVER CUPÓN ======
function removeCoupon() {
    appliedCoupon = null;
    updateCartUI();
}

// ====== CALCULAR DESCUENTO ======
function calculateDiscount(subtotal) {
    if (!appliedCoupon) return 0;
    
    switch (appliedCoupon.tipo) {
        case 'porcentaje':
            return subtotal * (appliedCoupon.valor / 100);
        case 'fijo':
            return appliedCoupon.valor;
        default:
            return 0;
    }
}

// ====== TOTALES CON DESCUENTO ======
function getCartTotalsWithDiscount() {
    const baseTotals = getCartTotals();
    const subtotal = parseFloat(baseTotals.subtotal);
    const discount = calculateDiscount(subtotal);
    
    let shipping = subtotal > 100 ? 0 : 9.99;
    if (appliedCoupon?.tipo === 'envio_gratis') {
        shipping = 0;
    }
    
    const subtotalWithDiscount = subtotal - discount;
    const tax = subtotalWithDiscount * 0.16;
    const total = subtotalWithDiscount + shipping + tax;
    
    return {
        ...baseTotals,
        discount: discount.toFixed(2),
        shipping: shipping.toFixed(2),
        tax: tax.toFixed(2),
        total: total.toFixed(2),
        coupon: appliedCoupon
    };
}

// Event listener para el formulario de cupón
document.getElementById('coupon-form')?.addEventListener('submit', (e) => {
    e.preventDefault();
    const code = document.getElementById('coupon-input').value;
    applyCoupon(code);
});

// ============================================
// CARRITO CON JSON SERVER (CRUD COMPLETO)
// ============================================

const API_URL = 'http://localhost:3000';

// ====== OBTENER CARRITO ======
async function fetchCart() {
    const response = await fetch(`${API_URL}/cart`);
    return await response.json();
}

// ====== AGREGAR AL CARRITO ======
async function addToCartAPI(product, quantity = 1) {
    const cart = await fetchCart();
    const existing = cart.find(item => item.productId === product.id);
    
    if (existing) {
        // Actualizar cantidad (PATCH)
        const response = await fetch(`${API_URL}/cart/${existing.id}`, {
            method: 'PATCH',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                cantidad: existing.cantidad + quantity
            })
        });
        return await response.json();
    } else {
        // Agregar nuevo (POST)
        const response = await fetch(`${API_URL}/cart`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                productId: product.id,
                nombre: product.nombre,
                precio: product.precio,
                imagen: product.imagen,
                cantidad: quantity
            })
        });
        return await response.json();
    }
}

// ====== ACTUALIZAR CANTIDAD ======
async function updateQuantityAPI(cartItemId, cantidad) {
    if (cantidad <= 0) {
        return removeFromCartAPI(cartItemId);
    }
    
    const response = await fetch(`${API_URL}/cart/${cartItemId}`, {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ cantidad })
    });
    return await response.json();
}

// ====== REMOVER DEL CARRITO ======
async function removeFromCartAPI(cartItemId) {
    await fetch(`${API_URL}/cart/${cartItemId}`, {
        method: 'DELETE'
    });
}

// ====== CREAR ORDEN ======
async function createOrder(cartItems, customerInfo) {
    const totals = calculateTotals(cartItems);
    
    const order = {
        items: cartItems,
        customer: customerInfo,
        totals,
        status: 'pending',
        fecha: new Date().toISOString()
    };
    
    // Crear orden
    const response = await fetch(`${API_URL}/orders`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(order)
    });
    
    // Vaciar carrito
    for (const item of cartItems) {
        await removeFromCartAPI(item.id);
    }
    
    return await response.json();
}

// ====== OBTENER PRODUCTOS ======
async function fetchProducts() {
    const response = await fetch(`${API_URL}/products`);
    return await response.json();
}
