/**
 * CLASE CHECKOUT - Sistema de Pago Completo
 * Maneja todo el flujo de checkout: envío, pago y confirmación
 */
class Checkout {
    constructor() {
        // Estado del checkout
        this.currentStep = 1;
        this.shippingData = {};
        this.paymentData = {};
        this.cart = this.loadCart();
        this.coupon = null;
        
        // Inicializar
        this.init();
    }
    
    init() {
        // Escuchadores de formularios
        this.setupFormListeners();
        
        // Escuchadores de métodos de pago
        this.setupPaymentMethods();
        
        // Formateo de inputs de tarjeta
        this.setupCardInputs();
        
        // Escuchadores de envío
        this.setupShippingOptions();
        
        // Actualizar resumen
        this.updateOrderSummary();
    }
    
    /**
     * Cargar carrito desde localStorage
     */
    loadCart() {
        const saved = localStorage.getItem('cart');
        return saved ? JSON.parse(saved) : {
            items: [
                { id: 1, name: 'Auriculares Pro Max', price: 199.99, qty: 1 },
                { id: 2, name: 'Funda Premium', price: 24.99, qty: 2 }
            ]
        };
    }
    
    /**
     * Calcular totales
     */
    calculateTotals() {
        const subtotal = this.cart.items.reduce((sum, item) => {
            return sum + (item.price * item.qty);
        }, 0);
        
        const shipping = this.getShippingCost();
        const discount = this.coupon ? subtotal * 0.2 : 0; // 20% descuento
        const taxRate = 0.21; // 21% IVA
        const taxes = (subtotal - discount) * taxRate;
        const total = subtotal - discount + shipping + taxes;
        
        return { subtotal, shipping, discount, taxes, total };
    }
    
    /**
     * Obtener costo de envío
     */
    getShippingCost() {
        const expressInput = document.getElementById('expressShipping');
        return expressInput && expressInput.checked ? 9.99 : 0;
    }
    
    /**
     * Actualizar resumen del pedido
     */
    updateOrderSummary() {
        const totals = this.calculateTotals();
        
        document.getElementById('shippingCost').textContent = 
            totals.shipping === 0 ? 'Gratis' : `$${totals.shipping.toFixed(2)}`;
        
        document.getElementById('orderTotal').textContent = 
            `$${totals.total.toFixed(2)}`;
        
        document.getElementById('totalAmount').textContent = 
            totals.total.toFixed(2);
        
        // Mostrar descuento si hay cupón
        const discountRow = document.getElementById('discountRow');
        if (this.coupon) {
            discountRow.style.display = 'flex';
            discountRow.querySelector('.discount').textContent = 
                `-$${totals.discount.toFixed(2)}`;
        } else {
            discountRow.style.display = 'none';
        }
    }
    
    /**
     * Avanzar al siguiente paso
     */
    nextStep() {
        // Completar paso actual
        document.getElementById(`step-${this.getStepName(this.currentStep)}`)
            .classList.replace('active', 'completed');
        document.getElementById(`line-${this.getStepName(this.currentStep + 1)}`)
            .classList.add('completed');
        
        // Activar siguiente paso
        this.currentStep++;
        document.getElementById(`step-${this.getStepName(this.currentStep)}`)
            .classList.add('active');
        
        // Mostrar/ocultar secciones
        this.updateSections();
    }
    
    /**
     * Procesar pago (simulado)
     * GUARDA EL PEDIDO EN localStorage para que el Admin pueda verlo
     */
    async processPayment() {
        const btn = document.getElementById('completePurchase');
        btn.disabled = true;
        btn.innerHTML = 'Procesando...';
        
        // Simular llamada a API
        await new Promise(resolve => setTimeout(resolve, 2000));
        
        // ============================================
        // GUARDAR PEDIDO EN localStorage
        // El Admin Dashboard lee de 'allOrders'
        // ============================================
        const currentUser = JSON.parse(localStorage.getItem('currentUser')) || {};
        const orderNumber = Date.now().toString().slice(-6);
        
        const newOrder = {
            id: orderNumber,
            numero: orderNumber,
            fecha: new Date().toISOString(),
            cliente: currentUser.nombre || currentUser.name || 'Invitado',
            email: currentUser.email || document.getElementById('email')?.value || 'N/A',
            productos: this.cart.map(item => ({
                id: item.id,
                nombre: item.name || item.nombre,
                precio: item.price || item.precio,
                cantidad: item.quantity || item.cantidad,
                subtotal: (item.price || item.precio) * (item.quantity || item.cantidad)
            })),
            subtotal: this.calculateTotals().subtotal,
            envio: this.calculateTotals().shipping,
            descuento: this.calculateTotals().discount || 0,
            total: this.calculateTotals().total,
            estado: 'pendiente',
            direccion: {
                calle: document.getElementById('address')?.value || '',
                ciudad: document.getElementById('city')?.value || '',
                estado: document.getElementById('state')?.value || '',
                cp: document.getElementById('zip')?.value || ''
            },
            metodoPago: document.querySelector('input[name="paymentMethod"]:checked')?.value || 'tarjeta'
        };
        
        // Obtener pedidos existentes y agregar el nuevo
        const allOrders = JSON.parse(localStorage.getItem('allOrders')) || [];
        allOrders.push(newOrder);
        localStorage.setItem('allOrders', JSON.stringify(allOrders));
        
        console.log('📦 Pedido guardado:', newOrder);
        console.log('📊 Total pedidos:', allOrders.length);
        
        // Mostrar éxito
        this.showSuccessModal();
        
        // Limpiar carrito
        localStorage.removeItem('cart');
    }
    
    /**
     * Mostrar modal de éxito
     */
    showSuccessModal() {
        const modal = new bootstrap.Modal(document.getElementById('successModal'));
        modal.show();
    }
}

// Validación del formulario de envío
function validateShippingForm(form) {
    const fields = {
        firstName: { required: true, minLength: 2 },
        lastName: { required: true, minLength: 2 },
        email: { required: true, pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/ },
        phone: { required: true, pattern: /^\+?[0-9]{9,15}$/ },
        address: { required: true, minLength: 5 },
        city: { required: true, minLength: 2 },
        state: { required: true },
        zip: { required: true, pattern: /^[0-9]{5}$/ }
    };
    
    let isValid = true;
    
    for (const [fieldId, rules] of Object.entries(fields)) {
        const input = form.querySelector(`#${fieldId}`);
        const value = input.value.trim();
        let fieldValid = true;
        
        // Requerido
        if (rules.required && !value) {
            fieldValid = false;
        }
        
        // Longitud mínima
        if (rules.minLength && value.length < rules.minLength) {
            fieldValid = false;
        }
        
        // Pattern (regex)
        if (rules.pattern && !rules.pattern.test(value)) {
            fieldValid = false;
        }
        
        // Actualizar clases
        input.classList.remove('is-valid', 'is-invalid');
        input.classList.add(fieldValid ? 'is-valid' : 'is-invalid');
        
        if (!fieldValid) isValid = false;
    }
    
    return isValid;
}

// Validación de tarjeta de crédito (Algoritmo de Luhn)
function validateCreditCard(number) {
    // Eliminar espacios
    const cleaned = number.replace(/\s/g, '');
    
    // Verificar que solo sean dígitos
    if (!/^\d+$/.test(cleaned)) return false;
    
    // Verificar longitud (13-19 dígitos)
    if (cleaned.length < 13 || cleaned.length > 19) return false;
    
    // Algoritmo de Luhn
    let sum = 0;
    let isEven = false;
    
    for (let i = cleaned.length - 1; i >= 0; i--) {
        let digit = parseInt(cleaned[i]);
        
        if (isEven) {
            digit *= 2;
            if (digit > 9) digit -= 9;
        }
        
        sum += digit;
        isEven = !isEven;
    }
    
    return sum % 10 === 0;
}

// Detectar tipo de tarjeta
function detectCardType(number) {
    const patterns = {
        visa: /^4/,
        mastercard: /^5[1-5]/,
        amex: /^3[47]/,
        discover: /^6(?:011|5)/
    };
    
    const cleaned = number.replace(/\s/g, '');
    
    for (const [type, pattern] of Object.entries(patterns)) {
        if (pattern.test(cleaned)) {
            return type;
        }
    }
    
    return 'unknown';
}

// Validar fecha de expiración
function validateExpiry(expiry) {
    const [month, year] = expiry.split('/');
    
    if (!month || !year) return false;
    
    const m = parseInt(month);
    const y = parseInt('20' + year);
    
    if (m < 1 || m > 12) return false;
    
    const now = new Date();
    const expDate = new Date(y, m);
    
    return expDate > now;
}

// Validar CVV
function validateCVV(cvv, cardType) {
    const length = cardType === 'amex' ? 4 : 3;
    return new RegExp(`^\\d{${length}}$`).test(cvv);
}

// Formatear número de tarjeta (agregar espacios)
function formatCardNumber(input) {
    let value = input.value.replace(/\s/g, '').replace(/\D/g, '');
    
    // Limitar a 16 dígitos (o 15 para AMEX)
    const cardType = detectCardType(value);
    const maxLength = cardType === 'amex' ? 15 : 16;
    value = value.substring(0, maxLength);
    
    // Agregar espacios cada 4 dígitos
    let formatted = '';
    for (let i = 0; i < value.length; i++) {
        if (i > 0 && i % 4 === 0) formatted += ' ';
        formatted += value[i];
    }
    
    input.value = formatted;
    
    // Actualizar icono de tarjeta
    updateCardIcon(cardType);
}

// Formatear fecha de expiración
function formatExpiry(input) {
    let value = input.value.replace(/\D/g, '');
    
    if (value.length >= 2) {
        // Validar mes (01-12)
        let month = parseInt(value.substring(0, 2));
        if (month > 12) month = 12;
        if (month < 1 && value.length >= 2) month = '01';
        
        value = month.toString().padStart(2, '0') + '/' + value.substring(2, 4);
    }
    
    input.value = value;
}

// Actualizar icono según tipo de tarjeta
function updateCardIcon(type) {
    const icon = document.getElementById('cardTypeIcon');
    const icons = {
        visa: 'bi-credit-card-2-front',
        mastercard: 'bi-credit-card-2-back',
        amex: 'bi-credit-card',
        discover: 'bi-credit-card-fill',
        unknown: 'bi-credit-card'
    };
    
    icon.className = `bi ${icons[type]} card-icon`;
}

// Configurar listeners de formateo
function setupCardInputs() {
    const cardNumber = document.getElementById('cardNumber');
    const cardExpiry = document.getElementById('cardExpiry');
    const cardCVV = document.getElementById('cardCVV');
    
    // Formatear mientras escribe
    cardNumber.addEventListener('input', () => formatCardNumber(cardNumber));
    cardExpiry.addEventListener('input', () => formatExpiry(cardExpiry));
    
    // Solo números en CVV
    cardCVV.addEventListener('input', () => {
        cardCVV.value = cardCVV.value.replace(/\D/g, '');
    });
    
    // Validar al perder foco
    cardNumber.addEventListener('blur', () => {
        const isValid = validateCreditCard(cardNumber.value);
        cardNumber.classList.toggle('is-invalid', !isValid);
        cardNumber.classList.toggle('is-valid', isValid);
    });
    
    cardExpiry.addEventListener('blur', () => {
        const isValid = validateExpiry(cardExpiry.value);
        cardExpiry.classList.toggle('is-invalid', !isValid);
        cardExpiry.classList.toggle('is-valid', isValid);
    });
}
