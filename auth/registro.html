// JavaScript Multi-Step Form
let currentStep = 1;
const totalSteps = 3;

function nextStep(step) {
    // Validar paso actual
    if (!validateStep(step)) return;
    
    // Ocultar paso actual
    document.getElementById(`step${step}`)
        .classList.add('d-none');
    
    // Mostrar siguiente
    document.getElementById(`step${step + 1}`)
        .classList.remove('d-none');
    
    // Actualizar indicadores
    updateIndicators(step + 1);
    currentStep = step + 1;
}

function prevStep(step) {
    document.getElementById(`step${step}`)
        .classList.add('d-none');
    document.getElementById(`step${step - 1}`)
        .classList.remove('d-none');
    updateIndicators(step - 1);
    currentStep = step - 1;
}

function updateIndicators(activeStep) {
    for (let i = 1; i <= totalSteps; i++) {
        const indicator = document
            .getElementById(`step${i}-indicator`);
        const line = document
            .getElementById(`line${i - 1}`);
        
        indicator.classList.remove(
            'active', 'completed'
        );
        
        if (i < activeStep) {
            indicator.classList.add('completed');
            indicator.innerHTML = '✓';
            if (line) line.classList.add('completed');
        } else if (i === activeStep) {
            indicator.classList.add('active');
            indicator.innerHTML = i;
        } else {
            indicator.innerHTML = i;
            if (line) line.classList.remove('completed');
        }
    }
}

function validateStep(step) {
    const stepEl = document.getElementById(`step${step}`);
    const inputs = stepEl.querySelectorAll('input[required]');
    let valid = true;
    
    inputs.forEach(input => {
        if (!input.value.trim()) {
            input.classList.add('is-invalid');
            valid = false;
        } else {
            input.classList.remove('is-invalid');
        }
    });
    
    return valid;
}

// JavaScript - OAuth con Google
async function registerWithGoogle() {
    // Usando Google Identity Services
    google.accounts.id.initialize({
        client_id: 'TU_CLIENT_ID.apps.googleusercontent.com',
        callback: handleGoogleResponse
    });
    
    google.accounts.id.prompt();
}

async function handleGoogleResponse(response) {
    // Enviar token al backend
    const result = await fetch('/api/auth/google', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
            token: response.credential 
        })
    });
    
    const data = await result.json();
    
    if (data.success) {
        localStorage.setItem('authToken', data.token);
        window.location.href = '/dashboard';
    }
}

// JavaScript - OAuth con Facebook
async function registerWithFacebook() {
    FB.login(function(response) {
        if (response.authResponse) {
            const accessToken = response.authResponse.accessToken;
            sendToBackend('facebook', accessToken);
        }
    }, { scope: 'email,public_profile' });
}

// JavaScript - OAuth con GitHub
function registerWithGithub() {
    const clientId = 'TU_GITHUB_CLIENT_ID';
    const redirectUri = 'https://tuapp.com/auth/github/callback';
    const scope = 'user:email';
    
    window.location.href = 
        `https://github.com/login/oauth/authorize?client_id=${clientId}&redirect_uri=${redirectUri}&scope=${scope}`;
}

// JavaScript - Avatar Upload con Preview

const avatarInput = document.getElementById('avatarInput');
const avatarPreview = document.getElementById('avatarPreview');

avatarInput.addEventListener('change', function(e) {
    const file = e.target.files[0];
    
    if (!file) return;
    
    // Validar tipo de archivo
    if (!file.type.startsWith('image/')) {
        alert('Por favor selecciona una imagen');
        return;
    }
    
    // Validar tamaño (max 5MB)
    if (file.size > 5 * 1024 * 1024) {
        alert('La imagen no debe superar 5MB');
        return;
    }
    
    // Mostrar preview
    const reader = new FileReader();
    
    reader.onload = function(e) {
        avatarPreview.innerHTML = `
            <img src="${e.target.result}" alt="Avatar">
        `;
    };
    
    reader.readAsDataURL(file);
});

// Subir avatar al servidor
async function uploadAvatar(file) {
    const formData = new FormData();
    formData.append('avatar', file);
    
    try {
        const response = await fetch('/api/upload/avatar', {
            method: 'POST',
            body: formData
            // NO poner Content-Type header con FormData
        });
        
        const data = await response.json();
        return data.avatarUrl;
        
    } catch (error) {
        console.error('Error subiendo avatar:', error);
        throw error;
    }
}

// Registro completo con avatar
async function register(formData) {
    let avatarUrl = null;
    
    // Primero subir avatar si existe
    const avatarFile = document.getElementById('avatarInput').files[0];
    
    if (avatarFile) {
        avatarUrl = await uploadAvatar(avatarFile);
    }
    
    // Luego crear usuario
    const response = await fetch('/api/auth/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
            ...formData,
            avatar: avatarUrl
        })
    });
    
    return await response.json();
}

// JavaScript - Password Strength Meter

const passwordInput = document.getElementById('passwordStrength');
const strengthBar = document.getElementById('strengthBar');
const strengthText = document.getElementById('strengthText');

passwordInput.addEventListener('input', function() {
    const password = this.value;
    const strength = calculateStrength(password);
    
    updateStrengthUI(strength);
    updateRequirements(password);
});

function calculateStrength(password) {
    let strength = 0;
    
    // Longitud
    if (password.length >= 8) strength += 1;
    if (password.length >= 12) strength += 1;
    
    // Mayúsculas
    if (/[A-Z]/.test(password)) strength += 1;
    
    // Minúsculas
    if (/[a-z]/.test(password)) strength += 1;
    
    // Números
    if (/[0-9]/.test(password)) strength += 1;
    
    // Caracteres especiales
    if (/[!@#$%^&*(),.?":{}|<>]/.test(password)) strength += 1;
    
    return strength;
}

function updateStrengthUI(strength) {
    const levels = [
        { class: '', text: 'Ingresa una contraseña', color: '' },
        { class: 'strength-weak', text: 'Débil', color: '#ff4444' },
        { class: 'strength-fair', text: 'Regular', color: '#ffaa00' },
        { class: 'strength-good', text: 'Buena', color: '#00cc44' },
        { class: 'strength-strong', text: 'Muy fuerte', color: '#00ff88' }
    ];
    
    const level = Math.min(Math.floor(strength / 1.5), 4);
    
    strengthBar.className = 'password-strength-bar ' + levels[level].class;
    strengthText.textContent = levels[level].text;
    strengthText.style.color = levels[level].color;
}

function updateRequirements(password) {
    const requirements = {
        'req-length': password.length >= 8,
        'req-upper': /[A-Z]/.test(password),
        'req-lower': /[a-z]/.test(password),
        'req-number': /[0-9]/.test(password),
        'req-special': /[!@#$%^&*(),.?":{}|<>]/.test(password)
    };
    
    Object.entries(requirements).forEach(([id, met]) => {
        const el = document.getElementById(id);
        const icon = el.querySelector('i');
        
        if (met) {
            icon.className = 'bi bi-check-lg text-success';
            el.style.color = '#00ff88';
        } else {
            icon.className = 'bi bi-x text-danger';
            el.style.color = '#888';
        }
    });
}

// Toggle password visibility
function togglePassword(inputId) {
    const input = document.getElementById(inputId);
    input.type = input.type === 'password' ? 'text' : 'password';
}

class FormValidator {
    constructor(form) {
        this.form = form;
        this.errors = {};
        this.rules = {};
    }
    
    // Agregar reglas de validación
    addRule(fieldName, rules) {
        this.rules[fieldName] = rules;
        return this;
    }
    
    // Validar un campo
    validateField(fieldName, value) {
        const rules = this.rules[fieldName];
        const errors = [];
        
        if (!rules) return [];
        
        rules.forEach(rule => {
            if (rule.required && !value.trim()) {
                errors.push(rule.message || 'Este campo es requerido');
            }
            
            if (rule.minLength && value.length < rule.minLength) {
                errors.push(rule.message || `Mínimo ${rule.minLength} caracteres`);
            }
            
            if (rule.maxLength && value.length > rule.maxLength) {
                errors.push(rule.message || `Máximo ${rule.maxLength} caracteres`);
            }
            
            if (rule.pattern && !rule.pattern.test(value)) {
                errors.push(rule.message || 'Formato inválido');
            }
            
            if (rule.match) {
                const matchValue = this.form.querySelector(`[name="${rule.match}"]`)?.value;
                if (value !== matchValue) {
                    errors.push(rule.message || 'Los campos no coinciden');
                }
            }
            
            if (rule.custom && !rule.custom(value)) {
                errors.push(rule.message || 'Valor inválido');
            }
        });
        
        return errors;
    }
    
    // Validar todo el formulario
    validate() {
        this.errors = {};
        let isValid = true;
        
        Object.keys(this.rules).forEach(fieldName => {
            const input = this.form.querySelector(`[name="${fieldName}"]`);
            if (!input) return;
            
            const fieldErrors = this.validateField(fieldName, input.value);
            
            if (fieldErrors.length > 0) {
                this.errors[fieldName] = fieldErrors;
                isValid = false;
                this.showError(input, fieldErrors[0]);
            } else {
                this.showSuccess(input);
            }
        });
        
        return isValid;
    }
    
    showError(input, message) {
        input.classList.remove('is-valid');
        input.classList.add('is-invalid');
        
        let feedback = input.parentElement.querySelector('.invalid-feedback');
        if (!feedback) {
            feedback = document.createElement('div');
            feedback.className = 'invalid-feedback';
            input.parentElement.appendChild(feedback);
        }
        feedback.textContent = message;
    }
    
    showSuccess(input) {
        input.classList.remove('is-invalid');
        input.classList.add('is-valid');
    }
}

// ====== USO ======
const form = document.getElementById('registerForm');
const validator = new FormValidator(form);

validator
    .addRule('nombre', [
        { required: true, message: 'El nombre es requerido' },
        { minLength: 2, message: 'Mínimo 2 caracteres' }
    ])
    .addRule('email', [
        { required: true, message: 'El email es requerido' },
        { pattern: /^[^\s@]+@[^\s@]+\.[^\s@]+$/, message: 'Email inválido' }
    ])
    .addRule('password', [
        { required: true },
        { minLength: 8, message: 'Mínimo 8 caracteres' },
        { pattern: /[A-Z]/, message: 'Debe contener una mayúscula' },
        { pattern: /[0-9]/, message: 'Debe contener un número' }
    ])
    .addRule('confirmPassword', [
        { required: true },
        { match: 'password', message: 'Las contraseñas no coinciden' }
    ]);

form.addEventListener('submit', async (e) => {
    e.preventDefault();
    
    if (validator.validate()) {
        const formData = new FormData(form);
        await submitRegistration(Object.fromEntries(formData));
    }
});
