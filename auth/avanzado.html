1. JWT Authentication Manager

Sistema completo de autenticación con JWT y refresh tokens.

Nota de Seguridad: En producción, los tokens deben manejarse en el servidor. Este código es para entender el concepto. 
/**
 * JWT AUTHENTICATION MANAGER
 * Manejo completo de tokens JWT con refresh
 */

class JWTAuthManager {
    constructor(options = {}) {
        this.apiUrl = options.apiUrl || '/api/auth';
        this.tokenKey = options.tokenKey || 'access_token';
        this.refreshKey = options.refreshKey || 'refresh_token';
        this.userKey = options.userKey || 'auth_user';
        this.listeners = [];
        this.refreshPromise = null;
    }
    
    // ====== DECODIFICAR JWT (sin verificar firma) ======
    decodeToken(token) {
        try {
            const payload = token.split('.')[1];
            return JSON.parse(atob(payload));
        } catch (e) {
            return null;
        }
    }
    
    // Verificar si el token está expirado
    isTokenExpired(token) {
        const decoded = this.decodeToken(token);
        if (!decoded || !decoded.exp) return true;
        
        // Expirado si faltan menos de 30 segundos
        return decoded.exp * 1000 < Date.now() + 30000;
    }
    
    // ====== STORAGE ======
    setTokens(accessToken, refreshToken) {
        localStorage.setItem(this.tokenKey, accessToken);
        if (refreshToken) {
            localStorage.setItem(this.refreshKey, refreshToken);
        }
    }
    
    getAccessToken() {
        return localStorage.getItem(this.tokenKey);
    }
    
    getRefreshToken() {
        return localStorage.getItem(this.refreshKey);
    }
    
    clearTokens() {
        localStorage.removeItem(this.tokenKey);
        localStorage.removeItem(this.refreshKey);
        localStorage.removeItem(this.userKey);
    }
    
    // ====== LOGIN ======
    async login(email, password) {
        const response = await fetch(`${this.apiUrl}/login`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, password })
        });
        
        if (!response.ok) {
            const error = await response.json();
            throw new Error(error.message || 'Error de autenticación');
        }
        
        const data = await response.json();
        
        this.setTokens(data.accessToken, data.refreshToken);
        localStorage.setItem(this.userKey, JSON.stringify(data.user));
        
        this._notifyListeners('login', data.user);
        
        return data;
    }
    
    // ====== REFRESH TOKEN ======
    async refreshAccessToken() {
        // Evitar múltiples refresh simultáneos
        if (this.refreshPromise) {
            return this.refreshPromise;
        }
        
        const refreshToken = this.getRefreshToken();
        
        if (!refreshToken) {
            throw new Error('No refresh token available');
        }
        
        this.refreshPromise = (async () => {
            try {
                const response = await fetch(`${this.apiUrl}/refresh`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ refreshToken })
                });
                
                if (!response.ok) {
                    this.logout();
                    throw new Error('Session expired');
                }
                
                const data = await response.json();
                this.setTokens(data.accessToken, data.refreshToken);
                
                return data.accessToken;
            } finally {
                this.refreshPromise = null;
            }
        })();
        
        return this.refreshPromise;
    }
    
    // ====== OBTENER TOKEN VÁLIDO ======
    async getValidToken() {
        let token = this.getAccessToken();
        
        if (!token || this.isTokenExpired(token)) {
            token = await this.refreshAccessToken();
        }
        
        return token;
    }
    
    // ====== LOGOUT ======
    logout() {
        this.clearTokens();
        this._notifyListeners('logout');
    }
    
    // ====== ESTADO ======
    isAuthenticated() {
        const token = this.getAccessToken();
        return token && !this.isTokenExpired(token);
    }
    
    getCurrentUser() {
        try {
            return JSON.parse(localStorage.getItem(this.userKey));
        } catch {
            return null;
        }
    }
    
    // ====== LISTENERS ======
    onAuthChange(callback) {
        this.listeners.push(callback);
        return () => {
            this.listeners = this.listeners.filter(cb => cb !== callback);
        };
    }
    
    _notifyListeners(event, data) {
        this.listeners.forEach(cb => cb(event, data));
    }
}

// ====== USO ======
const auth = new JWTAuthManager({ apiUrl: 'https://api.ejemplo.com/auth' });

// Escuchar cambios de autenticación
auth.onAuthChange((event, data) => {
    if (event === 'login') {
        console.log('Usuario logueado:', data);
    } else if (event === 'logout') {
        window.location.href = '/login';
    }
});

// Login
await auth.login('user@email.com', 'password123');

2. Sistema de Roles y Permisos (RBAC)

Control de acceso basado en roles y permisos.

/**
 * ROLE-BASED ACCESS CONTROL (RBAC)
 */

class RBACManager {
    constructor() {
        // Definir permisos por rol
        this.roles = {
            'admin': {
                permissions: ['*'], // Todo
                inherits: []
            },
            'manager': {
                permissions: [
                    'users:read', 'users:create', 'users:update',
                    'products:*',
                    'orders:*',
                    'reports:read'
                ],
                inherits: ['user']
            },
            'editor': {
                permissions: [
                    'products:read', 'products:create', 'products:update',
                    'content:*'
                ],
                inherits: ['user']
            },
            'user': {
                permissions: [
                    'profile:read', 'profile:update',
                    'orders:read', 'orders:create'
                ],
                inherits: []
            }
        };
    }
    
    // Obtener todos los permisos de un rol (incluyendo heredados)
    getRolePermissions(roleName) {
        const role = this.roles[roleName];
        if (!role) return [];
        
        let permissions = [...role.permissions];
        
        // Agregar permisos heredados
        for (const inheritedRole of role.inherits) {
            permissions = [...permissions, ...this.getRolePermissions(inheritedRole)];
        }
        
        return [...new Set(permissions)]; // Eliminar duplicados
    }
    
    // Verificar si un permiso coincide con un patrón
    _matchPermission(required, userPermission) {
        // Permiso wildcard total
        if (userPermission === '*') return true;
        
        // Permiso exacto
        if (required === userPermission) return true;
        
        // Wildcard por recurso (products:* coincide con products:read)
        const [resource, action] = required.split(':');
        const [permResource, permAction] = userPermission.split(':');
        
        if (resource === permResource && permAction === '*') {
            return true;
        }
        
        return false;
    }
    
    // Verificar si el usuario tiene un permiso
    hasPermission(user, permission) {
        if (!user || !user.role) return false;
        
        const userPermissions = this.getRolePermissions(user.role);
        
        return userPermissions.some(p => this._matchPermission(permission, p));
    }
    
    // Verificar múltiples permisos (ANY)
    hasAnyPermission(user, permissions) {
        return permissions.some(p => this.hasPermission(user, p));
    }
    
    // Verificar múltiples permisos (ALL)
    hasAllPermissions(user, permissions) {
        return permissions.every(p => this.hasPermission(user, p));
    }
    
    // Crear middleware de autorización
    requirePermission(permission) {
        return (user) => {
            if (!this.hasPermission(user, permission)) {
                throw new Error(`Permiso denegado: ${permission}`);
            }
            return true;
        };
    }
}

// ====== USO ======
const rbac = new RBACManager();

const user = { id: 1, name: 'Juan', role: 'manager' };

// Verificar permisos
console.log(rbac.hasPermission(user, 'products:create')); // true
console.log(rbac.hasPermission(user, 'users:delete'));   // false

// Proteger acciones
function deleteProduct(productId) {
    rbac.requirePermission('products:delete')(currentUser);
    // ... eliminar producto
}

// En componentes/vistas
if (rbac.hasPermission(user, 'admin:access')) {
    showAdminPanel();
}

3. Protección de Rutas

Sistema de rutas protegidas con redirección.

/**
 * ROUTER CON PROTECCIÓN DE RUTAS
 */

class ProtectedRouter {
    constructor(authManager, rbacManager) {
        this.auth = authManager;
        this.rbac = rbacManager;
        this.routes = [];
        this.loginPath = '/login';
        this.unauthorizedPath = '/403';
    }
    
    // Definir ruta
    addRoute(path, config) {
        this.routes.push({ path, ...config });
        return this;
    }
    
    // Verificar acceso a ruta
    canAccess(path) {
        const route = this.findRoute(path);
        if (!route) return { allowed: true };
        
        // Ruta pública
        if (route.public) {
            return { allowed: true };
        }
        
        // Verificar autenticación
        if (!this.auth.isAuthenticated()) {
            return { 
                allowed: false, 
                redirect: this.loginPath,
                reason: 'not_authenticated'
            };
        }
        
        // Verificar permisos
        if (route.permission) {
            const user = this.auth.getCurrentUser();
            if (!this.rbac.hasPermission(user, route.permission)) {
                return { 
                    allowed: false, 
                    redirect: this.unauthorizedPath,
                    reason: 'no_permission'
                };
            }
        }
        
        // Verificar roles
        if (route.roles) {
            const user = this.auth.getCurrentUser();
            if (!route.roles.includes(user?.role)) {
                return { 
                    allowed: false, 
                    redirect: this.unauthorizedPath,
                    reason: 'invalid_role'
                };
            }
        }
        
        return { allowed: true };
    }
    
    // Navegar a ruta
    navigate(path) {
        const access = this.canAccess(path);
        
        if (!access.allowed) {
            // Guardar URL original para redirigir después del login
            if (access.reason === 'not_authenticated') {
                sessionStorage.setItem('redirectAfterLogin', path);
            }
            
            console.log(`🚫 Acceso denegado: ${access.reason}`);
            window.location.href = access.redirect;
            return false;
        }
        
        window.location.href = path;
        return true;
    }
    
    findRoute(path) {
        return this.routes.find(r => {
            if (r.path instanceof RegExp) {
                return r.path.test(path);
            }
            return r.path === path || path.startsWith(r.path + '/');
        });
    }
    
    // Verificar ruta actual al cargar página
    checkCurrentRoute() {
        const currentPath = window.location.pathname;
        const access = this.canAccess(currentPath);
        
        if (!access.allowed) {
            window.location.href = access.redirect;
        }
    }
}

// ====== USO ======
const router = new ProtectedRouter(auth, rbac);

// Definir rutas
router
    .addRoute('/login', { public: true })
    .addRoute('/register', { public: true })
    .addRoute('/dashboard', { }) // Solo requiere autenticación
    .addRoute('/admin', { roles: ['admin'] })
    .addRoute('/products/create', { permission: 'products:create' })
    .addRoute('/users', { permission: 'users:read' });

// Verificar al cargar página
document.addEventListener('DOMContentLoaded', () => {
    router.checkCurrentRoute();
});

// Usar en links
function handleNavigation(e, path) {
    e.preventDefault();
    router.navigate(path);
}

4. Gestión de Sesión

Control de tiempo de sesión, inactividad y sincronización entre pestañas.

/**
 * SESSION MANAGER
 * Control de sesión con timeout y sincronización
 */

class SessionManager {
    constructor(authManager, options = {}) {
        this.auth = authManager;
        this.inactivityTimeout = options.inactivityTimeout || 30 * 60 * 1000; // 30 min
        this.warningTime = options.warningTime || 5 * 60 * 1000; // 5 min antes
        this.lastActivity = Date.now();
        this.timeoutId = null;
        this.warningShown = false;
        this.onWarning = options.onWarning || null;
        this.onTimeout = options.onTimeout || null;
        
        this._initActivityTracking();
        this._initTabSync();
    }
    
    // Rastrear actividad del usuario
    _initActivityTracking() {
        const events = ['mousedown', 'keydown', 'scroll', 'touchstart'];
        
        events.forEach(event => {
            document.addEventListener(event, () => this._updateActivity(), { passive: true });
        });
        
        this._startTimer();
    }
    
    _updateActivity() {
        this.lastActivity = Date.now();
        this.warningShown = false;
        
        // Sincronizar con otras pestañas
        localStorage.setItem('lastActivity', this.lastActivity.toString());
        
        this._startTimer();
    }
    
    _startTimer() {
        if (this.timeoutId) {
            clearTimeout(this.timeoutId);
        }
        
        this.timeoutId = setTimeout(() => {
            this._checkInactivity();
        }, 1000); // Verificar cada segundo
    }
    
    _checkInactivity() {
        const now = Date.now();
        const elapsed = now - this.lastActivity;
        const remaining = this.inactivityTimeout - elapsed;
        
        if (remaining <= 0) {
            // Timeout - cerrar sesión
            if (this.onTimeout) {
                this.onTimeout();
            }
            this.auth.logout();
            return;
        }
        
        if (remaining <= this.warningTime && !this.warningShown) {
            // Mostrar advertencia
            this.warningShown = true;
            if (this.onWarning) {
                this.onWarning(Math.ceil(remaining / 1000));
            }
        }
        
        this._startTimer();
    }
    
    // Sincronizar entre pestañas
    _initTabSync() {
        window.addEventListener('storage', (e) => {
            if (e.key === 'lastActivity') {
                this.lastActivity = parseInt(e.newValue);
                this.warningShown = false;
            }
            
            if (e.key === 'logout') {
                // Otra pestaña cerró sesión
                window.location.href = '/login';
            }
        });
    }
    
    // Extender sesión manualmente
    extendSession() {
        this._updateActivity();
    }
    
    // Obtener tiempo restante
    getRemainingTime() {
        const elapsed = Date.now() - this.lastActivity;
        return Math.max(0, this.inactivityTimeout - elapsed);
    }
    
    destroy() {
        if (this.timeoutId) {
            clearTimeout(this.timeoutId);
        }
    }
}

// ====== USO ======
const session = new SessionManager(auth, {
    inactivityTimeout: 30 * 60 * 1000, // 30 minutos
    warningTime: 2 * 60 * 1000, // Avisar 2 min antes
    
    onWarning: (secondsLeft) => {
        // Mostrar modal de advertencia
        showSessionWarningModal({
            message: `Tu sesión expirará en ${secondsLeft} segundos`,
            onExtend: () => session.extendSession()
        });
    },
    
    onTimeout: () => {
        showNotification('Sesión cerrada por inactividad');
    }
});
