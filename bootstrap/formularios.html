Validación de Formularios

Validación nativa del navegador y personalizada

¡Se ve bien!
Por favor ingresa tu nombre.
@
Email inválido.
Ingresa tu ciudad.
Selecciona un estado.
CP inválido (5 dígitos).
Debes aceptar los términos.
<!-- HTML del Formulario con Validación -->
<form class="needs-validation" novalidate id="myForm">
    <!-- Campo con valid-feedback e invalid-feedback -->
    <div class="mb-3">
        <label for="name" class="form-label">Nombre</label>
        <input type="text" class="form-control" id="name" required>
        <div class="valid-feedback">¡Se ve bien!</div>
        <div class="invalid-feedback">Por favor ingresa tu nombre.</div>
    </div>
    
    <!-- Input Group con Validación -->
    <div class="input-group has-validation">
        <span class="input-group-text">@</span>
        <input type="email" class="form-control" required>
        <div class="invalid-feedback">Email inválido.</div>
    </div>
    
    <!-- Validación con Pattern (Regex) -->
    <input type="text" class="form-control" required pattern="[0-9]{5}">
    
    <button type="submit">Enviar</button>
</form>

<!-- JavaScript para Activar Validación -->
<script>
(function() {
    'use strict';
    
    // Seleccionar todos los forms con .needs-validation
    const forms = document.querySelectorAll('.needs-validation');
    
    forms.forEach(form => {
        form.addEventListener('submit', function(event) {
            // Si el form no es válido
            if (!form.checkValidity()) {
                event.preventDefault();
                event.stopPropagation();
            }
            
            // Agregar clase para mostrar estados de validación
            form.classList.add('was-validated');
        });
    });
})();
</script>

<!-- Formulario de Contacto Completo -->
<form class="needs-validation" novalidate id="contactForm">
    <div class="row g-3">
        <!-- Nombre -->
        <div class="col-md-6">
            <div class="form-floating">
                <input type="text" class="form-control" id="name" 
                       placeholder="Nombre" required>
                <label><i class="bi bi-person me-2"></i>Nombre</label>
                <div class="invalid-feedback">Ingresa tu nombre.</div>
            </div>
        </div>

                <input type="email" class="form-control" id="email" 
                       placeholder="Email" required>

                <div class="invalid-feedback">Email inválido.</div>
            </div>
        </div>
        
        <!-- Select con Floating -->
        <div class="col-md-6">
            <div class="form-floating">
                <select class="form-select" id="subject" required>
                    <option selected disabled value="">Seleccionar...</option>
                    <option>Consulta General</option>
                    <option>Soporte</option>
                </select>
                <label>Asunto</label>
            </div>
        </div>

            <div class="form-floating">
                <textarea class="form-control" id="message" 
                          placeholder="Mensaje" style="height: 150px" required></textarea>

        </div>

            <div class="form-check">
                <input class="form-check-input" type="checkbox" id="terms" required>
                <label class="form-check-label">Acepto los términos</label>
            </div>
        </div>

                <label class="form-check-label">Suscribirme</label>
