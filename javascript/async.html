1Síncrono vs Asíncrono

Síncrono

Las operaciones se ejecutan una tras otra, bloqueando el hilo principal.

Tarea 1 (completa)
Tarea 2 (completa)
Tarea 3 (completa)
Asíncrono

Las operaciones pueden ejecutarse sin bloquear. El código continúa.

Tarea 1 (inicia)
Tarea 2 (inicia mientras 1 espera)
Tarea 1 (completa)
// ========================================
// CÓDIGO SÍNCRONO
// ========================================

console.log('1. Inicio');
console.log('2. Proceso');
console.log('3. Fin');
// Salida: 1, 2, 3 (en orden)

// ========================================
// CÓDIGO ASÍNCRONO
// ========================================

console.log('1. Inicio');

setTimeout(() => {
    console.log('2. Proceso (después de 2 segundos)');
}, 2000);

console.log('3. Fin');

// Salida: 1, 3, 2
// El código NO espera al setTimeout

// ========================================
// OPERACIONES ASÍNCRONAS COMUNES
// ========================================

// - setTimeout, setInterval
// - Peticiones HTTP (fetch, XMLHttpRequest)
// - Lectura/escritura de archivos
// - Acceso a bases de datos
// - Eventos del DOM
// - Geolocalización

// El Event Loop maneja las operaciones asíncronas
console.log('Sync 1');

setTimeout(() => console.log('Timeout'), 0); // Aunque sea 0ms

Promise.resolve().then(() => console.log('Promise'));

console.log('Sync 2');

// Salida: Sync 1, Sync 2, Promise, Timeout
// Las microtareas (Promises) tienen prioridad sobre macrotareas (setTimeout)

2Callbacks (El Problema)

// ========================================
// CALLBACKS - LA FORMA ANTIGUA
// ========================================

// Simulamos operaciones asíncronas
function obtenerUsuario(id, callback) {
    setTimeout(() => {
        const usuario = { id, nombre: 'Ana', email: 'ana@email.com' };
        callback(null, usuario);
    }, 1000);
}

function obtenerPedidos(usuarioId, callback) {
    setTimeout(() => {
        const pedidos = [{ id: 1, producto: 'Laptop' }, { id: 2, producto: 'Mouse' }];
        callback(null, pedidos);
    }, 1000);
}

function obtenerDetalles(pedidoId, callback) {
    setTimeout(() => {
        const detalles = { precio: 1000, enviado: true };
        callback(null, detalles);
    }, 1000);
}

// CALLBACK HELL (Pirámide de la perdición)
obtenerUsuario(1, (error, usuario) => {
    if (error) {
        console.error('Error obteniendo usuario');
        return;
    }
    
    console.log('Usuario:', usuario.nombre);
    
    obtenerPedidos(usuario.id, (error, pedidos) => {
        if (error) {
            console.error('Error obteniendo pedidos');
            return;
        }
        
        console.log('Pedidos:', pedidos.length);
        
        obtenerDetalles(pedidos[0].id, (error, detalles) => {
            if (error) {
                console.error('Error obteniendo detalles');
                return;
            }
            
            console.log('Detalles:', detalles);
            
            // Y podría seguir...
            // Esto se vuelve IMPOSIBLE de mantener
        });
    });
});

// Problemas del Callback Hell:
// 1. Difícil de leer y entender
// 2. Manejo de errores repetitivo
// 3. Difícil de depurar
// 4. No se puede usar try/catch

3Promesas - Conceptos Básicos

Una Promise representa un valor que puede estar disponible ahora, en el futuro, o nunca.

Estados de una Promise
  • Pending: Estado inicial, ni cumplida ni rechazada
  • Fulfilled: La operación se completó exitosamente
  • Rejected: La operación falló
// ========================================
// CREAR PROMESAS
// ========================================

// Sintaxis básica
const miPromesa = new Promise((resolve, reject) => {
    // Código asíncrono aquí
    const exito = true;
    
    if (exito) {
        resolve('¡Operación exitosa!'); // Cumplida
    } else {
        reject('Algo salió mal'); // Rechazada
    }
});

// Consumir la promesa
miPromesa
    .then(resultado => {
        console.log('Éxito:', resultado);
    })
    .catch(error => {
        console.error('Error:', error);
    })
    .finally(() => {
        console.log('Siempre se ejecuta');
    });

// Ejemplo práctico: Simular petición
function obtenerUsuario(id) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            if (id > 0) {
                resolve({ id, nombre: 'Ana', email: 'ana@email.com' });
            } else {
                reject(new Error('ID inválido'));
            }
        }, 1000);
    });
}

// Usar la función
obtenerUsuario(1)
    .then(usuario => console.log(usuario))
    .catch(error => console.error(error.message));

// Promesas ya resueltas/rechazadas
const resuelta = Promise.resolve('Valor inmediato');
const rechazada = Promise.reject('Error inmediato');

resuelta.then(v => console.log(v)); // 'Valor inmediato'
rechazada.catch(e => console.log(e)); // 'Error inmediato'

// Convertir callback a promesa
function esperar(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}

esperar(2000).then(() => console.log('2 segundos después'));

4Métodos Estáticos de Promise

// ========================================
// MÉTODOS ESTÁTICOS DE PROMISE
// ========================================

// Simulamos peticiones con diferentes tiempos
const p1 = new Promise(r => setTimeout(() => r('P1: 1s'), 1000));
const p2 = new Promise(r => setTimeout(() => r('P2: 2s'), 2000));
const p3 = new Promise((_, rej) => setTimeout(() => rej('P3: Error'), 1500));

// ========================================
// Promise.all() - Espera TODAS
// ========================================
// Resuelve cuando TODAS se resuelven
// Rechaza al PRIMER error

Promise.all([p1, p2])
    .then(resultados => {
        console.log(resultados); // ['P1: 1s', 'P2: 2s']
    });

// Con error
Promise.all([p1, p2, p3])
    .catch(error => {
        console.log('Falló:', error); // 'Falló: P3: Error'
    });

// ========================================
// Promise.allSettled() - Espera todas sin importar resultado
// ========================================
// Siempre resuelve con array de resultados

Promise.allSettled([p1, p2, p3])
    .then(resultados => {
        console.log(resultados);
        /*
        [
            { status: 'fulfilled', value: 'P1: 1s' },
            { status: 'fulfilled', value: 'P2: 2s' },
            { status: 'rejected', reason: 'P3: Error' }
        ]
        */
    });

// ========================================
// Promise.race() - La primera que termine
// ========================================
// Resuelve/rechaza con la PRIMERA que termine

Promise.race([p1, p2])
    .then(primero => {
        console.log(primero); // 'P1: 1s' (terminó primero)
    });

// Ejemplo: Timeout para peticiones
function fetchConTimeout(url, timeout = 5000) {
    const fetchPromise = fetch(url);
    const timeoutPromise = new Promise((_, reject) => {
        setTimeout(() => reject(new Error('Timeout!')), timeout);
    });
    
    return Promise.race([fetchPromise, timeoutPromise]);
}

// ========================================
// Promise.any() - La primera exitosa
// ========================================
// Resuelve con la primera que SE RESUELVA (ignora rechazos)
// Solo rechaza si TODAS fallan

Promise.any([p3, p1, p2])
    .then(primeroExitoso => {
        console.log(primeroExitoso); // 'P1: 1s' (primera exitosa)
    });

// ========================================
// Promise.resolve() y Promise.reject()
// ========================================

const inmediata = Promise.resolve(42);
inmediata.then(v => console.log(v)); // 42

const error = Promise.reject(new Error('Fallo'));
error.catch(e => console.log(e.message)); // 'Fallo'

5Encadenamiento de Promesas

// ========================================
// ENCADENAMIENTO DE PROMESAS
// ========================================

// Cada .then() retorna una nueva promesa

function obtenerUsuario(id) {
    return new Promise(resolve => {
        setTimeout(() => resolve({ id, nombre: 'Ana' }), 500);
    });
}

function obtenerPedidos(usuarioId) {
    return new Promise(resolve => {
        setTimeout(() => resolve([
            { id: 1, producto: 'Laptop' },
            { id: 2, producto: 'Mouse' }
        ]), 500);
    });
}

function obtenerDetalles(pedidoId) {
    return new Promise(resolve => {
        setTimeout(() => resolve({ precio: 1000, enviado: true }), 500);
    });
}

// Encadenamiento elegante (Adiós Callback Hell!)
obtenerUsuario(1)
    .then(usuario => {
        console.log('Usuario:', usuario.nombre);
        return obtenerPedidos(usuario.id);
    })
    .then(pedidos => {
        console.log('Pedidos:', pedidos.length);
        return obtenerDetalles(pedidos[0].id);
    })
    .then(detalles => {
        console.log('Detalles:', detalles);
    })
    .catch(error => {
        console.error('Error en algún paso:', error);
    })
    .finally(() => {
        console.log('Proceso completado');
    });

// Retornar valores en .then()
Promise.resolve(1)
    .then(x => x + 1)     // 2
    .then(x => x * 2)     // 4
    .then(x => x + 10)    // 14
    .then(x => console.log(x)); // 14

// Retornar promesa en .then()
Promise.resolve('Inicio')
    .then(msg => {
        console.log(msg);
        return new Promise(r => setTimeout(() => r('Después de 1s'), 1000));
    })
    .then(msg => console.log(msg));

6Async/Await (ES2017)

Sintaxis más limpia para trabajar con promesas. Hace que el código asíncrono parezca síncrono.

// ========================================
// ASYNC/AWAIT - SINTAXIS MODERNA
// ========================================

// Función async - Siempre retorna una promesa
async function miFuncion() {
    return 'Hola';
}

miFuncion().then(v => console.log(v)); // 'Hola'

// await - Espera a que la promesa se resuelva
async function obtenerDatos() {
    const usuario = await obtenerUsuario(1);
    console.log(usuario); // Espera hasta que resuelva
    
    const pedidos = await obtenerPedidos(usuario.id);
    console.log(pedidos);
    
    const detalles = await obtenerDetalles(pedidos[0].id);
    console.log(detalles);
    
    return { usuario, pedidos, detalles };
}

obtenerDatos().then(data => console.log('Todo:', data));

// Arrow function async
const fetchData = async () => {
    const response = await fetch('/api/data');
    const data = await response.json();
    return data;
};

// Método async en objeto
const api = {
    async getUser(id) {
        const response = await fetch(`/api/users/${id}`);
        return response.json();
    }
};

// Método async en clase
class ApiService {
    async fetchUsers() {
        const response = await fetch('/api/users');
        return response.json();
    }
}

// IIFE async
(async () => {
    const resultado = await Promise.resolve('Inmediato');
    console.log(resultado);
})();

// await solo funciona dentro de async functions
// En módulos ES, puedes usar top-level await

7Manejo de Errores

// ========================================
// MANEJO DE ERRORES CON ASYNC/AWAIT
// ========================================

// 1. try/catch (Recomendado)
async function fetchData() {
    try {
        const response = await fetch('/api/data');
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const data = await response.json();
        return data;
        
    } catch (error) {
        console.error('Error fetching data:', error.message);
        // Puedes relanzar, retornar default, o manejar
        throw error; // Relanzar
        // return null; // Retornar default
    } finally {
        console.log('Petición completada');
    }
}

// 2. Múltiples try/catch
async function procesoComplejo() {
    let usuario, pedidos;
    
    try {
        usuario = await obtenerUsuario(1);
    } catch (error) {
        console.error('Error obteniendo usuario');
        return;
    }
    
    try {
        pedidos = await obtenerPedidos(usuario.id);
    } catch (error) {
        console.error('Error obteniendo pedidos');
        pedidos = []; // Default
    }
    
    return { usuario, pedidos };
}

// 3. Wrapper para evitar try/catch repetitivo
async function safeAsync(promesa) {
    try {
        const data = await promesa;
        return [data, null];
    } catch (error) {
        return [null, error];
    }
}

// Uso del wrapper
async function usarWrapper() {
    const [usuario, error] = await safeAsync(obtenerUsuario(1));
    
    if (error) {
        console.error('Error:', error);
        return;
    }
    
    console.log('Usuario:', usuario);
}

// 4. Usar .catch() con async/await
async function conCatch() {
    const usuario = await obtenerUsuario(1).catch(e => null);
    
    if (!usuario) {
        console.log('No se pudo obtener usuario');
        return;
    }
}

// 5. Error handling con Promise.allSettled
async function fetchMultiple() {
    const resultados = await Promise.allSettled([
        fetch('/api/users'),
        fetch('/api/posts'),
        fetch('/api/comments')
    ]);
    
    const exitosos = resultados
        .filter(r => r.status === 'fulfilled')
        .map(r => r.value);
    
    const fallidos = resultados
        .filter(r => r.status === 'rejected')
        .map(r => r.reason);
    
    console.log('Exitosos:', exitosos.length);
    console.log('Fallidos:', fallidos.length);
}

8Ejecución en Paralelo

// ========================================
// EJECUCIÓN EN PARALELO VS SECUENCIAL
// ========================================

function esperar(ms, valor) {
    return new Promise(r => setTimeout(() => r(valor), ms));
}

// ❌ SECUENCIAL (Lento) - 3 segundos total
async function secuencial() {
    console.time('secuencial');
    
    const a = await esperar(1000, 'A');
    const b = await esperar(1000, 'B');
    const c = await esperar(1000, 'C');
    
    console.timeEnd('secuencial'); // ~3000ms
    return [a, b, c];
}

// ✅ PARALELO (Rápido) - 1 segundo total
async function paralelo() {
    console.time('paralelo');
    
    const [a, b, c] = await Promise.all([
        esperar(1000, 'A'),
        esperar(1000, 'B'),
        esperar(1000, 'C')
    ]);
    
    console.timeEnd('paralelo'); // ~1000ms
    return [a, b, c];
}

// ✅ Iniciar en paralelo, await después
async function paraleloAlternativo() {
    // Inician TODAS inmediatamente
    const promesaA = esperar(1000, 'A');
    const promesaB = esperar(1000, 'B');
    const promesaC = esperar(1000, 'C');
    
    // Esperar resultados
    const a = await promesaA;
    const b = await promesaB;
    const c = await promesaC;
    
    return [a, b, c];
}

// Procesar array en paralelo
async function procesarEnParalelo(ids) {
    const promesas = ids.map(id => fetchUsuario(id));
    const usuarios = await Promise.all(promesas);
    return usuarios;
}

// Procesar con límite de concurrencia
async function procesarConLimite(items, limite, procesador) {
    const resultados = [];
    
    for (let i = 0; i < items.length; i += limite) {
        const chunk = items.slice(i, i + limite);
        const chunkResultados = await Promise.all(
            chunk.map(item => procesador(item))
        );
        resultados.push(...chunkResultados);
    }
    
    return resultados;
}

// Uso: procesar 100 items, 10 a la vez
const items = Array.from({ length: 100 }, (_, i) => i);
const resultados = await procesarConLimite(items, 10, async (item) => {
    await esperar(100);
    return item * 2;
});

// Clase API completa
class ApiClient {
    constructor(baseUrl) {
        this.baseUrl = baseUrl;
    }
    
    async request(endpoint, options = {}) {
        const url = `${this.baseUrl}${endpoint}`;
        
        const config = {
            headers: {
                'Content-Type': 'application/json',
                ...options.headers
            },
            ...options
        };
        
        try {
            const response = await fetch(url, config);
            
            if (!response.ok) {
                throw new Error(`HTTP error! status: ${response.status}`);
            }
            
            return await response.json();
            
        } catch (error) {
            console.error('API Error:', error);
            throw error;
        }
    }
    
    get(endpoint) {
        return this.request(endpoint);
    }
    
    post(endpoint, data) {
        return this.request(endpoint, {
            method: 'POST',
            body: JSON.stringify(data)
        });
    }
    
    put(endpoint, data) {
        return this.request(endpoint, {
            method: 'PUT',
            body: JSON.stringify(data)
        });
    }
    
    delete(endpoint) {
        return this.request(endpoint, { method: 'DELETE' });
    }
}

// Uso
const api = new ApiClient('https://jsonplaceholder.typicode.com');
const users = await api.get('/users');
const newUser = await api.post('/users', { name: 'Ana' });

// EJEMPLO 1: Búsqueda con Debounce
// ========================================

class SearchComponent {
    constructor(inputElement, resultsElement) {
        this.input = inputElement;
        this.results = resultsElement;
        this.abortController = null;
        
        this.input.addEventListener('input', 
            this.debounce(this.search.bind(this), 300)
        );
    }
    
    debounce(fn, delay) {
        let timeoutId;
        return (...args) => {
            clearTimeout(timeoutId);
            timeoutId = setTimeout(() => fn(...args), delay);
        };
    }
    
    async search(event) {
        const query = event.target.value.trim();
        
        if (query.length < 2) {
            this.results.innerHTML = '';
            return;
        }
        
        // Cancelar búsqueda anterior
        if (this.abortController) {
            this.abortController.abort();
        }
        this.abortController = new AbortController();
        
        try {
            this.results.innerHTML = '<p>Buscando...</p>';
            
            const response = await fetch(
                `/api/search?q=${encodeURIComponent(query)}`,
                { signal: this.abortController.signal }
            );
            
            const data = await response.json();
            this.renderResults(data);
            
        } catch (error) {
            if (error.name !== 'AbortError') {
                this.results.innerHTML = '<p>Error en la búsqueda</p>';
            }
        }
    }
    
    renderResults(data) {
        this.results.innerHTML = data
            .map(item => `<div class="result">${item.name}</div>`)
            .join('');
    }
}

// EJEMPLO 2: Carga de Datos con Caché
// ========================================

class DataService {
    constructor() {
        this.cache = new Map();
        this.pendingRequests = new Map();
    }
    
    async fetchWithCache(url, maxAge = 60000) {
        // Verificar caché
        const cached = this.cache.get(url);
        if (cached && Date.now() - cached.timestamp < maxAge) {
            return cached.data;
        }
        
        // Evitar peticiones duplicadas
        if (this.pendingRequests.has(url)) {
            return this.pendingRequests.get(url);
        }
        
        // Realizar petición
        const promise = this.fetch(url);
        this.pendingRequests.set(url, promise);
        
        try {
            const data = await promise;
            this.cache.set(url, { data, timestamp: Date.now() });
            return data;
        } finally {
            this.pendingRequests.delete(url);
        }
    }
    
    async fetch(url) {
        const response = await fetch(url);
        return response.json();
    }
    
    clearCache(url) {
        if (url) {
            this.cache.delete(url);
        } else {
            this.cache.clear();
        }
    }
}

// EJEMPLO 3: Retry con Backoff Exponencial
// ========================================

async function fetchWithRetry(url, options = {}) {
    const {
        maxRetries = 3,
        baseDelay = 1000,
        maxDelay = 10000
    } = options;
    
    let lastError;
    
    for (let attempt = 0; attempt <= maxRetries; attempt++) {
        try {
            const response = await fetch(url, options);
            
            if (!response.ok) {
                throw new Error(`HTTP ${response.status}`);
            }
            
            return await response.json();
            
        } catch (error) {
            lastError = error;
            
            if (attempt < maxRetries) {
                // Backoff exponencial: 1s, 2s, 4s...
                const delay = Math.min(
                    baseDelay * Math.pow(2, attempt),
                    maxDelay
                );
                
                console.log(`Retry ${attempt + 1}/${maxRetries} en ${delay}ms`);
                await new Promise(r => setTimeout(r, delay));
            }
        }
    }
    
    throw lastError;
}

// EJEMPLO 4: Carga de Imágenes
// ========================================

function cargarImagen(src) {
    return new Promise((resolve, reject) => {
        const img = new Image();
        img.onload = () => resolve(img);
        img.onerror = () => reject(new Error(`Error cargando: ${src}`));
        img.src = src;
    });
}

async function cargarGaleria(imagenes) {
    const resultados = await Promise.allSettled(
        imagenes.map(src => cargarImagen(src))
    );
    
    const cargadas = resultados
        .filter(r => r.status === 'fulfilled')
        .map(r => r.value);
    
    const fallidas = resultados
        .filter(r => r.status === 'rejected')
        .map(r => r.reason.message);
    
    console.log(`Cargadas: ${cargadas.length}, Fallidas: ${fallidas.length}`);
    
    return cargadas;
}
