1Declaración de Funciones (Function Declaration)

La forma tradicional de crear funciones. Son "hoisted" (elevadas), pueden usarse antes de su declaración.

// ========================================
// DECLARACIÓN DE FUNCIONES
// ========================================

// Sintaxis básica
function saludar() {
    console.log('¡Hola!');
}

// Con parámetros
function saludarA(nombre) {
    console.log(`¡Hola, ${nombre}!`);
}

// Con return
function sumar(a, b) {
    return a + b;
}

// Llamar funciones
saludar(); // '¡Hola!'
saludarA('Ana'); // '¡Hola, Ana!'
const resultado = sumar(5, 3); // 8

// Hoisting - Funciona antes de la declaración
console.log(multiplicar(4, 5)); // 20 ✓

function multiplicar(a, b) {
    return a * b;
}

// Función dentro de función
function externa() {
    function interna() {
        console.log('Soy interna');
    }
    interna();
}
externa(); // 'Soy interna'

2Expresiones de Función (Function Expression)

Funciones asignadas a variables. NO son hoisted.

// ========================================
// EXPRESIONES DE FUNCIÓN
// ========================================

// Función anónima asignada a variable
const saludar = function() {
    console.log('¡Hola!');
};

// Función con nombre (named function expression)
const factorial = function fact(n) {
    if (n <= 1) return 1;
    return n * fact(n - 1); // Puede llamarse a sí misma
};
console.log(factorial(5)); // 120

// NO hay hoisting - esto da error
// console.log(dividir(10, 2)); // ❌ Error!

const dividir = function(a, b) {
    return a / b;
};

console.log(dividir(10, 2)); // 5 ✓

// Función como argumento
const numeros = [1, 2, 3, 4];
const dobles = numeros.map(function(n) {
    return n * 2;
});
console.log(dobles); // [2, 4, 6, 8]

// Función que retorna función
const crearSumador = function(x) {
    return function(y) {
        return x + y;
    };
};

const sumar5 = crearSumador(5);
console.log(sumar5(3)); // 8
console.log(sumar5(10)); // 15

3Arrow Functions (ES6)

Sintaxis corta y moderna. No tienen su propio this.

// ========================================
// ARROW FUNCTIONS (=>)
// ========================================

// Sintaxis básica
const saludar = () => {
    console.log('¡Hola!');
};

// Con un parámetro (paréntesis opcionales)
const doble = n => n * 2;
const doble2 = (n) => n * 2; // Equivalente

// Con múltiples parámetros (paréntesis obligatorios)
const sumar = (a, b) => a + b;

// Return implícito (una sola expresión)
const cuadrado = x => x * x;
console.log(cuadrado(4)); // 16

// Return explícito (con llaves)
const calcular = (a, b) => {
    const suma = a + b;
    const producto = a * b;
    return { suma, producto };
};
console.log(calcular(3, 4)); // { suma: 7, producto: 12 }

// Retornar objeto literal (envolver en paréntesis)
const crearUsuario = (nombre, edad) => ({ nombre, edad });
console.log(crearUsuario('Ana', 25)); // { nombre: 'Ana', edad: 25 }

// Con arrays
const numeros = [1, 2, 3, 4, 5];
const pares = numeros.filter(n => n % 2 === 0);
const dobles = numeros.map(n => n * 2);
const suma = numeros.reduce((acc, n) => acc + n, 0);

// Arrow functions y 'this'
const objeto = {
    nombre: 'Mi Objeto',
    
    // Método tradicional - 'this' es el objeto
    metodoNormal() {
        console.log(this.nombre); // 'Mi Objeto'
    },
    
    // Arrow - 'this' es el contexto donde se definió
    metodoArrow: () => {
        console.log(this.nombre); // undefined (this es window)
    },
    
    // Útil dentro de callbacks
    iniciar() {
        setTimeout(() => {
            console.log(this.nombre); // 'Mi Objeto' ✓
        }, 1000);
    }
};
Cuándo NO usar Arrow Functions
  • Como métodos de objetos (si necesitas this)
  • Como constructores (new)
  • Cuando necesitas arguments
  • En event handlers del DOM (si necesitas this = elemento)

4Parámetros y Argumentos

// ========================================
// PARÁMETROS EN FUNCIONES
// ========================================

// Parámetros vs Argumentos
// Parámetros: variables en la definición
// Argumentos: valores pasados al llamar

// 1. Valores por defecto (ES6)
function saludar(nombre = 'Invitado', saludo = 'Hola') {
    console.log(`${saludo}, ${nombre}!`);
}
saludar(); // 'Hola, Invitado!'
saludar('Ana'); // 'Hola, Ana!'
saludar('Ana', 'Buenos días'); // 'Buenos días, Ana!'

// Default con expresiones
function crearId(prefijo = 'ID', numero = Date.now()) {
    return `${prefijo}-${numero}`;
}

// 2. Rest parameters (...args)
function sumarTodos(...numeros) {
    return numeros.reduce((acc, n) => acc + n, 0);
}
console.log(sumarTodos(1, 2, 3, 4, 5)); // 15

// Rest con parámetros normales
function mostrarEquipo(capitan, ...jugadores) {
    console.log(`Capitán: ${capitan}`);
    console.log(`Jugadores: ${jugadores.join(', ')}`);
}
mostrarEquipo('Ana', 'Carlos', 'Elena', 'David');

// 3. Destructuring en parámetros
function mostrarUsuario({ nombre, edad, ciudad = 'Desconocida' }) {
    console.log(`${nombre}, ${edad} años, ${ciudad}`);
}
mostrarUsuario({ nombre: 'Ana', edad: 25 });

// Destructuring de array
function procesar([primero, segundo, ...resto]) {
    console.log(primero, segundo, resto);
}
procesar([1, 2, 3, 4]); // 1 2 [3, 4]

// 4. Objeto arguments (solo funciones normales)
function antiguaForma() {
    console.log(arguments.length);
    console.log(arguments[0]);
    // Convertir a array
    const args = Array.from(arguments);
}
antiguaForma('a', 'b', 'c'); // 3, 'a'

// 5. Spread para pasar argumentos
const nums = [1, 2, 3];
console.log(Math.max(...nums)); // 3

function crear(a, b, c) {
    return a + b + c;
}
console.log(crear(...nums)); // 6

5Return y Valores de Retorno

// ========================================
// RETURN EN FUNCIONES
// ========================================

// Return termina la ejecución
function verificarEdad(edad) {
    if (edad < 0) {
        return 'Edad inválida'; // Sale aquí
    }
    if (edad < 18) {
        return 'Menor de edad';
    }
    return 'Mayor de edad';
}

// Sin return = undefined
function sinReturn() {
    console.log('Ejecutando...');
}
const resultado = sinReturn();
console.log(resultado); // undefined

// Return vacío = undefined
function returnVacio() {
    return;
}
console.log(returnVacio()); // undefined

// Retornar múltiples valores (con objeto o array)
function estadisticas(numeros) {
    const suma = numeros.reduce((a, b) => a + b, 0);
    const promedio = suma / numeros.length;
    const maximo = Math.max(...numeros);
    const minimo = Math.min(...numeros);
    
    return { suma, promedio, maximo, minimo };
}

const stats = estadisticas([5, 10, 15, 20]);
console.log(stats.promedio); // 12.5

// Destructuring del return
const { suma, maximo } = estadisticas([1, 2, 3]);

// Retornar con array
function dividirConResto(a, b) {
    return [Math.floor(a / b), a % b];
}
const [cociente, resto] = dividirConResto(17, 5);
console.log(cociente, resto); // 3, 2

// Retornar función
function crearMultiplicador(factor) {
    return (numero) => numero * factor;
}
const triplicar = crearMultiplicador(3);
console.log(triplicar(10)); // 30

6Scope (Alcance)

// ========================================
// SCOPE - ALCANCE DE VARIABLES
// ========================================

// GLOBAL SCOPE
var globalVar = 'Soy global';
let globalLet = 'También global';

// FUNCTION SCOPE (var)
function ejemploVar() {
    var dentroFuncion = 'Solo visible aquí';
    console.log(dentroFuncion); // ✓
    
    if (true) {
        var dentroIf = 'var ignora bloques';
    }
    console.log(dentroIf); // ✓ Accesible (var no respeta bloques)
}
// console.log(dentroFuncion); // ❌ Error

// BLOCK SCOPE (let, const)
function ejemploLet() {
    let funcionLet = 'En función';
    
    if (true) {
        let bloqueLet = 'Solo en este bloque';
        const bloqueConst = 'También en bloque';
        console.log(bloqueLet); // ✓
    }
    // console.log(bloqueLet); // ❌ Error
}

// LEXICAL SCOPE (Alcance léxico)
function externa() {
    const externaVar = 'Soy de externa';
    
    function interna() {
        const internaVar = 'Soy de interna';
        console.log(externaVar); // ✓ Accede a variable de externa
        console.log(globalVar); // ✓ Accede a global
    }
    
    interna();
    // console.log(internaVar); // ❌ No puede acceder
}

// SCOPE CHAIN (Cadena de alcance)
const nivel0 = 'Global';

function nivel1() {
    const varNivel1 = 'Nivel 1';
    
    function nivel2() {
        const varNivel2 = 'Nivel 2';
        
        function nivel3() {
            console.log(nivel0); // ✓ Busca hacia arriba
            console.log(varNivel1); // ✓
            console.log(varNivel2); // ✓
        }
        nivel3();
    }
    nivel2();
}

// SHADOWING (Sombreado)
const x = 10;

function sombra() {
    const x = 20; // Sombrea la x global
    console.log(x); // 20
}
sombra();
console.log(x); // 10

7Closures (Clausuras)

Una función que recuerda el entorno donde fue creada, incluso después de que ese entorno haya terminado.

// ========================================
// CLOSURES - CLAUSURAS
// ========================================

// Ejemplo básico de closure
function crearContador() {
    let contador = 0; // Variable "privada"
    
    return function() {
        contador++;
        return contador;
    };
}

const contador1 = crearContador();
console.log(contador1()); // 1
console.log(contador1()); // 2
console.log(contador1()); // 3

const contador2 = crearContador();
console.log(contador2()); // 1 (contador independiente)

// Closure con múltiples funciones
function crearCuentaBancaria(saldoInicial) {
    let saldo = saldoInicial;
    
    return {
        depositar(cantidad) {
            saldo += cantidad;
            return saldo;
        },
        retirar(cantidad) {
            if (cantidad <= saldo) {
                saldo -= cantidad;
                return saldo;
            }
            return 'Saldo insuficiente';
        },
        verSaldo() {
            return saldo;
        }
    };
}

const miCuenta = crearCuentaBancaria(1000);
console.log(miCuenta.depositar(500)); // 1500
console.log(miCuenta.retirar(200)); // 1300
console.log(miCuenta.verSaldo()); // 1300

// Closure en loop (problema clásico)
// ❌ PROBLEMA con var
for (var i = 0; i < 3; i++) {
    setTimeout(function() {
        console.log(i); // 3, 3, 3 (todas ven el mismo i)
    }, 100);
}

// ✓ SOLUCIÓN con let (block scope)
for (let j = 0; j < 3; j++) {
    setTimeout(function() {
        console.log(j); // 0, 1, 2
    }, 100);
}

// ✓ SOLUCIÓN con closure
for (var k = 0; k < 3; k++) {
    (function(num) {
        setTimeout(function() {
            console.log(num); // 0, 1, 2
        }, 100);
    })(k);
}

// Módulo con closure (Module Pattern)
const miModulo = (function() {
    let privado = 'Soy privado';
    
    function metodoPrivado() {
        console.log(privado);
    }
    
    return {
        metodoPublico() {
            metodoPrivado();
        },
        cambiarPrivado(nuevo) {
            privado = nuevo;
        }
    };
})();

miModulo.metodoPublico(); // 'Soy privado'

8Callbacks

Una función pasada como argumento a otra función.

// ========================================
// CALLBACKS
// ========================================

// Callback básico
function procesarUsuario(id, callback) {
    // Simulamos obtener usuario
    const usuario = { id, nombre: 'Ana' };
    callback(usuario);
}

procesarUsuario(1, function(user) {
    console.log(`Usuario: ${user.nombre}`);
});

// Con arrow function
procesarUsuario(2, user => console.log(user));

// Callback con error (patrón Node.js)
function dividir(a, b, callback) {
    if (b === 0) {
        callback(new Error('No se puede dividir por cero'), null);
    } else {
        callback(null, a / b);
    }
}

dividir(10, 2, (error, resultado) => {
    if (error) {
        console.error(error.message);
    } else {
        console.log('Resultado:', resultado); // 5
    }
});

// Callbacks con setTimeout
console.log('Inicio');

setTimeout(() => {
    console.log('Después de 1 segundo');
}, 1000);

console.log('Fin');
// Salida: Inicio, Fin, Después de 1 segundo

// Callback Hell (Problema)
obtenerUsuario(1, (usuario) => {
    obtenerPedidos(usuario.id, (pedidos) => {
        obtenerProductos(pedidos[0].id, (productos) => {
            obtenerDetalles(productos[0].id, (detalles) => {
                console.log(detalles);
                // Esto se vuelve difícil de mantener
            });
        });
    });
});

// Callbacks en array methods
const numeros = [1, 2, 3, 4, 5];

numeros.forEach((num, index) => {
    console.log(`Índice ${index}: ${num}`);
});

const dobles = numeros.map(n => n * 2);
const pares = numeros.filter(n => n % 2 === 0);
const suma = numeros.reduce((acc, n) => acc + n, 0);

// Callback para ordenar
const usuarios = [
    { nombre: 'Carlos', edad: 30 },
    { nombre: 'Ana', edad: 25 },
    { nombre: 'Elena', edad: 28 }
];

usuarios.sort((a, b) => a.edad - b.edad); // Por edad
usuarios.sort((a, b) => a.nombre.localeCompare(b.nombre)); // Por nombre

9Higher-Order Functions

Funciones que reciben funciones como argumento o retornan funciones.

// ========================================
// HIGHER-ORDER FUNCTIONS
// ========================================

// 1. Función que recibe función
function ejecutarDosVeces(fn) {
    fn();
    fn();
}

ejecutarDosVeces(() => console.log('¡Hola!'));
// '¡Hola!' x2

// 2. Función que retorna función
function crearSaludador(saludo) {
    return function(nombre) {
        console.log(`${saludo}, ${nombre}!`);
    };
}

const saludarES = crearSaludador('Hola');
const saludarEN = crearSaludador('Hello');

saludarES('Ana'); // 'Hola, Ana!'
saludarEN('John'); // 'Hello, John!'

// 3. Crear HOF propias
function miMap(array, transformar) {
    const resultado = [];
    for (const item of array) {
        resultado.push(transformar(item));
    }
    return resultado;
}

const nums = [1, 2, 3];
console.log(miMap(nums, x => x * 2)); // [2, 4, 6]

function miFilter(array, condicion) {
    const resultado = [];
    for (const item of array) {
        if (condicion(item)) {
            resultado.push(item);
        }
    }
    return resultado;
}

// 4. Composición de funciones
const duplicar = x => x * 2;
const incrementar = x => x + 1;
const cuadrado = x => x * x;

// Componer manualmente
const resultado = cuadrado(duplicar(incrementar(3)));
console.log(resultado); // ((3+1)*2)² = 64

// Función compose
const compose = (...fns) => x => fns.reduceRight((acc, fn) => fn(acc), x);
const pipe = (...fns) => x => fns.reduce((acc, fn) => fn(acc), x);

const calcular = pipe(incrementar, duplicar, cuadrado);
console.log(calcular(3)); // 64

// 5. Curry
function curry(fn) {
    return function curried(...args) {
        if (args.length >= fn.length) {
            return fn.apply(this, args);
        }
        return function(...nextArgs) {
            return curried.apply(this, args.concat(nextArgs));
        };
    };
}

function suma(a, b, c) {
    return a + b + c;
}

const sumaCurried = curry(suma);
console.log(sumaCurried(1)(2)(3)); // 6
console.log(sumaCurried(1, 2)(3)); // 6
console.log(sumaCurried(1)(2, 3)); // 6

// 6. Partial application
function partial(fn, ...presetArgs) {
    return function(...laterArgs) {
        return fn(...presetArgs, ...laterArgs);
    };
}

const sumar10 = partial(suma, 10);
console.log(sumar10(5, 3)); // 18

10IIFE (Immediately Invoked Function Expression)

// ========================================
// IIFE - Función Autoejecutable
// ========================================

// Sintaxis básica
(function() {
    console.log('¡Me ejecuto inmediatamente!');
})();

// Con arrow function
(() => {
    console.log('IIFE con arrow');
})();

// Con parámetros
((nombre) => {
    console.log(`Hola, ${nombre}!`);
})('Ana');

// IIFE con return
const resultado = (function() {
    const privado = 10;
    return privado * 2;
})();
console.log(resultado); // 20

// Uso: Crear scope privado
(function() {
    var secreto = 'Nadie me ve fuera';
    // ... código que usa secreto
})();
// console.log(secreto); // ❌ Error

// Module Pattern con IIFE
const Calculadora = (function() {
    // Variables privadas
    let historial = [];
    
    // Funciones privadas
    function guardarEnHistorial(operacion, resultado) {
        historial.push({ operacion, resultado, fecha: new Date() });
    }
    
    // API pública
    return {
        sumar(a, b) {
            const resultado = a + b;
            guardarEnHistorial(`${a} + ${b}`, resultado);
            return resultado;
        },
        restar(a, b) {
            const resultado = a - b;
            guardarEnHistorial(`${a} - ${b}`, resultado);
            return resultado;
        },
        verHistorial() {
            return [...historial];
        },
        limpiarHistorial() {
            historial = [];
        }
    };
})();

console.log(Calculadora.sumar(5, 3)); // 8
console.log(Calculadora.restar(10, 4)); // 6
console.log(Calculadora.verHistorial());

// Async IIFE
(async () => {
    const response = await fetch('/api/datos');
    const data = await response.json();
    console.log(data);
})();

11Recursión

Una función que se llama a sí misma.

// ========================================
// RECURSIÓN
// ========================================

// Factorial: n! = n * (n-1)!
function factorial(n) {
    // Caso base (condición de salida)
    if (n <= 1) return 1;
    
    // Caso recursivo
    return n * factorial(n - 1);
}

console.log(factorial(5)); // 120 (5*4*3*2*1)

// Fibonacci: fib(n) = fib(n-1) + fib(n-2)
function fibonacci(n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}

console.log(fibonacci(10)); // 55

// Contar regresivamente
function cuentaRegresiva(n) {
    if (n < 0) return;
    console.log(n);
    cuentaRegresiva(n - 1);
}
cuentaRegresiva(5); // 5, 4, 3, 2, 1, 0

// Sumar array recursivamente
function sumarArray(arr) {
    if (arr.length === 0) return 0;
    return arr[0] + sumarArray(arr.slice(1));
}

console.log(sumarArray([1, 2, 3, 4])); // 10

// Aplanar array anidado
function aplanar(arr) {
    let resultado = [];
    
    for (const item of arr) {
        if (Array.isArray(item)) {
            resultado = resultado.concat(aplanar(item));
        } else {
            resultado.push(item);
        }
    }
    
    return resultado;
}

const anidado = [1, [2, [3, [4]], 5]];
console.log(aplanar(anidado)); // [1, 2, 3, 4, 5]

// Recorrer objeto profundamente
function recorrerObjeto(obj, prefijo = '') {
    for (const [key, value] of Object.entries(obj)) {
        const ruta = prefijo ? `${prefijo}.${key}` : key;
        
        if (typeof value === 'object' && value !== null) {
            recorrerObjeto(value, ruta);
        } else {
            console.log(`${ruta}: ${value}`);
        }
    }
}

const persona = {
    nombre: 'Ana',
    direccion: {
        ciudad: 'Madrid',
        cp: '28001'
    }
};
recorrerObjeto(persona);
// nombre: Ana
// direccion.ciudad: Madrid
// direccion.cp: 28001

// Tail recursion (optimización)
function factorialTail(n, acumulador = 1) {
    if (n <= 1) return acumulador;
    return factorialTail(n - 1, n * acumulador);
}

12Técnicas Avanzadas

// ========================================
// TÉCNICAS AVANZADAS
// ========================================

// 1. Memoization (caché de resultados)
function memoize(fn) {
    const cache = new Map();
    
    return function(...args) {
        const key = JSON.stringify(args);
        
        if (cache.has(key)) {
            console.log('Desde caché');
            return cache.get(key);
        }
        
        const resultado = fn.apply(this, args);
        cache.set(key, resultado);
        return resultado;
    };
}

const fibMemo = memoize(function fib(n) {
    if (n <= 1) return n;
    return fibMemo(n - 1) + fibMemo(n - 2);
});

console.log(fibMemo(40)); // Muy rápido con memoization

// 2. Debounce (espera antes de ejecutar)
function debounce(fn, delay) {
    let timeoutId;
    
    return function(...args) {
        clearTimeout(timeoutId);
        timeoutId = setTimeout(() => {
            fn.apply(this, args);
        }, delay);
    };
}

// Uso: búsqueda mientras escribes
const buscar = debounce((texto) => {
    console.log('Buscando:', texto);
}, 300);

// buscar('a'); buscar('ab'); buscar('abc');
// Solo ejecuta con 'abc' (espera 300ms sin nuevas llamadas)

// 3. Throttle (ejecuta máximo una vez por intervalo)
function throttle(fn, limit) {
    let inThrottle;
    
    return function(...args) {
        if (!inThrottle) {
            fn.apply(this, args);
            inThrottle = true;
            setTimeout(() => inThrottle = false, limit);
        }
    };
}

// Uso: scroll, resize
const handleScroll = throttle(() => {
    console.log('Scroll position:', window.scrollY);
}, 100);

// 4. Once (ejecuta solo una vez)
function once(fn) {
    let executed = false;
    let result;
    
    return function(...args) {
        if (!executed) {
            executed = true;
            result = fn.apply(this, args);
        }
        return result;
    };
}

const inicializar = once(() => {
    console.log('Inicializando...');
    return { config: true };
});

inicializar(); // 'Inicializando...'
inicializar(); // (nada, ya se ejecutó)

// 5. bind(), call(), apply()
const persona = {
    nombre: 'Ana',
    saludar(saludo, signo) {
        console.log(`${saludo}, soy ${this.nombre}${signo}`);
    }
};

const otraPersona = { nombre: 'Carlos' };

// call: ejecuta con this diferente
persona.saludar.call(otraPersona, 'Hola', '!'); // 'Hola, soy Carlos!'

// apply: igual pero args en array
persona.saludar.apply(otraPersona, ['Hey', '?']); // 'Hey, soy Carlos?'

// bind: crea nueva función con this fijo
const saludarCarlos = persona.saludar.bind(otraPersona);
saludarCarlos('Buenos días', '.'); // 'Buenos días, soy Carlos.'

// Bind con argumentos parciales
const saludarHola = persona.saludar.bind(otraPersona, 'Hola');
saludarHola('!'); // 'Hola, soy Carlos!'

// 6. Generator functions
function* generadorNumeros(max) {
    for (let i = 0; i < max; i++) {
        yield i;
    }
}

const gen = generadorNumeros(3);
console.log(gen.next()); // { value: 0, done: false }
console.log(gen.next()); // { value: 1, done: false }
console.log(gen.next()); // { value: 2, done: false }
console.log(gen.next()); // { value: undefined, done: true }
