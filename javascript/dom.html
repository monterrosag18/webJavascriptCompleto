Selectores DOM - 5 Ejemplos

Los selectores son la base para manipular el DOM. Permiten encontrar elementos en el documento HTML.

1getElementById()

Selecciona UN elemento por su ID (el más rápido)

Este párrafo tiene id="demo-id"

// ====== getElementById ======
// Selecciona UN ÚNICO elemento por su atributo ID
// Retorna: el elemento o null si no existe

// HTML: <p id="mi-parrafo">Texto</p>

const elemento = document.getElementById('mi-parrafo');

// Cambiar el texto
elemento.textContent = 'Nuevo texto';

// Cambiar HTML interno
elemento.innerHTML = '<strong>Texto en negrita</strong>';

// Verificar si existe antes de usar
if (elemento) {
    elemento.style.color = 'red';
}

// ⚠️ IMPORTANTE: Los IDs deben ser únicos en el documento
2querySelector()

Selecciona el PRIMER elemento que coincida con el selector CSS

Primer div con clase demo-class
Segundo div con clase demo-class
// ====== querySelector ======
// Selecciona el PRIMER elemento que coincida con el selector CSS
// Acepta cualquier selector CSS válido

// Por clase (usa punto .)
const porClase = document.querySelector('.mi-clase');

// Por ID (usa numeral #)
const porId = document.querySelector('#mi-id');

// Por etiqueta
const porTag = document.querySelector('p');

// Selectores combinados
const combinado = document.querySelector('div.card .card-title');

// Por atributo
const porAtributo = document.querySelector('[data-id="123"]');
const inputEmail = document.querySelector('input[type="email"]');

// Pseudo-selectores
const primerHijo = document.querySelector('ul li:first-child');
const ultimoHijo = document.querySelector('ul li:last-child');

// 💡 TIP: querySelector es muy versátil y el más usado
3querySelectorAll()

Selecciona TODOS los elementos que coincidan (retorna NodeList)

  • Item 1
  • Item 2
  • Item 3
// ====== querySelectorAll ======
// Selecciona TODOS los elementos que coincidan
// Retorna: NodeList (similar a un array)

// Seleccionar todos los párrafos
const parrafos = document.querySelectorAll('p');

// Seleccionar por clase
const items = document.querySelectorAll('.item');

// Seleccionar múltiples selectores
const varios = document.querySelectorAll('h1, h2, h3');

// ====== RECORRER LOS ELEMENTOS ======

// Método 1: forEach (más común)
items.forEach((item, index) => {
    item.textContent = `Item ${index + 1}`;
    item.style.color = 'blue';
});

// Método 2: for...of
for (const item of items) {
    console.log(item.textContent);
}

// Método 3: Convertir a Array para usar métodos de array
const itemsArray = Array.from(items);
// o también: const itemsArray = [...items];

const filtrados = itemsArray.filter(item => 
    item.classList.contains('activo')
);

// Acceder por índice
const primero = items[0];
const ultimo = items[items.length - 1];
4getElementsByClassName()

Selecciona elementos por nombre de clase (retorna HTMLCollection VIVA)

// ====== getElementsByClassName ======
// Retorna: HTMLCollection (colección VIVA que se actualiza automáticamente)

const elementos = document.getElementsByClassName('mi-clase');

// ⚠️ IMPORTANTE: HTMLCollection es VIVA
// Si agregas/eliminas elementos del DOM, la colección se actualiza

// Recorrer con for tradicional
for (let i = 0; i < elementos.length; i++) {
    elementos[i].style.backgroundColor = 'yellow';
}

// Convertir a array si necesitas métodos de array
const array = Array.from(elementos);
array.forEach(el => console.log(el));

// Múltiples clases (elemento debe tener TODAS)
const conDosClases = document.getElementsByClassName('clase1 clase2');

// 💡 DIFERENCIA con querySelectorAll:
// - getElementsByClassName retorna colección VIVA
// - querySelectorAll retorna NodeList ESTÁTICA

Modificar Contenido - 5 Ejemplos

1textContent vs innerHTML vs innerText
Texto visible Texto oculto 
// ====== textContent ======
// - Obtiene/establece TODO el texto (incluyendo oculto)
// - NO interpreta HTML (más seguro contra XSS)
// - Más rápido que innerHTML

const elemento = document.querySelector('#mi-elemento');

// Leer texto
const texto = elemento.textContent;

// Establecer texto (reemplaza todo)
elemento.textContent = 'Nuevo texto plano';

// HTML se muestra como texto literal
elemento.textContent = '<b>Esto se ve así</b>'; // Muestra las etiquetas


// ====== innerHTML ======
// - Obtiene/establece HTML interno
// - SÍ interpreta HTML
// - ⚠️ CUIDADO: Riesgo de XSS si usas datos del usuario

// Leer HTML
const html = elemento.innerHTML;

// Establecer HTML
elemento.innerHTML = '<strong>Texto en negrita</strong>';

// Agregar contenido (NO recomendado en loops)
elemento.innerHTML += '<p>Párrafo nuevo</p>';


// ====== innerText ======
// - Similar a textContent pero...
// - Respeta estilos CSS (no muestra texto oculto)
// - Más lento porque calcula estilos

const textoVisible = elemento.innerText;


// ====== outerHTML ======
// - Incluye el elemento mismo
const htmlCompleto = elemento.outerHTML;
// Retorna: "<div id="mi-elemento">contenido</div>"


// 💡 RECOMENDACIÓN:
// - Usa textContent para texto plano (más seguro)
// - Usa innerHTML solo cuando necesites HTML
// - Nunca uses innerHTML con datos del usuario sin sanitizar
2Value - Inputs y Formularios
// ====== OBTENER Y ESTABLECER VALORES DE FORMULARIOS ======

// INPUT TEXT
const input = document.querySelector('#mi-input');
const valor = input.value;           // Obtener valor
input.value = 'Nuevo valor';        // Establecer valor
input.value = '';                   // Limpiar input

// INPUT NUMBER
const numero = document.querySelector('input[type="number"]');
const valorNum = parseInt(numero.value); // Convertir a número
const valorDec = parseFloat(numero.value); // Convertir a decimal

// SELECT
const select = document.querySelector('#mi-select');
const valorSelect = select.value;    // Valor de la opción seleccionada
const textoOpcion = select.options[select.selectedIndex].text; // Texto visible
select.value = 'opcion2';           // Seleccionar opción por valor

// TEXTAREA
const textarea = document.querySelector('textarea');
const texto = textarea.value;

// CHECKBOX
const checkbox = document.querySelector('input[type="checkbox"]');
const estaChecked = checkbox.checked;  // true o false
checkbox.checked = true;               // Marcar
checkbox.checked = false;              // Desmarcar

// RADIO BUTTONS
const radios = document.querySelectorAll('input[name="opcion"]');
let valorRadio = '';
radios.forEach(radio => {
    if (radio.checked) {
        valorRadio = radio.value;
    }
});

// Forma más elegante con querySelector
const radioSeleccionado = document.querySelector('input[name="opcion"]:checked');
if (radioSeleccionado) {
    console.log(radioSeleccionado.value);
}

// OBTENER TODOS LOS VALORES DE UN FORMULARIO
const form = document.querySelector('form');
const formData = new FormData(form);

// Convertir a objeto
const datos = Object.fromEntries(formData);
console.log(datos); // { nombre: "valor", email: "valor", ... }
3Crear y Agregar Elementos

Contenedor original

// ====== CREAR ELEMENTOS ======

// Crear un elemento
const nuevoDiv = document.createElement('div');
const nuevoParrafo = document.createElement('p');
const nuevaImagen = document.createElement('img');

// Agregar contenido y atributos
nuevoDiv.textContent = 'Contenido del div';
nuevoDiv.className = 'mi-clase otra-clase';
nuevoDiv.id = 'mi-nuevo-div';

// ====== AGREGAR AL DOM ======

const contenedor = document.querySelector('#contenedor');

// appendChild - Agrega al FINAL del contenedor
contenedor.appendChild(nuevoDiv);

// append - Puede agregar múltiples elementos y texto
contenedor.append(nuevoDiv, 'texto directo', otroElemento);

// prepend - Agrega al INICIO del contenedor
contenedor.prepend(nuevoDiv);

// before - Agrega ANTES del elemento
elementoExistente.before(nuevoDiv);

// after - Agrega DESPUÉS del elemento
elementoExistente.after(nuevoDiv);

// insertBefore - Insertar antes de un hijo específico
contenedor.insertBefore(nuevoDiv, contenedor.firstChild);

// ====== insertAdjacentHTML (muy útil) ======
// Posiciones: 'beforebegin', 'afterbegin', 'beforeend', 'afterend'

elemento.insertAdjacentHTML('beforeend', '<p>Nuevo párrafo</p>');

// beforebegin: Antes del elemento
// afterbegin: Dentro, al principio
// beforeend: Dentro, al final
// afterend: Después del elemento

// ====== CLONAR ELEMENTOS ======
const clon = elemento.cloneNode(true);  // true = incluye hijos
const clonSuperficial = elemento.cloneNode(false); // solo el elemento

// ====== CREAR ELEMENTOS COMPLEJOS ======
function crearCard(titulo, descripcion) {
    const card = document.createElement('div');
    card.className = 'card';
    card.innerHTML = `
        <div class="card-body">
            <h5 class="card-title">${titulo}</h5>
            <p class="card-text">${descripcion}</p>
        </div>
    `;
    return card;
}
4Eliminar y Reemplazar Elementos
// ====== ELIMINAR ELEMENTOS ======

const elemento = document.querySelector('.elemento-a-eliminar');

// Método moderno: remove()
elemento.remove();

// Método antiguo: removeChild() - aún usado para compatibilidad
elemento.parentNode.removeChild(elemento);

// Eliminar todos los hijos de un contenedor
const contenedor = document.querySelector('#contenedor');

// Método 1: innerHTML (rápido pero elimina event listeners)
contenedor.innerHTML = '';

// Método 2: Loop (preserva referencias)
while (contenedor.firstChild) {
    contenedor.removeChild(contenedor.firstChild);
}

// Método 3: replaceChildren() - moderno
contenedor.replaceChildren(); // Sin argumentos = vacía el contenedor

// ====== REEMPLAZAR ELEMENTOS ======

// replaceWith - Reemplaza el elemento con otro(s)
const nuevoElemento = document.createElement('div');
nuevoElemento.textContent = 'Soy el reemplazo';
elementoViejo.replaceWith(nuevoElemento);

// replaceChild - Método del padre
padre.replaceChild(nuevoElemento, elementoViejo);

// Reemplazar con HTML
elemento.outerHTML = '<div class="nuevo">Contenido nuevo</div>';

// ====== EJEMPLO PRÁCTICO: Eliminar items de lista ======
function eliminarItem(boton) {
    const item = boton.closest('li'); // Encuentra el padre li
    item.remove();
}

Elemento para modificar estilos 
// ====== MODIFICAR ESTILOS INLINE ======

const elemento = document.querySelector('#mi-elemento');

// Establecer estilos individuales (camelCase)
elemento.style.color = 'red';
elemento.style.backgroundColor = '#f0f0f0';
elemento.style.fontSize = '18px';
elemento.style.padding = '10px 20px';
elemento.style.border = '2px solid blue';
elemento.style.borderRadius = '8px';
elemento.style.display = 'flex';
elemento.style.justifyContent = 'center';

// Establecer múltiples estilos con cssText
elemento.style.cssText = 'color: red; font-size: 20px; padding: 10px;';

// Leer estilos computados (incluyendo CSS)
const estilos = getComputedStyle(elemento);
console.log(estilos.color);
console.log(estilos.fontSize);


// ====== MODIFICAR CLASES ======

// classList es la forma moderna y recomendada

// Agregar clase(s)
elemento.classList.add('nueva-clase');
elemento.classList.add('clase1', 'clase2', 'clase3'); // múltiples

// Eliminar clase(s)
elemento.classList.remove('clase-a-eliminar');
elemento.classList.remove('clase1', 'clase2'); // múltiples

// Toggle - agregar si no existe, eliminar si existe
elemento.classList.toggle('activo');

// Toggle con condición
elemento.classList.toggle('visible', condicion); // agrega si true, quita si false

// Verificar si tiene una clase
if (elemento.classList.contains('mi-clase')) {
    console.log('Tiene la clase');
}

// Reemplazar una clase por otra
elemento.classList.replace('clase-vieja', 'clase-nueva');

// Obtener todas las clases como array
const clases = [...elemento.classList];

// Forma antigua (aún funciona pero menos flexible)
elemento.className = 'clase1 clase2 clase3'; // Reemplaza todas
elemento.className += ' nueva-clase'; // Agrega (cuidado con espacios)

Atributos - Ejemplos Completos

1Manipular Atributos
// ====== ATRIBUTOS ESTÁNDAR ======

const elemento = document.querySelector('#mi-elemento');
const imagen = document.querySelector('img');
const enlace = document.querySelector('a');

// getAttribute - Obtener valor de atributo
const id = elemento.getAttribute('id');
const src = imagen.getAttribute('src');
const href = enlace.getAttribute('href');

// setAttribute - Establecer atributo
imagen.setAttribute('src', 'nueva-imagen.jpg');
imagen.setAttribute('alt', 'Descripción de la imagen');
enlace.setAttribute('target', '_blank');

// hasAttribute - Verificar si existe
if (elemento.hasAttribute('disabled')) {
    console.log('Está deshabilitado');
}

// removeAttribute - Eliminar atributo
elemento.removeAttribute('disabled');

// Acceso directo a atributos comunes
imagen.src = 'imagen.jpg';      // Más corto que setAttribute
enlace.href = 'https://...';
elemento.id = 'nuevo-id';
input.placeholder = 'Escribe aquí';
input.disabled = true;            // Atributos booleanos
input.required = true;


// ====== DATA ATTRIBUTES (data-*) ======
// HTML: <div data-id="123" data-user-name="Juan" data-is-active="true">

const card = document.querySelector('[data-id]');

// Acceso con dataset (camelCase para nombres compuestos)
const id = card.dataset.id;           // "123"
const userName = card.dataset.userName; // "Juan" (data-user-name → userName)
const isActive = card.dataset.isActive; // "true" (string!)

// Establecer data attributes
card.dataset.categoria = 'tecnologia'; // Crea data-categoria
card.dataset.itemCount = '5';

// Eliminar data attribute
delete card.dataset.isActive;

// ⚠️ Los valores siempre son strings, convertir si es necesario
const idNumero = parseInt(card.dataset.id);
const activo = card.dataset.isActive === 'true';


// ====== EJEMPLO PRÁCTICO: Cards con data attributes ======

// ====== NAVEGACIÓN POR EL DOM ======

const elemento = document.querySelector('#mi-elemento');

// ====== PADRES ======
const padre = elemento.parentElement;       // Elemento padre directo
const padreNode = elemento.parentNode;      // Nodo padre (puede ser document)

// closest - Encuentra el ancestro más cercano que coincida
const card = elemento.closest('.card');         // Busca hacia arriba
const contenedor = elemento.closest('[data-container]');

// ====== HIJOS ======
const hijos = elemento.children;            // HTMLCollection de hijos elementos
const primerHijo = elemento.firstElementChild;
const ultimoHijo = elemento.lastElementChild;
const cantidadHijos = elemento.childElementCount;

// childNodes incluye nodos de texto
const todosLosNodos = elemento.childNodes;  // Incluye texto, comentarios, etc.

// ====== HERMANOS ======
const siguiente = elemento.nextElementSibling;     // Hermano siguiente
const anterior = elemento.previousElementSibling;  // Hermano anterior

// ====== EJEMPLO PRÁCTICO: Acordeón ======
document.querySelectorAll('.accordion-header').forEach(header => {
    header.addEventListener('click', () => {
        // El contenido es el siguiente hermano
        const contenido = header.nextElementSibling;
        contenido.classList.toggle('show');
    });
});

// ====== EJEMPLO PRÁCTICO: Eliminar item de lista ======
function eliminarItem(botonEliminar) {
    // closest encuentra el li padre
    const item = botonEliminar.closest('li');
    item.remove();
}

// ====== EJEMPLO PRÁCTICO: Tabs ======
function activarTab(tabButton) {
    // Desactivar todos los hermanos
    const contenedor = tabButton.parentElement;
    contenedor.children.forEach(tab => {
        tab.classList.remove('active');
    });
    
    // Activar el clickeado
    tabButton.classList.add('active');
    
    // Mostrar contenido correspondiente
    const targetId = tabButton.dataset.target;
    const contenido = document.querySelector(targetId);
    contenido.classList.add('show');
}
