const boton = document.querySelector('#mi-boton');

// ====== FORMAS DE AGREGAR EVENTOS ======

// 1. Función anónima (más común)
boton.addEventListener('click', function() {
    console.log('¡Click!');
});

// 2. Arrow function (más corta)
boton.addEventListener('click', () => {
    console.log('¡Click con arrow!');
});

// 3. Función nombrada (necesaria para remover)
function manejarClick() {
    console.log('¡Click con función nombrada!');
}
boton.addEventListener('click', manejarClick);

// 4. Con el objeto event
boton.addEventListener('click', function(event) {
    console.log('Elemento clickeado:', event.target);
    console.log('Tipo de evento:', event.type);
});

// ====== REMOVER EVENT LISTENER ======
// ⚠️ Solo funciona con funciones nombradas
boton.removeEventListener('click', manejarClick);

// ====== OPCIONES ADICIONALES ======
boton.addEventListener('click', handler, {
    once: true,      // Se ejecuta solo UNA vez y se remueve
    capture: false,  // Fase de captura vs burbuja
    passive: true    // Mejora performance (no usa preventDefault)
});

// Evento que se ejecuta solo una vez
boton.addEventListener('click', () => {
    console.log('Solo se ve una vez');
}, { once: true });

// ====== MÚLTIPLES EVENTOS AL MISMO ELEMENTO ======
const input = document.querySelector('input');

['focus', 'blur', 'input'].forEach(evento => {
    input.addEventListener(evento, (e) => {
        console.log(`Evento: ${e.type}`);
    });
});

// ====== ❌ FORMA ANTIGUA (no recomendada) ======
// boton.onclick = function() { }; // Solo permite UN handler

// ====== preventDefault ======
// Previene el comportamiento por defecto del elemento

// Prevenir navegación de un enlace
const link = document.querySelector('a');
link.addEventListener('click', (e) => {
    e.preventDefault(); // No navega
    console.log('Click en link sin navegar');
});

// Prevenir envío de formulario
const form = document.querySelector('form');
form.addEventListener('submit', (e) => {
    e.preventDefault(); // No envía el form
    // Aquí validamos y enviamos con fetch
});

// Prevenir menú contextual
elemento.addEventListener('contextmenu', (e) => {
    e.preventDefault(); // No muestra menú del navegador
});


// ====== stopPropagation ======
// Detiene la propagación del evento (bubbling)

const padre = document.querySelector('#padre');
const hijo = document.querySelector('#hijo');

padre.addEventListener('click', () => {
    console.log('Click en padre');
});

hijo.addEventListener('click', (e) => {
    e.stopPropagation(); // El evento NO llega al padre
    console.log('Click en hijo (evento detenido aquí)');
});


// ====== stopImmediatePropagation ======
// Detiene TODOS los handlers, incluso del mismo elemento

elemento.addEventListener('click', (e) => {
    console.log('Handler 1');
    e.stopImmediatePropagation();
});

elemento.addEventListener('click', () => {
    console.log('Handler 2 - NUNCA se ejecuta');
});

2Sistema de Atajos de Teclado
// ====== SISTEMA COMPLETO DE ATAJOS ======

const shortcuts = {
    'ctrl+s': () => guardar(),
    'ctrl+n': () => nuevo(),
    'ctrl+o': () => abrir(),
    'ctrl+z': () => deshacer(),
    'ctrl+shift+z': () => rehacer(),
    'escape': () => cerrar(),
    'f1': () => mostrarAyuda(),
    'delete': () => eliminarSeleccionado()
};

document.addEventListener('keydown', (e) => {
    // Construir la combinación de teclas
    let combo = [];
    
    if (e.ctrlKey) combo.push('ctrl');
    if (e.shiftKey) combo.push('shift');
    if (e.altKey) combo.push('alt');
    combo.push(e.key.toLowerCase());
    
    const shortcutKey = combo.join('+');
    
    // Ejecutar si existe el atajo
    if (shortcuts[shortcutKey]) {
        e.preventDefault();
        shortcuts[shortcutKey]();
    }
});

// ====== BÚSQUEDA EN TIEMPO REAL ======
let searchTimeout;
const searchInput = document.querySelector('#search');

searchInput.addEventListener('keyup', (e) => {
    // Debounce: esperar que el usuario termine de escribir
    clearTimeout(searchTimeout);
    
    searchTimeout = setTimeout(() => {
        const query = e.target.value;
        if (query.length >= 3) {
            buscar(query);
        }
    }, 300); // Espera 300ms después de la última tecla
});

1Eventos de Formulario Completos

// ====== EVENTOS DE FORMULARIO ======

const form = document.querySelector('#mi-formulario');
const input = document.querySelector('#mi-input');
const select = document.querySelector('#mi-select');

// ====== SUBMIT - Envío del formulario ======
form.addEventListener('submit', (e) => {
    e.preventDefault(); // ⚠️ IMPORTANTE: Prevenir recarga
    
    // Obtener todos los datos del formulario
    const formData = new FormData(form);
    const datos = Object.fromEntries(formData);
    
    console.log('Datos del formulario:', datos);
    
    // Enviar con fetch
    fetch('/api/submit', {
        method: 'POST',
        body: formData
    });
});

// ====== INPUT - Cada cambio en tiempo real ======
input.addEventListener('input', (e) => {
    console.log('Valor actual:', e.target.value);
    // Útil para validación en tiempo real
    validarCampo(e.target);
});

// ====== CHANGE - Cuando el valor cambia y pierde foco ======
input.addEventListener('change', (e) => {
    console.log('Valor final:', e.target.value);
});

// Para SELECT es más útil 'change'
select.addEventListener('change', (e) => {
    console.log('Opción seleccionada:', e.target.value);
});

// ====== FOCUS y BLUR ======
input.addEventListener('focus', () => {
    console.log('Input enfocado');
    input.parentElement.classList.add('focused');
});

input.addEventListener('blur', () => {
    console.log('Input perdió foco');
    input.parentElement.classList.remove('focused');
    validarCampo(input); // Validar al salir
});

// ====== FOCUSIN/FOCUSOUT (con bubbling) ======
// Útil para detectar foco en cualquier hijo del form
form.addEventListener('focusin', (e) => {
    console.log('Elemento enfocado:', e.target.name);
});

// ====== RESET - Cuando se resetea el formulario ======
form.addEventListener('reset', () => {
    console.log('Formulario reseteado');
});


// ====== VALIDACIÓN COMPLETA ======
function validarFormulario(form) {
    const inputs = form.querySelectorAll('input, select, textarea');
    let esValido = true;
    
    inputs.forEach(input => {
        // Limpiar estado previo
        input.classList.remove('is-valid', 'is-invalid');
        
        // Validar requeridos
        if (input.required && !input.value.trim()) {
            input.classList.add('is-invalid');
            esValido = false;
            return;
        }
        
        // Validar email
        if (input.type === 'email' && input.value) {
            const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            if (!emailRegex.test(input.value)) {
                input.classList.add('is-invalid');
                esValido = false;
                return;
            }
        }
        
        // Campo válido
        if (input.value) {
            input.classList.add('is-valid');
        }
    });
    
    return esValido;
}

Event Delegation - Patrón Importante

¿Qué es Event Delegation?
En lugar de agregar eventos a cada elemento hijo, agregamos UN solo evento al padre y detectamos qué hijo fue clickeado. Es más eficiente y funciona con elementos dinámicos.

// El patrón más importante para elementos dinámicos

// ❌ FORMA INCORRECTA (no funciona con elementos nuevos)
document.querySelectorAll('.delete-btn').forEach(btn => {
    btn.addEventListener('click', () => {
        // Solo funciona para elementos que existían al cargar
    });
});


// ✅ FORMA CORRECTA: Event Delegation
const lista = document.querySelector('#mi-lista');

lista.addEventListener('click', (e) => {
    // Verificar si el click fue en un botón de eliminar
    if (e.target.classList.contains('delete-btn')) {
        // Encontrar el item padre y eliminarlo
        const item = e.target.closest('li');
        item.remove();
    }
    
    // También podemos detectar clics en el item
    if (e.target.matches('li') || e.target.closest('li')) {
        const item = e.target.closest('li');
        console.log('Item clickeado:', item.textContent);
    }
});


// ====== USAR matches() para selectores más complejos ======
document.addEventListener('click', (e) => {
    // matches() verifica si el elemento coincide con el selector
    if (e.target.matches('button.btn-primary')) {
        console.log('Click en botón primario');
    }
    
    if (e.target.matches('[data-action="delete"]')) {
        const id = e.target.dataset.id;
        eliminarElemento(id);
    }
});


// ====== EJEMPLO: Cards de productos ======
const productosContainer = document.querySelector('#productos');

productosContainer.addEventListener('click', (e) => {
    const card = e.target.closest('.producto-card');
    if (!card) return; // Click fuera de una card
    
    const productoId = card.dataset.productoId;
    
    // Botón agregar al carrito
    if (e.target.matches('.btn-add-cart')) {
        agregarAlCarrito(productoId);
    }
    
    // Botón favorito
    if (e.target.matches('.btn-favorite')) {
        toggleFavorito(productoId);
        e.target.classList.toggle('active');
    }
    
    // Click en la card (ver detalles)
    if (e.target.matches('.card-body, .card-img-top')) {
        verDetalles(productoId);
    }
});


// ====== EJEMPLO: Tabla con acciones ======
const tabla = document.querySelector('#tabla-usuarios');

tabla.addEventListener('click', (e) => {
    const fila = e.target.closest('tr');
    if (!fila) return;
    
    const userId = fila.dataset.userId;
    
    if (e.target.matches('.btn-edit')) {
        editarUsuario(userId);

// ====== EVENTOS DEL MOUSE ======

const elemento = document.querySelector('#mi-elemento');

// mouseenter - Entra al elemento (no hace bubble)
elemento.addEventListener('mouseenter', () => {
    elemento.classList.add('hover');
});

// mouseleave - Sale del elemento (no hace bubble)
elemento.addEventListener('mouseleave', () => {
    elemento.classList.remove('hover');
});

// mouseover/mouseout - Similar pero hace bubble (detecta hijos)
elemento.addEventListener('mouseover', (e) => {
    console.log('Mouse sobre:', e.target);
});

// mousemove - Movimiento continuo del mouse
elemento.addEventListener('mousemove', (e) => {
    // Posición relativa al elemento
    console.log('Posición:', e.offsetX, e.offsetY);
    
    // Posición relativa al viewport
    console.log('Viewport:', e.clientX, e.clientY);
    
    // Posición relativa al documento
    console.log('Documento:', e.pageX, e.pageY);
});

// mousedown/mouseup - Presionar/soltar botón
elemento.addEventListener('mousedown', () => console.log('Presionado'));
elemento.addEventListener('mouseup', () => console.log('Soltado'));

// ====== DRAG & DROP BÁSICO ======
let isDragging = false;
let offsetX, offsetY;
const draggable = document.querySelector('.draggable');

draggable.addEventListener('mousedown', (e) => {
    isDragging = true;
    offsetX = e.offsetX;
    offsetY = e.offsetY;
});

document.addEventListener('mousemove', (e) => {
    if (!isDragging) return;
    
    draggable.style.left = (e.pageX - offsetX) + 'px';
    draggable.style.top = (e.pageY - offsetY) + 'px';
});

document.addEventListener('mouseup', () => {
    isDragging = false;
});

// ====== SCROLL EVENT ======
window.addEventListener('scroll', () => {
    const scrollTop = window.scrollY || document.documentElement.scrollTop;
    console.log('Scroll:', scrollTop);
    
    // Mostrar botón "volver arriba" después de cierto scroll
    if (scrollTop > 300) {
        btnScrollTop.classList.add('visible');
    } else {
        btnScrollTop.classList.remove('visible');
    }
});

El Objeto Event

Propiedades del Objeto Event
// ====== PROPIEDADES DEL OBJETO EVENT ======

elemento.addEventListener('click', (e) => {
    // ====== PROPIEDADES PRINCIPALES ======
    
    e.type           // "click", "submit", "keydown", etc.
    e.target         // Elemento que disparó el evento (donde hizo click)
    e.currentTarget  // Elemento con el event listener (puede ser diferente)
    e.timeStamp      // Timestamp del evento
    
    // ====== DIFERENCIA target vs currentTarget ======
    // Si tienes: <div id="padre"><button>Click</button></div>
    // Y el listener está en #padre:
    // - e.target = el button (donde clickeaste)
    // - e.currentTarget = el #padre (donde está el listener)
    
    // ====== MÉTODOS ======
    
    e.preventDefault()       // Previene acción por defecto
    e.stopPropagation()      // Detiene propagación (bubbling)
    e.stopImmediatePropagation() // Detiene todos los handlers
    
    // ====== PARA EVENTOS DE MOUSE ======
    
    e.clientX, e.clientY  // Posición relativa al viewport
    e.pageX, e.pageY      // Posición relativa al documento
    e.offsetX, e.offsetY  // Posición relativa al elemento
    e.screenX, e.screenY  // Posición relativa a la pantalla
    e.button              // 0=izq, 1=medio, 2=der
    e.buttons             // Bitmask de botones presionados
    
    // ====== PARA EVENTOS DE TECLADO ======
    
    e.key        // "a", "Enter", "Escape", etc.
    e.code       // "KeyA", "Enter", "Escape", etc.
    e.keyCode    // 65, 13, 27, etc. (deprecado pero usado)
    e.shiftKey   // true si Shift presionado
    e.ctrlKey    // true si Ctrl presionado
    e.altKey     // true si Alt presionado
    e.metaKey    // true si Cmd/Win presionado
    e.repeat     // true si es repetición por mantener tecla
    
    // ====== PARA EVENTOS DE FORMULARIO ======
    
    e.target.value     // Valor del input
    e.target.checked   // Estado de checkbox
    e.target.files     // Archivos seleccionados (input file)
});

// ====== EJEMPLO PRÁCTICO: Menú contextual personalizado ======
document.addEventListener('contextmenu', (e) => {
    e.preventDefault();
    
    const menu = document.querySelector('#custom-menu');
    menu.style.left = e.pageX + 'px';
