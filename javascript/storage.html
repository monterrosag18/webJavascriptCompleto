Web Storage API

LocalStorage y SessionStorage - Almacenamiento en el navegador

LocalStorage
  • Persiste después de cerrar el navegador
  • ~5-10 MB de espacio
  • Compartido entre pestañas
  • Ideal para preferencias, carritos
SessionStorage
  • Se borra al cerrar la pestaña
  • ~5-10 MB de espacio
  • Solo en la pestaña actual
  • Ideal para datos temporales

1Métodos Básicos

// ========================================
// MÉTODOS BÁSICOS DE STORAGE
// ========================================

// Misma API para localStorage y sessionStorage

// GUARDAR - setItem(clave, valor)
localStorage.setItem('nombre', 'Ana');
localStorage.setItem('edad', '25'); // Siempre strings

// También se puede con notación de objeto
localStorage.ciudad = 'Madrid';
localStorage['pais'] = 'España';

// OBTENER - getItem(clave)
const nombre = localStorage.getItem('nombre'); // 'Ana'
const noExiste = localStorage.getItem('xyz'); // null

// Notación de objeto
const ciudad = localStorage.ciudad; // 'Madrid'

// ELIMINAR - removeItem(clave)
localStorage.removeItem('nombre');

// LIMPIAR TODO - clear()
localStorage.clear();

// OBTENER CLAVE POR ÍNDICE - key(índice)
localStorage.setItem('a', '1');
localStorage.setItem('b', '2');
localStorage.key(0); // 'a' (primera clave)

// CONTAR ITEMS - length
console.log(localStorage.length); // 2

// VERIFICAR SI EXISTE
if (localStorage.getItem('nombre') !== null) {
    console.log('Existe');
}

// Con operador nullish
const valor = localStorage.getItem('tema') ?? 'light';

    { id: 1, nombre: 'Laptop', precio: 1000 },

    { id: 3, nombre: 'Teclado', precio: 75 }
];

localStorage.setItem('productos', JSON.stringify(productos));

// RECUPERAR ARRAY

// Funciones helper recomendadas
const Storage = {
    set(key, value) {
        localStorage.setItem(key, JSON.stringify(value));
    },
    
    get(key, defaultValue = null) {
        const item = localStorage.getItem(key);
        try {
            return item ? JSON.parse(item) : defaultValue;
        } catch {
            return item ?? defaultValue;
        }
    },
    
    remove(key) {
        localStorage.removeItem(key);
    },
    
    clear() {
        localStorage.clear();
    }
};

3Ejemplo: Carrito de Compras Persistente

// ========================================
// CARRITO DE COMPRAS CON LOCALSTORAGE
// ========================================

class Carrito {
    constructor() {
        this.storageKey = 'carrito';
        this.items = this.cargar();
    }
    
    // Cargar desde localStorage
    cargar() {
        const data = localStorage.getItem(this.storageKey);
        return data ? JSON.parse(data) : [];
    }
    
    // Guardar en localStorage
    guardar() {
        localStorage.setItem(this.storageKey, JSON.stringify(this.items));
    }
    
    // Agregar producto
    agregar(producto) {
        const existe = this.items.find(item => item.id === producto.id);
        
        if (existe) {
            existe.cantidad++;
        } else {
            this.items.push({ ...producto, cantidad: 1 });
        }
        
        this.guardar();
        return this;
    }
    
    // Eliminar producto
    eliminar(productoId) {
        this.items = this.items.filter(item => item.id !== productoId);
        this.guardar();
        return this;
    }
    
    // Actualizar cantidad
    actualizarCantidad(productoId, cantidad) {
        const item = this.items.find(i => i.id === productoId);
        if (item) {
            item.cantidad = Math.max(0, cantidad);
            if (item.cantidad === 0) this.eliminar(productoId);
            else this.guardar();
        }
        return this;
    }
    
    // Calcular total
    get total() {
        return this.items.reduce((sum, item) => 
            sum + (item.precio * item.cantidad), 0);
    }
    
    // Contar items
    get cantidadTotal() {
        return this.items.reduce((sum, item) => sum + item.cantidad, 0);
    }
    
    // Vaciar carrito
    vaciar() {
        this.items = [];
        this.guardar();
        return this;
    }
}

// Uso
const carrito = new Carrito();

carrito.agregar({ id: 1, nombre: 'Laptop', precio: 1000 });
carrito.agregar({ id: 2, nombre: 'Mouse', precio: 25 });
carrito.agregar({ id: 1, nombre: 'Laptop', precio: 1000 }); // +1 cantidad

console.log(carrito.items);
console.log('Total: $' + carrito.total);
console.log('Items:', carrito.cantidadTotal);

4Ejemplo: Preferencias de Usuario

// ========================================
// SISTEMA DE PREFERENCIAS
// ========================================

const Preferencias = {
    defaults: {
        tema: 'light',
        idioma: 'es',
        fontSize: 16,
        notificaciones: true,
        sidebar: 'expanded'
    },
    
    getAll() {
        const saved = localStorage.getItem('preferencias');
        return saved 
            ? { ...this.defaults, ...JSON.parse(saved) }
            : { ...this.defaults };
    },
    
    get(key) {
        return this.getAll()[key];
    },
    
    set(key, value) {
        const prefs = this.getAll();
        prefs[key] = value;
        localStorage.setItem('preferencias', JSON.stringify(prefs));
        this.aplicar(key, value);
    },
    
    aplicar(key, value) {
        switch(key) {
            case 'tema':
                document.body.className = `theme-${value}`;
                break;
            case 'fontSize':
                document.documentElement.style.fontSize = `${value}px`;
                break;
        }
    },
    
    init() {
        const prefs = this.getAll();
        Object.entries(prefs).forEach(([k, v]) => this.aplicar(k, v));
    },
    
    reset() {
        localStorage.removeItem('preferencias');
        this.init();
    }
};

// Uso
Preferencias.init(); // Al cargar la página

// Cambiar tema
Preferencias.set('tema', 'dark');

// Obtener idioma
const idioma = Preferencias.get('idioma');

    // Actualizar UI basado en el cambio
    if (event.key === 'carrito') {
        actualizarContadorCarrito();
    }
    
    if (event.key === 'tema') {

const AuthSync = {
    login(userData) {
        localStorage.setItem('usuario', JSON.stringify(userData));
        localStorage.setItem('loginTime', Date.now().toString());
    },
    
    logout() {
        localStorage.removeItem('usuario');
        localStorage.removeItem('loginTime');
        // Esto disparará el evento en otras pestañas
    },
    
    isLoggedIn() {
        return localStorage.getItem('usuario') !== null;
    }
};

6Storage con Expiración (TTL)

// ========================================
// STORAGE CON TIEMPO DE EXPIRACIÓN
// ========================================

const StorageWithExpiry = {
    set(key, value, ttlMinutes) {
        const item = {
            value,
            expiry: ttlMinutes ? Date.now() + (ttlMinutes * 60 * 1000) : null
        };
        localStorage.setItem(key, JSON.stringify(item));
    },
    
    get(key) {
        const itemStr = localStorage.getItem(key);
        
        if (!itemStr) return null;
        
        try {
            const item = JSON.parse(itemStr);
            
            // Si no tiene expiración o no ha expirado
            if (!item.expiry || Date.now() < item.expiry) {
                return item.value;
            }
            
            // Expirado - eliminar y retornar null
            localStorage.removeItem(key);
            return null;
            
        } catch {
            return null;
        }
    },
    
    isExpired(key) {
        const itemStr = localStorage.getItem(key);
        if (!itemStr) return true;
        
        try {
            const { expiry } = JSON.parse(itemStr);
            return expiry && Date.now() >= expiry;
        } catch {
            return true;
        }
    },
    
    getTimeRemaining(key) {
        const itemStr = localStorage.getItem(key);
        if (!itemStr) return 0;
        
        try {
            const { expiry } = JSON.parse(itemStr);
            if (!expiry) return Infinity;
            return Math.max(0, expiry - Date.now());
        } catch {
            return 0;
        }
    }
};

// Uso
// Guardar token que expira en 60 minutos
StorageWithExpiry.set('authToken', 'abc123', 60);

// Cache de API que expira en 5 minutos
StorageWithExpiry.set('usersCache', usersData, 5);

// Obtener (retorna null si expiró)
const token = StorageWithExpiry.get('authToken');

if (!token) {
    console.log('Token expirado, iniciar sesión');
}
