1Crear Arrays

🔹 Diferentes formas de crear arrays
// ========================================
// 1. CREAR ARRAYS - Múltiples Formas
// ========================================

// Forma 1: Array literal (más común)
const frutas = ['manzana', 'banana', 'naranja'];

// Forma 2: Constructor Array
const numeros = new Array(1, 2, 3, 4, 5);

// Forma 3: Array vacío con tamaño
const vacio = new Array(5); // [empty × 5]

// Forma 4: Array.of() - crea array con elementos
const arrayOf = Array.of(1, 2, 3);

// Forma 5: Array.from() - desde iterables
const fromString = Array.from('Hola'); // ['H', 'o', 'l', 'a']
const fromSet = Array.from(new Set([1, 2, 2, 3])); // [1, 2, 3]

// Forma 6: Array.from() con función de mapeo
const pares = Array.from({ length: 5 }, (_, i) => i * 2);
console.log(pares); // [0, 2, 4, 6, 8]

// Forma 7: fill() para llenar array
const lleno = new Array(3).fill('x'); // ['x', 'x', 'x']

// Arrays mixtos (cualquier tipo de dato)
const mixto = [1, 'texto', true, null, { a: 1 }, [1, 2]];

// Arrays anidados (matrices)
const matriz = [
    [1, 2, 3],
    [4, 5, 6],
    [7, 8, 9]
];
console.log(matriz[1][2]); // 6

2Acceder y Modificar Elementos

// ========================================
// 2. ACCEDER Y MODIFICAR ELEMENTOS
// ========================================

const colores = ['rojo', 'verde', 'azul', 'amarillo'];

// Acceder por índice (0-based)
console.log(colores[0]); // 'rojo' (primero)
console.log(colores[2]); // 'azul' (tercero)

// Acceder al último elemento
console.log(colores[colores.length - 1]); // 'amarillo'

// at() - permite índices negativos (ES2022)
console.log(colores.at(-1)); // 'amarillo' (último)
console.log(colores.at(-2)); // 'azul' (penúltimo)

// Modificar elemento
colores[1] = 'verde oscuro';
console.log(colores); // ['rojo', 'verde oscuro', 'azul', 'amarillo']

// Agregar elemento en posición específica
colores[10] = 'negro'; // Crea huecos vacíos
console.log(colores.length); // 11

// Propiedad length
console.log(colores.length); // Número de elementos

// Truncar array modificando length
const nums = [1, 2, 3, 4, 5];
nums.length = 3;
console.log(nums); // [1, 2, 3]

// Verificar si es un array
console.log(Array.isArray(colores)); // true
console.log(Array.isArray('texto')); // false

3Métodos para Agregar/Eliminar

¡Importante! Estos métodos modifican el array original (mutadores).
push() Mutador

Agrega elementos al final

const arr = [1, 2, 3];

// Agregar un elemento
arr.push(4);
console.log(arr); // [1, 2, 3, 4]

// Agregar múltiples
arr.push(5, 6, 7);
console.log(arr); // [1, 2, 3, 4, 5, 6, 7]

// Retorna la nueva longitud
const len = arr.push(8);
console.log(len); // 8
pop() Mutador

Elimina el último elemento

const arr = [1, 2, 3, 4];

// Eliminar último
const ultimo = arr.pop();
console.log(ultimo); // 4
console.log(arr); // [1, 2, 3]

// pop() en array vacío
const vacio = [];
console.log(vacio.pop()); // undefined
unshift() Mutador

Agrega elementos al inicio

const arr = [3, 4, 5];

// Agregar al inicio
arr.unshift(1, 2);
console.log(arr); // [1, 2, 3, 4, 5]

// Retorna nueva longitud
const len = arr.unshift(0);
console.log(len); // 6
shift() Mutador

Elimina el primer elemento

const arr = [1, 2, 3, 4];

// Eliminar primero
const primero = arr.shift();
console.log(primero); // 1
console.log(arr); // [2, 3, 4]
splice() Mutador

El método más versátil: agrega, elimina y reemplaza

// ========================================
// splice(inicio, cantidadEliminar, ...elementosAgregar)
// ========================================

let arr = ['a', 'b', 'c', 'd', 'e'];

// 1. ELIMINAR elementos
let eliminados = arr.splice(2, 2); // Desde índice 2, eliminar 2
console.log(eliminados); // ['c', 'd']
console.log(arr); // ['a', 'b', 'e']

// 2. INSERTAR elementos (sin eliminar)
arr = ['a', 'b', 'e'];
arr.splice(2, 0, 'c', 'd'); // En índice 2, eliminar 0, insertar 'c','d'
console.log(arr); // ['a', 'b', 'c', 'd', 'e']

// 3. REEMPLAZAR elementos
arr = ['a', 'b', 'c', 'd', 'e'];
arr.splice(1, 2, 'X', 'Y', 'Z'); // Eliminar 2, insertar 3
console.log(arr); // ['a', 'X', 'Y', 'Z', 'd', 'e']

// 4. Índices negativos (desde el final)
arr = [1, 2, 3, 4, 5];
arr.splice(-2, 1); // Eliminar 1 elemento desde el penúltimo
console.log(arr); // [1, 2, 3, 5]

4Métodos de Búsqueda

// ========================================
// MÉTODOS DE BÚSQUEDA
// ========================================

const numeros = [10, 20, 30, 40, 20, 50];
const usuarios = [
    { id: 1, nombre: 'Ana', edad: 25 },
    { id: 2, nombre: 'Carlos', edad: 30 },
    { id: 3, nombre: 'Elena', edad: 28 }
];

// indexOf() - Busca valor, retorna índice o -1
console.log(numeros.indexOf(20)); // 1 (primera ocurrencia)
console.log(numeros.indexOf(100)); // -1 (no encontrado)
console.log(numeros.indexOf(20, 2)); // 4 (buscar desde índice 2)

// lastIndexOf() - Busca desde el final
console.log(numeros.lastIndexOf(20)); // 4

// includes() - Verifica si existe (boolean)
console.log(numeros.includes(30)); // true
console.log(numeros.includes(100)); // false

// find() - Busca con función, retorna ELEMENTO
const encontrado = usuarios.find(u => u.edad > 26);
console.log(encontrado); // { id: 2, nombre: 'Carlos', edad: 30 }

// findIndex() - Busca con función, retorna ÍNDICE
const indice = usuarios.findIndex(u => u.nombre === 'Elena');
console.log(indice); // 2

// findLast() - Busca desde el final (ES2023)
const ultimoMayor = usuarios.findLast(u => u.edad > 26);
console.log(ultimoMayor); // { id: 3, nombre: 'Elena', edad: 28 }

// findLastIndex() - Índice desde el final (ES2023)
const ultimoIndice = usuarios.findLastIndex(u => u.edad > 26);
console.log(ultimoIndice); // 2

// some() - ¿Alguno cumple la condición?
console.log(usuarios.some(u => u.edad >= 30)); // true

// every() - ¿Todos cumplen la condición?
console.log(usuarios.every(u => u.edad >= 18)); // true
console.log(usuarios.every(u => u.edad >= 30)); // false

5Métodos de Iteración

// ========================================
// MÉTODOS DE ITERACIÓN
// ========================================

const productos = [
    { nombre: 'Laptop', precio: 1000, stock: 5 },
    { nombre: 'Mouse', precio: 25, stock: 50 },
    { nombre: 'Teclado', precio: 75, stock: 30 },
    { nombre: 'Monitor', precio: 300, stock: 10 }
];

// forEach() - Ejecuta función por cada elemento
productos.forEach((producto, index) => {
    console.log(`${index + 1}. ${producto.nombre}: $${producto.precio}`);
});

// map() - Transforma cada elemento, retorna nuevo array
const nombres = productos.map(p => p.nombre);
console.log(nombres); // ['Laptop', 'Mouse', 'Teclado', 'Monitor']

const conIVA = productos.map(p => ({
    ...p,
    precioConIVA: p.precio * 1.21
}));

// filter() - Filtra elementos, retorna nuevo array
const caros = productos.filter(p => p.precio > 100);
console.log(caros); // [Laptop, Monitor]

const conStock = productos.filter(p => p.stock > 20);

// reduce() - Reduce a un solo valor
const total = productos.reduce((suma, p) => suma + p.precio, 0);
console.log(total); // 1400

const stockTotal = productos.reduce((acc, p) => acc + p.stock, 0);
console.log(stockTotal); // 95

// reduce() para crear objeto
const porNombre = productos.reduce((obj, p) => {
    obj[p.nombre] = p.precio;
    return obj;
}, {});
console.log(porNombre); // { Laptop: 1000, Mouse: 25, ... }

// reduceRight() - Reduce desde el final
const concatenado = ['a', 'b', 'c'].reduceRight((acc, v) => acc + v, '');
console.log(concatenado); // 'cba'

// Encadenar métodos
const resultado = productos
    .filter(p => p.precio > 50)
    .map(p => p.nombre)
    .join(', ');
console.log(resultado); // 'Laptop, Teclado, Monitor'

6Métodos de Transformación

// ========================================
// MÉTODOS DE TRANSFORMACIÓN
// ========================================

// slice() - Extrae porción (NO modifica original)
const arr = [1, 2, 3, 4, 5];
console.log(arr.slice(1, 4)); // [2, 3, 4] (índice 1 a 3)
console.log(arr.slice(2)); // [3, 4, 5] (desde índice 2)
console.log(arr.slice(-2)); // [4, 5] (últimos 2)
console.log(arr.slice()); // [1,2,3,4,5] (copia completa)

// concat() - Combina arrays
const a = [1, 2];
const b = [3, 4];
const c = [5, 6];
console.log(a.concat(b, c)); // [1, 2, 3, 4, 5, 6]

// join() - Convierte a string
const letras = ['H', 'o', 'l', 'a'];
console.log(letras.join('')); // 'Hola'
console.log(letras.join('-')); // 'H-o-l-a'
console.log(letras.join()); // 'H,o,l,a' (coma por defecto)

// reverse() - Invierte (MODIFICA original)
const nums = [1, 2, 3];
nums.reverse();
console.log(nums); // [3, 2, 1]

// toReversed() - Invierte sin modificar (ES2023)
const original = [1, 2, 3];
const invertido = original.toReversed();
console.log(original); // [1, 2, 3] (sin cambios)
console.log(invertido); // [3, 2, 1]

// sort() - Ordena (MODIFICA original)
const nombres = ['Carlos', 'Ana', 'Elena'];
nombres.sort();
console.log(nombres); // ['Ana', 'Carlos', 'Elena']

// ¡Cuidado! sort() ordena como strings por defecto
const numeros = [10, 2, 30, 1];
numeros.sort();
console.log(numeros); // [1, 10, 2, 30] ¡Incorrecto!

// Ordenar números correctamente
numeros.sort((a, b) => a - b); // Ascendente
console.log(numeros); // [1, 2, 10, 30]

numeros.sort((a, b) => b - a); // Descendente
console.log(numeros); // [30, 10, 2, 1]

// toSorted() - Ordena sin modificar (ES2023)
const orig = [3, 1, 2];
const ordenado = orig.toSorted((a, b) => a - b);

// flat() - Aplana arrays anidados
const anidado = [1, [2, 3], [4, [5, 6]]];
console.log(anidado.flat()); // [1, 2, 3, 4, [5, 6]]
console.log(anidado.flat(2)); // [1, 2, 3, 4, 5, 6]
console.log(anidado.flat(Infinity)); // Aplana todo

// flatMap() - map() + flat(1)
const frases = ['Hola Mundo', 'JavaScript ES6'];
const palabras = frases.flatMap(f => f.split(' '));
console.log(palabras); // ['Hola', 'Mundo', 'JavaScript', 'ES6']

7Spread y Destructuring

// ========================================
// SPREAD OPERATOR (...)
// ========================================

// Copiar array
const original = [1, 2, 3];
const copia = [...original];

// Combinar arrays
const arr1 = [1, 2];
const arr2 = [3, 4];
const combinado = [...arr1, ...arr2]; // [1, 2, 3, 4]

// Agregar elementos
const conMas = [0, ...arr1, 5]; // [0, 1, 2, 5]

// Pasar como argumentos
const nums = [5, 2, 8, 1];
console.log(Math.max(...nums)); // 8

// ========================================
// DESTRUCTURING DE ARRAYS
// ========================================

const colores = ['rojo', 'verde', 'azul', 'amarillo'];

// Destructuring básico
const [primero, segundo] = colores;
console.log(primero); // 'rojo'
console.log(segundo); // 'verde'

// Saltar elementos
const [, , tercero] = colores;
console.log(tercero); // 'azul'

// Rest pattern (resto de elementos)
const [head, ...tail] = colores;
console.log(head); // 'rojo'
console.log(tail); // ['verde', 'azul', 'amarillo']

// Valores por defecto
const [a, b, c, d, e = 'blanco'] = colores;
console.log(e); // 'blanco' (no existe índice 4)

// Intercambiar variables
let x = 1, y = 2;
[x, y] = [y, x];
console.log(x, y); // 2, 1

// Destructuring en funciones
function procesar([primero, ...resto]) {
    console.log('Primero:', primero);
    console.log('Resto:', resto);
}
procesar([1, 2, 3, 4]);

// Destructuring anidado
const matriz = [[1, 2], [3, 4]];
const [[a1, a2], [b1, b2]] = matriz;
console.log(a1, a2, b1, b2); // 1 2 3 4

const calculadora = {

const usuario = {

// ========================================
// MÉTODOS DEL OBJETO
// ========================================

const cuenta = {
    titular: 'Juan',
    saldo: 1000,
    
    // Método usando this
    depositar(cantidad) {
        this.saldo += cantidad;
        console.log(`Depositado: $${cantidad}. Nuevo saldo: $${this.saldo}`);
    },
    
    retirar(cantidad) {
        if (cantidad <= this.saldo) {
            this.saldo -= cantidad;
            console.log(`Retirado: $${cantidad}. Nuevo saldo: $${this.saldo}`);
        } else {
            console.log('Saldo insuficiente');
        }
    },
    
    // Getter
    get info() {
        return `${this.titular}: $${this.saldo}`;
    },
    
    // Setter
    set nuevoSaldo(valor) {
        if (valor >= 0) {
            this.saldo = valor;
        }
    }
};

cuenta.depositar(500);
console.log(cuenta.info); // Getter: 'Juan: $1500'
cuenta.nuevoSaldo = 2000; // Setter

10Object Methods (Métodos Estáticos)

// ========================================
// MÉTODOS ESTÁTICOS DE OBJECT
// ========================================

const persona = {
    nombre: 'Ana',
    edad: 25,
    ciudad: 'Madrid'
};

// Object.keys() - Array de claves
console.log(Object.keys(persona)); 
// ['nombre', 'edad', 'ciudad']

// Object.values() - Array de valores
console.log(Object.values(persona)); 
// ['Ana', 25, 'Madrid']

// Object.entries() - Array de [clave, valor]
console.log(Object.entries(persona));
// [['nombre', 'Ana'], ['edad', 25], ['ciudad', 'Madrid']]

// Iterar con entries
for (const [key, value] of Object.entries(persona)) {
    console.log(`${key}: ${value}`);
}

// Object.fromEntries() - De entries a objeto
const entries = [['a', 1], ['b', 2]];
const obj = Object.fromEntries(entries);
console.log(obj); // { a: 1, b: 2 }

// Object.assign() - Copiar/Combinar objetos
const target = { a: 1, b: 2 };
const source = { b: 3, c: 4 };
const resultado = Object.assign(target, source);
console.log(resultado); // { a: 1, b: 3, c: 4 }

// Clonar objeto (shallow copy)
const clon = Object.assign({}, persona);

// Object.freeze() - Inmutable (no modifica)
const config = Object.freeze({
    apiUrl: 'https://api.example.com',
    timeout: 5000
});
config.timeout = 10000; // No tiene efecto
console.log(Object.isFrozen(config)); // true

// Object.seal() - No agregar/eliminar, pero sí modificar
const sealed = Object.seal({ x: 1 });
sealed.x = 2; // OK
sealed.y = 3; // No tiene efecto
console.log(Object.isSealed(sealed)); // true

// Object.getOwnPropertyNames()
console.log(Object.getOwnPropertyNames(persona));
// ['nombre', 'edad', 'ciudad']

// Object.getOwnPropertyDescriptor()
console.log(Object.getOwnPropertyDescriptor(persona, 'nombre'));
// { value: 'Ana', writable: true, enumerable: true, configurable: true }

// Object.defineProperty() - Definir con opciones
Object.defineProperty(persona, 'id', {
    value: 123,
    writable: false,
    enumerable: true
});

11Clases ES6

// ========================================
// CLASES EN JAVASCRIPT (ES6+)
// ========================================

class Usuario {
    // Propiedades privadas (ES2022)
    #password;
    
    // Propiedades públicas
    rol = 'user';
    
    // Propiedad estática
    static contador = 0;
    
    // Constructor
    constructor(nombre, email, password) {
        this.nombre = nombre;
        this.email = email;
        this.#password = password;
        this.id = ++Usuario.contador;
    }
    
    // Método público
    saludar() {
        return `Hola, soy ${this.nombre}`;
    }
    
    // Método privado (ES2022)
    #encriptarPassword() {
        return this.#password.split('').reverse().join('');
    }
    
    // Getter
    get info() {
        return `${this.nombre} (${this.email})`;
    }
    
    // Setter
    set nuevoEmail(email) {
        if (email.includes('@')) {
            this.email = email;
        }
    }
    
    // Método estático
    static crearAnonimo() {
        return new Usuario('Anónimo', 'anon@example.com', '12345');
    }
    
    // Verificar password
    verificarPassword(password) {
        return this.#password === password;
    }
}

// Crear instancias
const usuario1 = new Usuario('Ana', 'ana@email.com', 'secret123');
const usuario2 = new Usuario('Carlos', 'carlos@email.com', 'pass456');

console.log(usuario1.saludar()); // 'Hola, soy Ana'
console.log(usuario1.info); // Getter
console.log(Usuario.contador); // 2

// Método estático
const anonimo = Usuario.crearAnonimo();

// Usar setter
usuario1.nuevoEmail = 'nueva@email.com';

// Verificar tipo
console.log(usuario1 instanceof Usuario); // true

12Herencia

// ========================================
// HERENCIA CON EXTENDS
// ========================================

// Clase base (padre)
class Animal {
    constructor(nombre, edad) {
        this.nombre = nombre;
        this.edad = edad;
    }
    
    hablar() {
        console.log('El animal hace un sonido');
    }
    
    presentarse() {
        return `Soy ${this.nombre}, tengo ${this.edad} años`;
    }
}

// Clase hija (hereda de Animal)
class Perro extends Animal {
    constructor(nombre, edad, raza) {
        // super() llama al constructor del padre
        super(nombre, edad);
        this.raza = raza;
    }
    
    // Sobrescribir método (override)
    hablar() {
        console.log('¡Guau guau!');
    }
    
    // Llamar método del padre con super
    presentarseCompleto() {
        return `${super.presentarse()} y soy un ${this.raza}`;
    }
    
    // Nuevo método
    buscarPelota() {
        console.log(`${this.nombre} va a buscar la pelota`);
    }
}

class Gato extends Animal {
    constructor(nombre, edad, color) {
        super(nombre, edad);
        this.color = color;
    }
    
    hablar() {
        console.log('¡Miau!');
    }
    
    ronronear() {
        console.log(`${this.nombre} está ronroneando`);
    }
}

// Usar las clases
const max = new Perro('Max', 3, 'Labrador');
const michi = new Gato('Michi', 2, 'naranja');

max.hablar(); // '¡Guau guau!'
michi.hablar(); // '¡Miau!'

console.log(max.presentarseCompleto());
// 'Soy Max, tengo 3 años y soy un Labrador'

// Verificar herencia
console.log(max instanceof Perro); // true
console.log(max instanceof Animal); // true
console.log(max instanceof Gato); // false

        cp: '28001'

};

console.log(jsonString);
// '{"nombre":"Ana","edad":25,"activo":true,...}'

// Con formato (indentación)
const jsonFormateado = JSON.stringify(usuario, null, 2);
console.log(jsonFormateado);
/*
{
  "nombre": "Ana",
  "edad": 25,
  ...
}
*/

// Con replacer (filtrar propiedades)
const parcial = JSON.stringify(usuario, ['nombre', 'email']);
// Solo incluye nombre

// Replacer función
const filtrado = JSON.stringify(usuario, (key, value) => {
    if (key === 'edad') return undefined; // Omitir
    return value;
});

// JSON.parse() - JSON string a Objeto
const jsonData = '{"nombre":"Carlos","edad":30}';
const objeto = JSON.parse(jsonData);
console.log(objeto.nombre); // 'Carlos'

// Con reviver (transformar valores)
const conFecha = '{"fecha":"2024-01-15T10:30:00.000Z"}';
const parseado = JSON.parse(conFecha, (key, value) => {
    if (key === 'fecha') return new Date(value);
    return value;
});
console.log(parseado.fecha instanceof Date); // true

// Clonar objeto profundo (deep clone)
const clonProfundo = JSON.parse(JSON.stringify(usuario));

// Limitaciones de JSON
const objComplejo = {
    fecha: new Date(),
    funcion: () => 'hola',
    indefinido: undefined,
    simbolo: Symbol('id'),
    regex: /test/gi
};

// JSON NO soporta: Date, funciones, undefined, Symbol, RegExp
console.log(JSON.stringify(objComplejo));
// Solo convierte fecha a string, ignora el resto

// EJEMPLO: CARRITO DE COMPRAS COMPLETO
// ========================================

class CarritoCompras {
    #items = [];
    
    // Agregar producto
    agregar(producto, cantidad = 1) {
        const existente = this.#items.find(item => item.id === producto.id);
        
        if (existente) {
            existente.cantidad += cantidad;
        } else {
            this.#items.push({
                ...producto,
                cantidad
            });
        }
    }
    
    // Eliminar producto
    eliminar(productoId) {
        this.#items = this.#items.filter(item => item.id !== productoId);
    }
    
    // Actualizar cantidad
    actualizarCantidad(productoId, cantidad) {
        const item = this.#items.find(i => i.id === productoId);
        if (item) {
            item.cantidad = Math.max(0, cantidad);
            if (item.cantidad === 0) this.eliminar(productoId);
        }
    }
    
    // Calcular subtotal
    get subtotal() {
        return this.#items.reduce((sum, item) => 
            sum + (item.precio * item.cantidad), 0
        );
    }
    
    // Calcular IVA
    get iva() {
        return this.subtotal * 0.21;
    }
    
    // Total con IVA
    get total() {
        return this.subtotal + this.iva;
    }
    
    // Cantidad de items
    get cantidadItems() {
        return this.#items.reduce((sum, item) => sum + item.cantidad, 0);
    }
    
    // Obtener items
    get items() {
        return [...this.#items];
    }
    
    // Vaciar carrito
    vaciar() {
        this.#items = [];
    }
    
    // Aplicar descuento
    aplicarDescuento(porcentaje) {
        return this.total * (1 - porcentaje / 100);
    }
    
    // Resumen del carrito
    resumen() {
        console.log('=== CARRITO ===');
        this.#items.forEach(item => {
            console.log(`${item.nombre} x${item.cantidad} = $${item.precio * item.cantidad}`);
        });
        console.log(`Subtotal: $${this.subtotal.toFixed(2)}`);
        console.log(`IVA (21%): $${this.iva.toFixed(2)}`);
        console.log(`TOTAL: $${this.total.toFixed(2)}`);
    }
}

// Uso del carrito
const carrito = new CarritoCompras();

carrito.agregar({ id: 1, nombre: 'Laptop', precio: 1000 });
carrito.agregar({ id: 2, nombre: 'Mouse', precio: 25 }, 2);
carrito.agregar({ id: 1, nombre: 'Laptop', precio: 1000 }); // +1

carrito.resumen();
console.log('Con 10% descuento: $' + carrito.aplicarDescuento(10).toFixed(2));

// EJEMPLO: GESTIÓN DE USUARIOS
// ========================================

class GestorUsuarios {
    static STORAGE_KEY = 'usuarios_app';
    
    // Cargar usuarios de localStorage
    static cargar() {
        const data = localStorage.getItem(this.STORAGE_KEY);
        return data ? JSON.parse(data) : [];
    }
    
    // Guardar en localStorage
    static guardar(usuarios) {
        localStorage.setItem(this.STORAGE_KEY, JSON.stringify(usuarios));
    }
    
    // Crear usuario
    static crear(nombre, email) {
        const usuarios = this.cargar();
        
        // Verificar email único
        if (usuarios.some(u => u.email === email)) {
            throw new Error('El email ya existe');
        }
        
        const nuevoUsuario = {
            id: Date.now(),
            nombre,
            email,
            creado: new Date().toISOString(),
            activo: true
        };
        
        usuarios.push(nuevoUsuario);
        this.guardar(usuarios);
        return nuevoUsuario;
    }
    
    // Obtener todos
    static obtenerTodos() {
        return this.cargar();
    }
    
    // Buscar por ID
    static buscarPorId(id) {
        return this.cargar().find(u => u.id === id);
    }
    
    // Buscar por email
    static buscarPorEmail(email) {
        return this.cargar().find(u => u.email === email);
    }
    
    // Actualizar
    static actualizar(id, datos) {
        const usuarios = this.cargar();
        const index = usuarios.findIndex(u => u.id === id);
        
        if (index === -1) {
            throw new Error('Usuario no encontrado');
        }
        
        usuarios[index] = { ...usuarios[index], ...datos };
        this.guardar(usuarios);
        return usuarios[index];
    }
    
    // Eliminar
    static eliminar(id) {
        const usuarios = this.cargar().filter(u => u.id !== id);
        this.guardar(usuarios);
    }
    
    // Filtrar activos
    static obtenerActivos() {
        return this.cargar().filter(u => u.activo);
    }
    
    // Buscar por nombre (parcial)
    static buscar(termino) {
        return this.cargar().filter(u => 
            u.nombre.toLowerCase().includes(termino.toLowerCase())
        );
    }
}

// Uso
try {
    GestorUsuarios.crear('Ana García', 'ana@email.com');
    GestorUsuarios.crear('Carlos López', 'carlos@email.com');
    
    console.log(GestorUsuarios.obtenerTodos());
    console.log(GestorUsuarios.buscar('ana'));
} catch (error) {
    console.error(error.message);
}

// EJEMPLO: ANÁLISIS DE VENTAS
// ========================================

const ventas = [
    { id: 1, producto: 'Laptop', cantidad: 5, precio: 1000, categoria: 'Electrónica', fecha: '2024-01-15' },
    { id: 2, producto: 'Mouse', cantidad: 20, precio: 25, categoria: 'Electrónica', fecha: '2024-01-16' },
    { id: 3, producto: 'Silla', cantidad: 3, precio: 200, categoria: 'Muebles', fecha: '2024-01-17' },
    { id: 4, producto: 'Escritorio', cantidad: 2, precio: 350, categoria: 'Muebles', fecha: '2024-01-18' },
    { id: 5, producto: 'Teclado', cantidad: 15, precio: 75, categoria: 'Electrónica', fecha: '2024-01-19' }
];

// 1. Total de ventas
const totalVentas = ventas.reduce((sum, v) => sum + (v.precio * v.cantidad), 0);
console.log('Total ventas: $' + totalVentas); // $7825

// 2. Ventas por categoría
const ventasPorCategoria = ventas.reduce((acc, v) => {
    const total = v.precio * v.cantidad;
    acc[v.categoria] = (acc[v.categoria] || 0) + total;
    return acc;
}, {});
console.log(ventasPorCategoria);
// { Electrónica: 6625, Muebles: 1200 }

// 3. Producto más vendido (por cantidad)
const masVendido = ventas.reduce((max, v) => 
    v.cantidad > max.cantidad ? v : max
);
console.log('Más vendido:', masVendido.producto);

// 4. Promedio de precio
const promedioPrecio = ventas.reduce((sum, v) => sum + v.precio, 0) / ventas.length;
console.log('Precio promedio: $' + promedioPrecio);

// 5. Productos con precio > promedio
const sobrePromedio = ventas.filter(v => v.precio > promedioPrecio);
console.log('Sobre promedio:', sobrePromedio.map(v => v.producto));

// 6. Ordenar por total de venta
const ordenadoPorTotal = [...ventas]
    .map(v => ({ ...v, total: v.precio * v.cantidad }))
    .sort((a, b) => b.total - a.total);
console.log(ordenadoPorTotal);

// 7. Agrupar por fecha
const porFecha = ventas.reduce((acc, v) => {
    if (!acc[v.fecha]) acc[v.fecha] = [];
    acc[v.fecha].push(v);
    return acc;
}, {});

// 8. Resumen completo
const resumen = {
    totalVentas,
    cantidadProductos: ventas.length,
    unidadesVendidas: ventas.reduce((sum, v) => sum + v.cantidad, 0),
    promedioPorVenta: totalVentas / ventas.length,
    categorias: [...new Set(ventas.map(v => v.categoria))],
    productoMasCaro: ventas.reduce((max, v) => v.precio > max.precio ? v : max).producto,
    productoMasBarato: ventas.reduce((min, v) => v.precio < min.precio ? v : min).producto
};
console.log(resumen);
