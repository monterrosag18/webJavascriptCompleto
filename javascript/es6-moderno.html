ES6 let y const

Nuevas formas de declarar variables con block scope.

// ========================================
// VAR vs LET vs CONST
// ========================================

// var - Function scope, hoisting
var x = 1;
var x = 2; // ✓ Redeclaración permitida
x = 3;     // ✓ Reasignación permitida

// let - Block scope, sin hoisting
let y = 1;
// let y = 2; // ✗ Error: ya declarada
y = 3;     // ✓ Reasignación permitida

// const - Block scope, sin reasignación
const z = 1;
// z = 2;    // ✗ Error: no reasignable

// const con objetos/arrays - el contenido SÍ puede cambiar
const usuario = { nombre: 'Ana' };
usuario.nombre = 'María'; // ✓ Esto SÍ funciona
usuario.edad = 25;        // ✓ Añadir propiedades
// usuario = {}; // ✗ Error: no reasignable

const numeros = [1, 2, 3];
numeros.push(4);  // ✓ Modificar array
numeros[0] = 10; // ✓ Cambiar elementos
// numeros = []; // ✗ Error

// Block Scope
if (true) {
    var varVariable = 'var';
    let letVariable = 'let';
    const constVariable = 'const';
}
console.log(varVariable);   // 'var' - accesible
// console.log(letVariable);   // Error: not defined
// console.log(constVariable); // Error: not defined

// Temporal Dead Zone
console.log(a); // undefined (hoisting de var)
// console.log(b); // Error: Cannot access before initialization
var a = 1;
let b = 2;

ES6 Template Literals 

// ========================================
// TEMPLATE LITERALS (Backticks `)
// ========================================

const nombre = 'Ana';
const edad = 25;

// Antes (ES5)
const mensajeViejo = 'Hola, me llamo ' + nombre + ' y tengo ' + edad + ' años.';

// Ahora (ES6+)
const mensaje = `Hola, me llamo ${nombre} y tengo ${edad} años.`;

// Expresiones en interpolación
const precio = 100;
const descuento = 0.2;
const total = `Total: $${precio * (1 - descuento)}`; // Total: $80

// Strings multilínea
const html = `
<div class="card">
    <h2>${nombre}</h2>
    <p>Edad: ${edad}</p>
</div>
`;

// Llamar funciones
const mayusculas = `Nombre: ${nombre.toUpperCase()}`;

// Condicionales
const estado = `Usuario ${edad >= 18 ? 'adulto' : 'menor'}`;

// Tagged Templates (avanzado)
function destacar(strings, ...values) {
    return strings.reduce((result, str, i) => {
        const value = values[i] ? `<strong>${values[i]}</strong>` : '';
        return result + str + value;
    }, '');
}

const resultado = destacar`Mi nombre es ${nombre} y tengo ${edad} años`;
// "Mi nombre es <strong>Ana</strong> y tengo <strong>25</strong> años"

ES6 Arrow Functions 

// ========================================
// ARROW FUNCTIONS (=>)
// ========================================

// Sintaxis tradicional
function sumar(a, b) {
    return a + b;
}

// Arrow function
const sumarArrow = (a, b) => {
    return a + b;
};

// Return implícito (sin llaves)
const sumarCorta = (a, b) => a + b;

// Un solo parámetro (sin paréntesis)
const doble = x => x * 2;

// Sin parámetros
const saludo = () => 'Hola!';

// Retornar objeto (con paréntesis)
const crearUsuario = (nombre, edad) => ({ nombre, edad });

// Con arrays
const numeros = [1, 2, 3, 4, 5];
const dobles = numeros.map(n => n * 2);
const pares = numeros.filter(n => n % 2 === 0);
const suma = numeros.reduce((acc, n) => acc + n, 0);

// this en Arrow Functions
// Las arrow functions NO tienen su propio this

const objeto = {
    nombre: 'Objeto',
    
    // Función tradicional - this es el objeto
    metodoTradicional() {
        console.log(this.nombre); // 'Objeto'
    },
    
    // Arrow - this es el contexto donde se definió
    metodoArrow: () => {
        console.log(this.nombre); // undefined (window)
    },
    
    // Útil en callbacks
    iniciarTimer() {
        // Arrow mantiene el this del método
        setTimeout(() => {
            console.log(this.nombre); // 'Objeto'
        }, 1000);
    }
};

// Cuándo NO usar arrow functions:
// - Métodos de objeto que necesitan this
// - Event handlers del DOM
// - Constructores (no tienen prototype)

ES6 Destructuring 

// ========================================
// DESTRUCTURING - Extraer valores
// ========================================

// DESTRUCTURING DE ARRAYS
const colores = ['rojo', 'verde', 'azul'];

// Extraer elementos
const [primero, segundo, tercero] = colores;
console.log(primero); // 'rojo'

// Omitir elementos
const [, , ultimo] = colores;
console.log(ultimo); // 'azul'

// Valores por defecto
const [a, b, c, d = 'blanco'] = colores;
console.log(d); // 'blanco'

// Rest operator
const [head, ...tail] = colores;
console.log(tail); // ['verde', 'azul']

// Intercambiar variables
let x = 1, y = 2;
[x, y] = [y, x];
console.log(x, y); // 2, 1

// DESTRUCTURING DE OBJETOS
const usuario = {
    nombre: 'Ana',
    edad: 25,
    email: 'ana@email.com',
    direccion: {
        ciudad: 'Madrid',
        pais: 'España'
    }
};

// Extraer propiedades
const { nombre, edad } = usuario;
console.log(nombre, edad); // 'Ana', 25

// Renombrar variables
const { nombre: nombreUsuario, edad: edadUsuario } = usuario;
console.log(nombreUsuario); // 'Ana'

// Valores por defecto
const { telefono = 'No disponible' } = usuario;
console.log(telefono); // 'No disponible'

// Destructuring anidado
const { direccion: { ciudad, pais } } = usuario;
console.log(ciudad, pais); // 'Madrid', 'España'

// En parámetros de función
function saludar({ nombre, edad = 0 }) {
    console.log(`Hola ${nombre}, tienes ${edad} años`);
}
saludar(usuario);

// Rest en objetos
const { email, ...resto } = usuario;
console.log(resto); // { nombre: 'Ana', edad: 25, direccion: {...} }

ES6 Spread Operator (...) 

// ========================================
// SPREAD OPERATOR (...) - Expandir
// ========================================

// ARRAYS
const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];

// Concatenar arrays
const combinado = [...arr1, ...arr2]; // [1, 2, 3, 4, 5, 6]

// Copiar array (shallow copy)
const copia = [...arr1]; // [1, 2, 3]

// Insertar elementos
const conCero = [0, ...arr1]; // [0, 1, 2, 3]
const conMedio = [1, 2, ...[2.5], 3]; // [1, 2, 2.5, 3]

// En llamadas a funciones
const numeros = [5, 2, 8, 1, 9];
const maximo = Math.max(...numeros); // 9

// OBJETOS (ES2018)
const usuario = { nombre: 'Ana', edad: 25 };
const direccion = { ciudad: 'Madrid', pais: 'España' };

// Combinar objetos
const completo = { ...usuario, ...direccion };
// { nombre: 'Ana', edad: 25, ciudad: 'Madrid', pais: 'España' }

// Copiar objeto (shallow copy)
const copiaUsuario = { ...usuario };

// Sobrescribir propiedades
const actualizado = { ...usuario, edad: 26 };
// { nombre: 'Ana', edad: 26 }

// Añadir propiedades
const conEmail = { ...usuario, email: 'ana@email.com' };

// REST vs SPREAD
// REST: Recoger en un array/objeto
const [primero, ...resto] = [1, 2, 3]; // resto = [2, 3]
const { nombre: n, ...otrosDatos } = usuario;

// SPREAD: Expandir un array/objeto
const nuevo = [...arr1]; // Expandir

ES6 Classes 

// ========================================
// CLASSES ES6+
// ========================================

class Persona {
    // Propiedades de clase (ES2022)
    tipo = 'humano';
    
    // Propiedades privadas (ES2022)
    #id;
    
    // Constructor
    constructor(nombre, edad) {
        this.nombre = nombre;
        this.edad = edad;
        this.#id = Math.random().toString(36).substr(2);
    }
    
    // Métodos
    saludar() {
        console.log(`Hola, soy ${this.nombre}`);
    }
    
    // Getters y Setters
    get info() {
        return `${this.nombre} (${this.edad})`;
    }
    
    set nuevaEdad(valor) {
        if (valor > 0) this.edad = valor;
    }
    
    // Método privado (ES2022)
    #generarToken() {
        return `token_${this.#id}`;
    }
    
    getToken() {
        return this.#generarToken();
    }
    
    // Métodos estáticos
    static crear(nombre, edad) {
        return new Persona(nombre, edad);
    }
    
    static especie = 'Homo Sapiens';
}

// Uso
const ana = new Persona('Ana', 25);
ana.saludar();           // 'Hola, soy Ana'
console.log(ana.info);    // 'Ana (25)'
ana.nuevaEdad = 26;
console.log(Persona.especie); // 'Homo Sapiens'

// HERENCIA
class Empleado extends Persona {
    constructor(nombre, edad, puesto) {
        super(nombre, edad); // Llamar constructor padre
        this.puesto = puesto;
    }
    
    // Override
    saludar() {
        super.saludar(); // Llamar método padre
        console.log(`Trabajo como ${this.puesto}`);
    }
}

const emp = new Empleado('Carlos', 30, 'Developer');
emp.saludar();
// 'Hola, soy Carlos'
// 'Trabajo como Developer'

ES2020 Optional Chaining & Nullish Coalescing 

// ========================================
// OPTIONAL CHAINING (?.) - ES2020
// ========================================

const usuario = {
    nombre: 'Ana',
    direccion: {
        ciudad: 'Madrid'
    }
};

// Sin optional chaining (antes)
const calle = usuario && usuario.direccion && usuario.direccion.calle;

// Con optional chaining
const calleSegura = usuario?.direccion?.calle; // undefined (no error)
const ciudad = usuario?.direccion?.ciudad; // 'Madrid'

// Con arrays
const items = usuario?.pedidos?.[0]; // undefined

// Con funciones
const resultado = usuario?.metodoOpcional?.(); // undefined

// ========================================
// NULLISH COALESCING (??) - ES2020
// ========================================

// ?? solo considera null/undefined
// || considera todos los falsy (0, '', false, null, undefined)

const valor1 = null ?? 'default';  // 'default'
const valor2 = undefined ?? 'default'; // 'default'
const valor3 = 0 ?? 'default';     // 0 (0 es válido)
const valor4 = '' ?? 'default';    // '' (string vacío es válido)
const valor5 = false ?? 'default'; // false (false es válido)

// Comparar con ||
const a = 0 || 'default';  // 'default' (0 es falsy)
const b = 0 ?? 'default';  // 0

// Combinando ambos
const config = {
    timeout: 0,
    retries: null
};

const timeout = config?.timeout ?? 5000; // 0 (el valor)
const retries = config?.retries ?? 3;    // 3 (default)

// Nullish assignment (ES2021)
let x = null;
x ??= 'default'; // x = 'default'

let y = 0;
y ??= 10; // y = 0 (no cambia porque 0 no es null/undefined)

ES2022+ Nuevos Métodos de Array 

// ========================================
// NUEVOS MÉTODOS DE ARRAY
// ========================================

const arr = [1, 2, 3, 4, 5];

// at() - ES2022 - Índices negativos
arr.at(0);   // 1
arr.at(-1);  // 5 (último)
arr.at(-2);  // 4 (penúltimo)

// findLast() / findLastIndex() - ES2023
const numeros = [1, 2, 3, 4, 5, 4, 3];
numeros.findLast(n => n > 3);        // 4 (el último mayor a 3)
numeros.findLastIndex(n => n > 3);   // 5 (índice)

// toReversed() - ES2023 - Inmutable
const original = [1, 2, 3];
const reversed = original.toReversed(); // [3, 2, 1]
console.log(original); // [1, 2, 3] (no cambia)

// toSorted() - ES2023 - Inmutable
const nums = [3, 1, 4, 1, 5];
const sorted = nums.toSorted((a, b) => a - b); // [1, 1, 3, 4, 5]
console.log(nums); // [3, 1, 4, 1, 5] (no cambia)

// toSpliced() - ES2023 - Inmutable
const items = ['a', 'b', 'c', 'd'];
const spliced = items.toSpliced(1, 2, 'x', 'y'); // ['a', 'x', 'y', 'd']
console.log(items); // ['a', 'b', 'c', 'd'] (no cambia)

// with() - ES2023 - Inmutable
const colores = ['rojo', 'verde', 'azul'];
const nuevo = colores.with(1, 'amarillo'); // ['rojo', 'amarillo', 'azul']
console.log(colores); // ['rojo', 'verde', 'azul'] (no cambia)

// flat() / flatMap() - ES2019
const anidado = [[1, 2], [3, [4, 5]]];
anidado.flat();      // [1, 2, 3, [4, 5]]
anidado.flat(2);     // [1, 2, 3, 4, 5]
anidado.flat(Infinity); // [1, 2, 3, 4, 5]

[1, 2, 3].flatMap(n => [n, n * 2]); // [1, 2, 2, 4, 3, 6]

// includes() - ES2016
[1, 2, 3].includes(2); // true
[1, 2, NaN].includes(NaN); // true (indexOf falla con NaN)

ES2017+ Métodos de Object 

// ========================================
// MÉTODOS DE OBJECT
// ========================================

const usuario = { nombre: 'Ana', edad: 25, ciudad: 'Madrid' };

// Object.keys() - ES5
Object.keys(usuario); // ['nombre', 'edad', 'ciudad']

// Object.values() - ES2017
Object.values(usuario); // ['Ana', 25, 'Madrid']

// Object.entries() - ES2017
Object.entries(usuario); 
// [['nombre', 'Ana'], ['edad', 25], ['ciudad', 'Madrid']]

// Object.fromEntries() - ES2019
const entries = [['a', 1], ['b', 2]];
Object.fromEntries(entries); // { a: 1, b: 2 }

// Convertir Map a Object
const map = new Map([['x', 1], ['y', 2]]);
Object.fromEntries(map); // { x: 1, y: 2 }

// Transformar objeto
const precios = { manzana: 1, naranja: 2, pera: 3 };
const preciosDobles = Object.fromEntries(
    Object.entries(precios).map(([k, v]) => [k, v * 2])
);
// { manzana: 2, naranja: 4, pera: 6 }

// Object.assign() - ES6
const target = { a: 1 };
const source = { b: 2 };
Object.assign(target, source); // target = { a: 1, b: 2 }

// Object.hasOwn() - ES2022 (reemplaza hasOwnProperty)
Object.hasOwn(usuario, 'nombre'); // true
Object.hasOwn(usuario, 'toString'); // false

// Object.groupBy() - ES2024
const personas = [
    { nombre: 'Ana', edad: 25 },
    { nombre: 'Juan', edad: 30 },
    { nombre: 'María', edad: 25 }
];

const porEdad = Object.groupBy(personas, p => p.edad);
// { 25: [{...}, {...}], 30: [{...}] }

ES6 Map y Set 

// ========================================
// MAP - Pares clave-valor (cualquier tipo de clave)
// ========================================

const map = new Map();

// Añadir
map.set('nombre', 'Ana');
map.set(1, 'uno');
map.set({ id: 1 }, 'objeto como clave');

// Obtener
map.get('nombre'); // 'Ana'
map.get(1);        // 'uno'

// Verificar
map.has('nombre'); // true
map.size;          // 3

// Eliminar
map.delete('nombre');
map.clear(); // Eliminar todo

// Inicializar con valores
const usuarios = new Map([
    ['u1', { nombre: 'Ana' }],
    ['u2', { nombre: 'Juan' }]
]);

// Iterar
for (const [clave, valor] of map) {
    console.log(clave, valor);
}

map.forEach((valor, clave) => console.log(clave, valor));

// ========================================
// SET - Valores únicos
// ========================================

const set = new Set();

// Añadir (ignora duplicados)
set.add(1);
set.add(2);
set.add(1); // Ignorado

// Verificar
set.has(1);  // true
set.size;    // 2

// Eliminar duplicados de array
const arrDuplicados = [1, 2, 2, 3, 3, 3];
const unicos = [...new Set(arrDuplicados)]; // [1, 2, 3]

// Operaciones de conjuntos
const a = new Set([1, 2, 3]);
const b = new Set([2, 3, 4]);

// Unión
const union = new Set([...a, ...b]); // {1, 2, 3, 4}

// Intersección
const interseccion = new Set([...a].filter(x => b.has(x))); // {2, 3}

// Diferencia
const diferencia = new Set([...a].filter(x => !b.has(x))); // {1}
