Singleton

Creacional

Garantiza que solo exista una instancia de una clase y proporciona un punto de acceso global.

// ========================================
// PATRÓN SINGLETON
// ========================================

// Implementación con Clase
class Database {
    static instance = null;
    
    constructor() {
        if (Database.instance) {
            return Database.instance;
        }
        
        this.connection = null;
        this.config = {};
        Database.instance = this;
    }
    
    connect(config) {
        if (!this.connection) {
            this.config = config;
            this.connection = `Connected to ${config.host}`;
            console.log('Nueva conexión creada');
        }
        return this.connection;
    }
    
    query(sql) {
        if (!this.connection) {
            throw new Error('No hay conexión');
        }
        console.log(`Ejecutando: ${sql}`);
    }
    
    static getInstance() {
        if (!Database.instance) {
            Database.instance = new Database();
        }
        return Database.instance;
    }
}

// Uso
const db1 = new Database();
const db2 = new Database();
console.log(db1 === db2); // true - Misma instancia

// Singleton con Module Pattern
const AppState = (() => {
    let instance;
    
    function createInstance() {
        return {
            user: null,
            theme: 'light',
            setUser(user) { this.user = user; },
            setTheme(theme) { this.theme = theme; }
        };
    }
    
    return {
        getInstance() {
            if (!instance) {
                instance = createInstance();
            }
            return instance;
        }
    };
})();

const state1 = AppState.getInstance();
const state2 = AppState.getInstance();
console.log(state1 === state2); // true

Factory

Creacional

Proporciona una interfaz para crear objetos sin especificar sus clases concretas.

// ========================================
// PATRÓN FACTORY
// ========================================

// Clases de Productos
class Button {
    constructor(text) {
        this.text = text;
    }
    render() {
        return `<button>${this.text}</button>`;
    }
}

class Input {
    constructor(type, placeholder) {
        this.type = type;
        this.placeholder = placeholder;
    }
    render() {
        return `<input type="${this.type}" placeholder="${this.placeholder}">`;
    }
}

class Select {
    constructor(options) {
        this.options = options;
    }
    render() {
        const opts = this.options.map(o => `<option>${o}</option>`).join('');
        return `<select>${opts}</select>`;
    }
}

// Factory
class UIFactory {
    static create(type, config) {
        switch (type) {
            case 'button':
                return new Button(config.text);
            case 'input':
                return new Input(config.type, config.placeholder);
            case 'select':
                return new Select(config.options);
            default:
                throw new Error(`Tipo ${type} no soportado`);
        }
    }
}

// Uso
const btn = UIFactory.create('button', { text: 'Click me' });
const input = UIFactory.create('input', { type: 'email', placeholder: 'Email' });
const select = UIFactory.create('select', { options: ['A', 'B', 'C'] });

console.log(btn.render());
console.log(input.render());
console.log(select.render());

// Factory con registro dinámico
class ComponentFactory {
    static components = new Map();
    
    static register(name, componentClass) {
        this.components.set(name, componentClass);
    }
    
    static create(name, ...args) {
        const Component = this.components.get(name);
        if (!Component) {
            throw new Error(`Componente ${name} no registrado`);
        }
        return new Component(...args);
    }
}

ComponentFactory.register('button', Button);
ComponentFactory.register('input', Input);

Module Pattern

Estructural

Encapsula código con variables y métodos privados, exponiendo solo una API pública.

// ========================================
// PATRÓN MODULE (IIFE)
// ========================================

const Calculator = (() => {
    // Variables privadas
    let result = 0;
    const history = [];
    
    // Funciones privadas
    function saveToHistory(operation) {
        history.push({
            operation,
            result,
            timestamp: new Date()
        });
    }
    
    // API Pública
    return {
        add(n) {
            result += n;
            saveToHistory(`+${n}`);
            return this;
        },
        
        subtract(n) {
            result -= n;
            saveToHistory(`-${n}`);
            return this;
        },
        
        multiply(n) {
            result *= n;
            saveToHistory(`*${n}`);
            return this;
        },
        
        getResult() {
            return result;
        },
        
        getHistory() {
            return [...history]; // Copia para no exponer original
        },
        
        reset() {
            result = 0;
            return this;
        }
    };
})();

// Uso con chaining
Calculator.add(10).multiply(2).subtract(5);
console.log(Calculator.getResult()); // 15
console.log(Calculator.getHistory());

// Revealing Module Pattern
const UserService = (() => {
    let users = [];
    
    function fetchUsers() {
        return fetch('/api/users').then(r => r.json());
    }
    
    function getById(id) {
        return users.find(u => u.id === id);
    }
    
    function add(user) {
        users.push({ ...user, id: Date.now() });
    }
    
    function remove(id) {
        users = users.filter(u => u.id !== id);
    }
    
    function getAll() {
        return [...users];
    }
    
    // Exponer solo lo necesario
    return {
        fetch: fetchUsers,
        get: getById,
        add,
        remove,
        list: getAll
    };
})();

Observer / Pub-Sub

Comportamiento

Define una dependencia uno a muchos. Cuando un objeto cambia de estado, notifica a todos sus observadores.

// ========================================
// PATRÓN OBSERVER
// ========================================

class EventEmitter {
    constructor() {
        this.events = new Map();
    }
    
    // Suscribirse a un evento
    on(event, callback) {
        if (!this.events.has(event)) {
            this.events.set(event, []);
        }
        this.events.get(event).push(callback);
        
        // Retornar función para desuscribirse
        return () => this.off(event, callback);
    }
    
    // Suscripción que se ejecuta solo una vez
    once(event, callback) {
        const wrapper = (...args) => {
            callback(...args);
            this.off(event, wrapper);
        };
        this.on(event, wrapper);
    }
    
    // Desuscribirse
    off(event, callback) {
        if (this.events.has(event)) {
            const callbacks = this.events.get(event);
            const index = callbacks.indexOf(callback);
            if (index > -1) callbacks.splice(index, 1);
        }
    }
    
    // Emitir evento
    emit(event, ...args) {
        if (this.events.has(event)) {
            this.events.get(event).forEach(cb => cb(...args));
        }
    }
}

// Ejemplo: Sistema de notificaciones
class Store extends EventEmitter {
    constructor() {
        super();
        this.cart = [];
    }
    
    addToCart(product) {
        this.cart.push(product);
        this.emit('cart:updated', this.cart);
        this.emit('product:added', product);
    }
    
    removeFromCart(id) {
        this.cart = this.cart.filter(p => p.id !== id);
        this.emit('cart:updated', this.cart);
    }
}

// Uso
const store = new Store();

// Suscribirse a cambios
store.on('cart:updated', (cart) => {
    console.log('Carrito actualizado:', cart.length, 'items');
    updateCartUI(cart);
});

store.on('product:added', (product) => {
    console.log('Producto añadido:', product.name);
    showNotification(`${product.name} añadido al carrito`);
});

store.addToCart({ id: 1, name: 'Laptop', price: 999 });

Strategy

Comportamiento

Define una familia de algoritmos intercambiables en tiempo de ejecución.

// ========================================
// PATRÓN STRATEGY
// ========================================

// Estrategias de pago
const paymentStrategies = {
    creditCard: {
        pay(amount, details) {
            console.log(`Pagando $${amount} con tarjeta ${details.cardNumber}`);
            return { success: true, method: 'credit_card' };
        },
        validate(details) {
            return details.cardNumber && details.cvv;
        }
    },
    
    paypal: {
        pay(amount, details) {
            console.log(`Pagando $${amount} con PayPal (${details.email})`);
            return { success: true, method: 'paypal' };
        },
        validate(details) {
            return details.email && details.password;
        }
    },
    
    bankTransfer: {
        pay(amount, details) {
            console.log(`Transferencia de $${amount} a cuenta ${details.accountNumber}`);
            return { success: true, method: 'bank_transfer', pending: true };
        },
        validate(details) {
            return details.accountNumber;
        }
    }
};

// Contexto que usa las estrategias
class PaymentProcessor {
    constructor() {
        this.strategy = null;
    }
    
    setStrategy(strategyName) {
        this.strategy = paymentStrategies[strategyName];
        if (!this.strategy) {
            throw new Error(`Estrategia ${strategyName} no soportada`);
        }
    }
    
    process(amount, details) {
        if (!this.strategy) {
            throw new Error('Selecciona un método de pago');
        }
        
        if (!this.strategy.validate(details)) {
            throw new Error('Datos de pago inválidos');
        }
        
        return this.strategy.pay(amount, details);
    }
}

// Uso
const processor = new PaymentProcessor();

processor.setStrategy('creditCard');
processor.process(100, { cardNumber: '4111111111111111', cvv: '123' });

processor.setStrategy('paypal');
processor.process(50, { email: 'user@email.com', password: '***' });

// Strategy con funciones
const sortStrategies = {
    price: (a, b) => a.price - b.price,
    name: (a, b) => a.name.localeCompare(b.name),
    date: (a, b) => new Date(b.date) - new Date(a.date),
    rating: (a, b) => b.rating - a.rating
};

function sortProducts(products, strategyName) {
    return [...products].sort(sortStrategies[strategyName]);
}

Decorator

Estructural

Añade funcionalidad a objetos de forma dinámica sin modificar su estructura.

// ========================================
// PATRÓN DECORATOR
// ========================================

// Clase base
class Coffee {
    constructor() {
        this.description = 'Café simple';
        this.cost = 2;
    }
    
    getDescription() { return this.description; }
    getCost() { return this.cost; }
}

// Decoradores
class MilkDecorator {
    constructor(coffee) {
        this.coffee = coffee;
    }
    
    getDescription() {
        return this.coffee.getDescription() + ' + Leche';
    }
    
    getCost() {
        return this.coffee.getCost() + 0.5;
    }
}

class SugarDecorator {
    constructor(coffee) {
        this.coffee = coffee;
    }
    
    getDescription() {
        return this.coffee.getDescription() + ' + Azúcar';
    }
    
    getCost() {
        return this.coffee.getCost() + 0.2;
    }
}

// Uso
let miCafe = new Coffee();
console.log(miCafe.getDescription(), '$' + miCafe.getCost());
// Café simple $2

miCafe = new MilkDecorator(miCafe);
console.log(miCafe.getDescription(), '$' + miCafe.getCost());
// Café simple + Leche $2.5

miCafe = new SugarDecorator(miCafe);
console.log(miCafe.getDescription(), '$' + miCafe.getCost());
// Café simple + Leche + Azúcar $2.7

// Decorator con funciones (más común en JS)
function withLogging(fn) {
    return function(...args) {
        console.log(`Llamando ${fn.name} con:`, args);
        const result = fn(...args);
        console.log(`Resultado:`, result);
        return result;
    };
}

function withTiming(fn) {
    return function(...args) {
        const start = performance.now();
        const result = fn(...args);
        console.log(`Tiempo: ${(performance.now() - start).toFixed(2)}ms`);
        return result;
    };
}

const sumar = (a, b) => a + b;
const sumarConLogging = withLogging(withTiming(sumar));
sumarConLogging(5, 3);

Proxy

Estructural

Proporciona un intermediario para controlar el acceso a un objeto.

// ========================================
// PATRÓN PROXY (JavaScript nativo)
// ========================================

// Proxy de validación
const usuario = { nombre: 'Ana', edad: 25 };

const usuarioProxy = new Proxy(usuario, {
    get(target, prop) {
        console.log(`Accediendo a ${prop}`);
        return target[prop];
    },
    
    set(target, prop, value) {
        if (prop === 'edad' && (typeof value !== 'number' || value < 0)) {
            throw new Error('Edad debe ser número positivo');
        }
        console.log(`Cambiando ${prop} a ${value}`);
        target[prop] = value;
        return true;
    }
});

usuarioProxy.nombre;      // "Accediendo a nombre"
usuarioProxy.edad = 26;   // "Cambiando edad a 26"
// usuarioProxy.edad = -5; // Error!

// Proxy reactivo (como Vue.js)
function reactive(obj, onChange) {
    return new Proxy(obj, {
        set(target, prop, value) {
            const oldValue = target[prop];
            target[prop] = value;
            onChange(prop, value, oldValue);
            return true;
        }
    });
}

const state = reactive({ count: 0 }, (prop, newVal, oldVal) => {
    console.log(`${prop}: ${oldVal} → ${newVal}`);
    updateUI();
});

state.count++; // "count: 0 → 1", actualiza UI

// Proxy con caché
function createCachedAPI(api) {
    const cache = new Map();
    
    return new Proxy(api, {
        async get(target, prop) {
            if (typeof target[prop] === 'function') {
                return async (...args) => {
                    const key = `${prop}:${JSON.stringify(args)}`;
                    
                    if (cache.has(key)) {
                        console.log('Retornando del caché');
                        return cache.get(key);
                    }
                    
                    const result = await target[prop](...args);
                    cache.set(key, result);
                    return result;
                };
            }
            return target[prop];
        }
    });
}
